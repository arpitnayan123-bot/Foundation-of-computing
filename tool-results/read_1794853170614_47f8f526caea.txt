   459→export function DictLenses() {
   460→  const mono = "ui-monospace, SFMono-Regular, Menlo, monospace";
   461→  const lenses = [
   462→    {
   463→      method: ".items()",

   467→      fill: "#c2410c",

   469→    {
   470→      method: ".keys()",
   471→      gives: "just the labels (the default)",

   480→      out: ["python", "c"],

   482→    },
   483→  ];
