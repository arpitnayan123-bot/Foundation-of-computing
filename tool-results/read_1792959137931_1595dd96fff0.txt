   122→  const progressRef = useRef<HTMLDivElement>(null);
   123→  const activeRef = useRef("");
   124→
   125→  useEffect(() => {
   126→    let ticking = false;
   127→    const update = () => {
   128→      ticking = false;
   129→      const doc = document.documentElement;
   130→      const total = doc.scrollHeight - doc.clientHeight;
   131→      const pct = total > 0 ? (doc.scrollTop / total) * 100 : 0;
   132→      if (progressRef.current) progressRef.current.style.width = `${pct}%`;
   133→

   135→      const probe = window.innerHeight * 0.32;

   138→        const el = document.getElementById(item.id);
   139→        if (el && el.getBoundingClientRect().top <= probe) current = item.id;
   140→      }
   141→      if (current !== activeRef.current) {
   142→        activeRef.current = current;
   143→        setActive(current);
   144→      }
   145→    };

   147→      if (!ticking) {
   148→        ticking = true;
   149→        requestAnimationFrame(update);
   150→      }
   151→    };
   152→    update();
   153→    window.addEventListener("scroll", onScroll, { passive: true });
   154→    window.addEventListener("resize", onScroll);
   155→    return () => {
   156→      window.removeEventListener("scroll", onScroll);
   157→      window.removeEventListener("resize", onScroll);
   158→    };
   159→  }, []);

   161→  // shared Python runtime status

   176→        setMobileOpen(false);
   177→      }
   178→    };

   180→    document.addEventListener("keydown", onKey);
   181→    return () => {
   182→      document.removeEventListener("mousedown", onDown);
   183→      document.removeEventListener("keydown", onKey);
   184→    };
   185→  }, []);
   186→
   187→  // lock body scroll while the mobile index is open

   192→    };

   217→          >

   267→                        className="absolute left-1/2 top-full z-50 w-[min(26rem,calc(100vw-2rem))] -translate-x-1/2 pt-2"

   331→                              >
   332→                                <PenLine className="h-3.5 w-3.5" />
   333→                              </span>
   334→                              <span className="min-w-0">

   336→                                  Test my knowledge

   339→                                  {EXAM_BANKS[group.id].questions.length} exam-style questions · pen-paper mode
