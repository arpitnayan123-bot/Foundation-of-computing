    25→/* ============================================================
    26→   Reveal — scroll-triggered fade/slide
    27→   ============================================================ */
    28→export function Reveal({
    29→  children,
    30→  delay = 0,
    31→  className = "",
    32→}: {
    33→  children: React.ReactNode;
    34→  delay?: number;
    35→  className?: string;
    36→}) {
    37→  const ref = useRef<HTMLDivElement>(null);
    38→  const [seen, setSeen] = useState(false);
    39→  useEffect(() => {
    40→    const el = ref.current;
    41→    if (!el) return;
    42→    const io = new IntersectionObserver(
    43→      ([e]) => {
    44→        if (e.isIntersecting) {
    45→          setSeen(true);
    46→          io.disconnect();
    47→        }
    48→      },
    49→      { threshold: 0.1, rootMargin: "0px 0px -36px 0px" }
    50→    );
    51→    io.observe(el);
    52→    return () => io.disconnect();
    53→  }, []);
    54→  return (
    55→    <div
    56→      ref={ref}
    57→      className={`reveal ${seen ? "reveal-in" : ""} ${className}`}
    58→      style={{ transitionDelay: `${delay}ms` }}
    59→    >
    60→      {children}
    61→    </div>
    62→  );
    63→}

   183→/* ============================================================
   184→   Python syntax highlighter (tiny, warm palette)
   185→   ============================================================ */
   186→const PY_KEYWORDS = new Set([
   187→  "False","None","True","and","as","assert","break","class","continue","def",
   188→  "del","elif","else","except","finally","for","from","global","if","import",
   189→  "in","is","lambda","nonlocal","not","or","pass","raise","return","try",
   190→  "while","with","yield",
   191→]);
   192→const PY_BUILTINS = new Set([
   193→  "print","len","range","list","sorted","min","max","sum","int","str","float",
   194→  "input","append","insert","pop","remove","sort","reverse","title","upper",
   195→  "lower","strip","lstrip","rstrip","keys","values","items","get","set",
   196→  "dict","tuple","enumerate","zip","abs","round","type","clear","copy","count","index",
   197→]);
   198→
   199→type Tok = { text: string; cls: string };
   200→
   201→function tokenizePython(code: string): Tok[][] {
   202→  const pattern =
   203→    /(#.*$)|("""[\s\S]*?"""|'''[\s\S]*?'''|"(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*')|\b(\d[\d_]*\.?\d*)\b|([A-Za-z_][A-Za-z0-9_]*)|(\s+)|([^\sA-Za-z0-9_]+)/gm;
   204→  const lines: Tok[][] = [[]];
   205→  let m: RegExpExecArray | null;
   206→  while ((m = pattern.exec(code)) !== null) {
   207→    const [full, comment, str, num, word, ws, sym] = m;
   208→    if (comment !== undefined) push(comment, "tk-c");
   209→    else if (str !== undefined) push(str, "tk-s");
   210→    else if (num !== undefined) push(num, "tk-n");
   211→    else if (word !== undefined) {
   212→      if (PY_KEYWORDS.has(word)) push(word, "tk-k");
   213→      else if (PY_BUILTINS.has(word)) push(word, "tk-b");
   214→      else push(word, "");
   215→    } else if (ws !== undefined) push(ws, "ws");
   216→    else if (sym !== undefined) push(sym, "tk-d");
   217→  }
   218→  function push(text: string, cls: string) {
   219→    const parts = text.split("\n");
   220→    parts.forEach((p, i) => {
   221→      if (i > 0) lines.push([]);
   222→      if (p) lines[lines.length - 1].push({ text: p, cls: cls === "ws" ? "" : cls });
   223→    });
   224→  }
   225→  return lines;
   226→}
   227→
   228→function HighlightedCode({ code }: { code: string }) {
   229→  const lines = tokenizePython(code);
   230→  return (
   231→    <pre className="code-body">
   232→      {lines.map((toks, i) => (
   233→        <div key={i} className="flex">
   234→          <span className="mr-4 w-6 shrink-0 select-none text-right text-[#6d5236]">
   235→            {i + 1}
   236→          </span>
   237→          <span className="flex-1">
   238→            {toks.length === 0 ? " " : null}
   239→            {toks.map((t, j) =>
   240→              t.cls ? (
   241→                <span key={j} className={t.cls}>
   242→                  {t.text}
   243→                </span>
   244→              ) : (
   245→                <React.Fragment key={j}>{t.text}</React.Fragment>
   246→              )
   247→            )}
   248→          </span>
   249→        </div>
   250→      ))}
   251→    </pre>
   252→  );
   253→}

   255→/* ============================================================
   256→   CodeBlock — syntax-highlighted snippet with an optional LIVE
   257→   Python runtime: ▶ Run executes the code in the browser (Pyodide),
   258→   ✎ lets the learner edit the code, ⟲ restores the original.
   259→   `pre`   → hidden Python executed before the visible code
   260→   `files` → virtual .py modules written before the run
   261→   ============================================================ */
   262→type RunPhase = "idle" | "running" | "done" | "error";
   263→
   264→export function CodeBlock({
   265→  title = "main.py",
   266→  code,
   267→  output,
   268→  note,
   269→  className = "",
   270→  runnable = true,
   271→  pre,
   272→  files,
   273→  dark = false,
   274→}: {
   275→  title?: string;
   276→  code: string;
   277→  output?: string;
   278→  note?: React.ReactNode;
   279→  className?: string;
   280→  runnable?: boolean;
   281→  pre?: string;
   282→  files?: Record<string, string>;
   283→  dark?: boolean;
   284→}) {
   285→  const [current, setCurrent] = useState(code);
   286→  const [editing, setEditing] = useState(false);
   287→  const [phase, setPhase] = useState<RunPhase>("idle");
   288→  const [bootMsg, setBootMsg] = useState("");
   289→  const [out, setOut] = useState("");
   290→  const [err, setErr] = useState("");
   291→  const [elapsed, setElapsed] = useState(0);
   292→  const [copied, setCopied] = useState(false);
   293→  const termRef = useRef<HTMLPreElement>(null);
   294→  const dirty = current !== code;
   295→  const live = phase !== "idle";
   296→  const busy = phase === "running";
   297→
   298→  useEffect(() => {
   299→    if (termRef.current) termRef.current.scrollTop = termRef.current.scrollHeight;
   300→  }, [out, err]);
   301→
   302→  async function run() {
   303→    if (busy) return;
   304→    setEditing(false);
   305→    setPhase("running");
   306→    setOut("");
   307→    setErr("");
   308→    setBootMsg("");
   309→    setElapsed(0);
   310→    const t0 = performance.now();
   311→    let py: PyodideLike | null = null;
   312→    try {
   313→      py = await getPyodide((m) => setBootMsg(m));
   314→      const pyr = py;
   315→      setBootMsg("");
   316→      let lastOut = "";
   317→      // capped push: a print flood can't grow the buffer unbounded
   318→      const push = makeCappedPush((full) => {
   319→        lastOut = full;
   320→        setOut(full);
   321→      });
   322→      py.setStdout({ batched: push });
   323→      py.setStderr({ batched: push });
   324→      py.setStdin({
   325→        stdin: () => {
   326→          try {
   327→            return window.prompt("Python input() is asking for a value — cancel = stop the program:");
   328→          } catch {
   329→            return null; // prompt blocked (e.g. sandboxed iframe) → treat as EOF
   330→          }
   331→        },
   332→      });
   333→      // Fresh module cache + virtual files for import demos.
   334→      await py.runPythonAsync(PURGE_USER_MODULES);
   335→      if (files) {
   336→        for (const [name, content] of Object.entries(files)) py.FS.writeFile(name, content);
   337→      }
   338→      const wf = current.match(/^%%writefile\s+([^\s#]+)/);
   339→      if (wf) {
   340→        // Notebook magic: save the cell as a real file inside the runtime.
   341→        const name = wf[1];
   342→        const body = current.slice(current.indexOf("\n") + 1).replace(/^\n+/, "");
   343→        py.FS.writeFile(name, body);
   344→        push(`✓ saved ${name} (${body.split("\n").length} lines) — import blocks can now use it`);
   345→      } else {
   346→        const ns = py.globals.get("dict")() as { destroy?: () => void };
   347→        try {
   348→          // step-guard: cooperative runaway-loop killer (see lib/pyguard.ts)
   349→          await withStepGuard(pyr, async () => {
   350→            if (pre) await pyr.runPythonAsync(pre, { globals: ns });
   351→            await pyr.runPythonAsync(current, { globals: ns });
   352→          });
   353→        } finally {
   354→          ns.destroy?.();
   355→        }
   356→      }
   357→      setElapsed((performance.now() - t0) / 1000);
   358→      setPhase("done");
   359→    } catch (e) {
   360→      const msg = e instanceof Error ? e.message : String(e);
   361→      setErr(cleanTraceback(msg) || msg);
   362→      setElapsed((performance.now() - t0) / 1000);
   363→      setPhase("error");
   364→    } finally {
   365→      if (py) {
   366→        try {
   367→          py.setStdout();
   368→          py.setStderr();
   369→          py.setStdin();
   370→        } catch {
   371→          /* already reset */
   372→        }
   373→      }
   374→    }
   375→  }
   376→
   377→  function copy() {
   378→    try {
   379→      navigator.clipboard.writeText(current).then(
   380→        () => {
   381→          setCopied(true);
   382→          window.setTimeout(() => setCopied(false), 1600);
   383→        },
   384→        () => undefined
   385→      );
   386→    } catch {
   387→      /* clipboard unavailable */
   388→    }
   389→  }
   390→
   391→  return (
   392→    <div className={`code-shell ${className}`}>
   393→      <div className="code-bar">
   394→        <span className="code-dot" style={{ background: "#e4674a" }} />
   395→        <span className="code-dot" style={{ background: "#e8a13c" }} />
   396→        <span className="code-dot" style={{ background: "#8fae5d" }} />
   397→        <span className="code-title ml-2">{title}</span>
   398→        {runnable ? (
   399→          <div className="ml-auto flex items-center gap-1">
   400→            <button
   401→              type="button"
   402→              onClick={copy}
   403→              className="code-icon-btn"
   404→              title="Copy code"
   405→              aria-label="Copy code"
   406→            >
   407→              {copied ? <Check className="h-3.5 w-3.5" style={{ color: "#8fae5d" }} /> : <Copy className="h-3.5 w-3.5" />}
   408→            </button>
   409→            <button
   410→              type="button"
   411→              onClick={() => {
   412→                setCurrent(code);
   413→                setEditing((v) => !v);
   414→              }}
   415→              className={`code-icon-btn ${editing ? "code-icon-btn-on" : ""}`}
   416→              title={dirty ? "Editing — click ⟲ to restore the original" : editing ? "Done editing" : "Edit the code, then run it"}
   417→              aria-label="Edit code"
   418→            >
   419→              <Pencil className="h-3.5 w-3.5" />
   420→            </button>
   421→            {dirty ? (
   422→              <button
   423→                type="button"
   424→                onClick={() => setCurrent(code)}
   425→                className="code-icon-btn"
   426→                title="Restore the original code"
   427→                aria-label="Restore original code"
   428→              >
   429→                <RotateCcw className="h-3.5 w-3.5" />
   430→              </button>
   431→            ) : null}
   432→            <button
   433→              type="button"
   434→              onClick={run}
   435→              disabled={busy}
   436→              className="run-btn"
   437→              title="Run this code live — real Python 3.12 in your browser"
   438→            >
   439→              {busy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Play className="h-3.5 w-3.5" />}
   440→              {busy ? "running" : "Run"}
   441→            </button>
   442→          </div>
   443→        ) : (
   444→          <span className="readonly-chip" title="Illustrative snippet — not executable">
   445→            <EyeOff className="h-3 w-3" />
   446→            read-only
   447→          </span>
   448→        )}
   449→      </div>
   450→
   451→      {editing ? (
   452→        <textarea
   453→          value={current}
   454→          onChange={(e) => setCurrent(e.target.value)}
   455→          spellCheck={false}
   456→          aria-label="Editable code"
   457→          className="code-edit"
   458→          rows={Math.max(6, current.split("\n").length + 1)}
   459→        />
   460→      ) : (
   461→        <HighlightedCode code={current} />
   462→      )}
   463→
   464→      <AnimatePresence>
   465→        {live ? (
   466→          <motion.div
   467→            key="live-term"
   468→            initial={{ opacity: 0, height: 0 }}
   469→            animate={{ opacity: 1, height: "auto" }}
   470→            exit={{ opacity: 0, height: 0 }}
   471→            transition={{ duration: 0.22 }}
   472→            style={{ overflow: "hidden" }}
   473→          >
   474→            <div className="live-term">
   475→              <div className="live-term-bar">
   476→                <Terminal className="h-3 w-3" />
   477→                LIVE RUN
   478→                <span className="ml-auto normal-case tracking-normal opacity-70">
   479→                  {bootMsg ? "python runtime" : "python 3.12 · webassembly"}
   480→                </span>
   481→              </div>
   482→              <pre ref={termRef} className="live-term-body">
   483→                {bootMsg ? (
   484→                  <span className="inline-flex items-center gap-2 text-[#f5c889]">
   485→                    <Loader2 className="h-3.5 w-3.5 animate-spin" /> {bootMsg}
   486→                  </span>
   487→                ) : null}
   488→                {out ? out : !bootMsg && busy ? "\u2026" : ""}
   489→                {!bootMsg && busy ? <span className="term-caret" aria-hidden /> : null}
   490→                {err ? <span className="tb-err">{err}\n</span> : null}
   491→              </pre>
   492→              {phase === "done" || phase === "error" ? (
   493→                <div className="live-term-foot">
   494→                  <span className={phase === "done" ? "foot-ok" : "foot-bad"}>
   495→                    {phase === "done"
   496→                      ? `✓ finished in ${elapsed.toFixed(2)}s`
   497→                      : "✗ the program raised an error"}
   498→                  </span>
   499→                  <span className="ml-auto opacity-60">CPython · Pyodide</span>
   500→                </div>
   501→              ) : null}
   502→            </div>
   503→          </motion.div>
   504→        ) : null}
   505→      </AnimatePresence>
   506→
   507→      {/* Pen-paper practice: runnable blocks HIDE the static output — the
   508→          learner predicts it first, then presses ▶ Run to see the real thing.
   509→          Read-only blocks keep their output (there is no Run button). */}
   510→      {output && !runnable ? (
   511→        <div className="code-out">
   512→          <span className="out-label">OUTPUT</span>
   513→          {output}
   514→        </div>
   515→      ) : null}
   516→      {note ? (
   517→        <div className="border-t border-[#4a3020] bg-[rgba(255,220,170,.04)] px-5 py-3 text-[.8rem] leading-relaxed text-[#d9b88c]">
   518→          {note}
   519→        </div>
   520→      ) : null}
   521→    </div>
   522→  );
   523→}

   540→  return (
   541→    <AnimatePresence>
   542→      {show ? (
   543→        <motion.div

   549→          role="status"
   550→          className="fixed bottom-4 left-4 z-[70] flex max-w-[calc(100vw-2rem)] items-center gap-2.5 rounded-2xl border border-[#ecd9bd] bg-[#fffaf1]/95 px-4 py-3 shadow-2xl backdrop-blur-md"

   555→          </span>
   556→          <span className="text-[.82rem] font-semibold leading-snug text-[#3e2c1d]">

   618→/* ============================================================
   619→   Quiz — interactive single-choice
   620→   ============================================================ */
   621→export function Quiz({
   622→  q,
   623→  options,
   624→  answer,
   625→  explain,
   626→  source,
   627→}: {
   628→  q: string;
   629→  options: string[];
   630→  answer: number;
   631→  explain: string;
   632→  source?: string;
   633→}) {
   634→  const [chosen, setChosen] = useState<number | null>(null);
   635→  const revealed = chosen !== null;
   636→  return (
   637→    <div className="card-warm p-5">
   638→      <div className="flex items-start gap-2">
   639→        <GraduationCap className="mt-0.5 h-5 w-5 shrink-0" style={{ color: "#c2410c" }} />
   640→        <div>
   641→          <p className="quiz-q font-semibold leading-snug text-[#3e2c1d]">{q}</p>
   642→          {source ? (
   643→            <p className="quiz-src mt-1 text-[.7rem] font-bold uppercase tracking-[.16em] text-[#8a5a10]">
   644→              {source}
   645→            </p>
   646→          ) : null}
   647→        </div>
   648→      </div>
   649→      <div className="mt-3 grid gap-2">
   650→        {options.map((opt, i) => {
   651→          const isAnswer = i === answer;
   652→          const cls = !revealed
   653→            ? "quiz-opt"
   654→            : isAnswer
   655→            ? "quiz-opt correct"
   656→            : i === chosen
   657→            ? "quiz-opt wrong"
   658→            : "quiz-opt";
   659→          return (
   660→            <button
   661→              key={i}
   662→              className={cls}
   663→              disabled={revealed}
   664→              onClick={() => setChosen(i)}
   665→              aria-label={`Option ${i + 1}: ${opt}`}
   666→            >
   667→              <span className="mr-2 inline-flex h-5 w-5 items-center justify-center rounded-full border border-current text-[.65rem] font-bold">
   668→                {revealed && isAnswer ? (
   669→                  <Check className="h-3 w-3" />
   670→                ) : revealed && i === chosen ? (
   671→                  <X className="h-3 w-3" />
   672→                ) : (
   673→                  String.fromCharCode(65 + i)
   674→                )}
   675→              </span>
   676→              {opt}
   677→            </button>
   678→          );
   679→        })}
   680→      </div>
   681→      {revealed ? (
   682→        <div className="anim-pop mt-3 rounded-xl border border-[#e8dcc0] bg-[#fbf3e3] p-3 text-[.85rem] leading-relaxed text-[#2b2014]">
   683→          <strong style={{ color: "#c2410c" }}>
   684→            {chosen === answer ? "Correct. " : "Not quite. "}
   685→          </strong>
   686→          {explain}
   687→        </div>
   688→      ) : null}
   689→    </div>
   690→  );
   691→}
