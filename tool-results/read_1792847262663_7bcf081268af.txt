     4→     4→   TEST MY KNOWLEDGE — THE ONE EXAM ENGINE.
     5→     5→
     6→     6→   All five shelves (FoC / ATB / FA / EBH / STM) used to carry a
     7→     7→   550-line clone of this window each. Every difference between
     8→     8→   them was pure presentation — colors, labels, copy — so the

    15→    15→   questions carry { id, type, q, options?, answer?, code?,

   124→   124→type QStatus = "unattempted" | "correct" | "wrong" | "revealed";
   125→   125→interface QState {
   126→   126→  chosen?: number;
   127→   127→  typed?: string;

   132→   132→function fmt(sec: number) {
   133→   133→  const m = Math.floor(sec / 60);
   134→   134→  const s = sec % 60;
   135→   135→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
   136→   136→}

   138→   138→export function ExamEngine({
   139→   139→  theme,
   140→   140→  bank,
   141→   141→  draftId,
   142→   142→  onClose,
   143→   143→}: {
   144→   144→  theme: ExamTheme;
   145→   145→  bank: EngineBank;
   146→   146→  /** stable per-window id (the exam group) — namespaces the draft */
   147→   147→  draftId: string;
   148→   148→  onClose: () => void;
   149→   149→}) {
   150→   150→  const qs = bank.questions;
   151→   151→
   152→   152→  /* ---- draft restore: a refresh or an accidental Escape keeps your work ---- */
   153→   153→  const DRAFT_KEY = examDraftKey(theme.key, draftId);
   154→   154→  const draft = useMemo(() => loadExamDraft(DRAFT_KEY), [DRAFT_KEY]);
   155→   155→  const restored = useMemo(() => {
   156→   156→    if (!draft) return null;
   157→   157→    let n = 0;
   158→   158→    for (const s of Object.values(draft.states)) {
   159→   159→      const st = s as QState | undefined;
   160→   160→      if (st && (st.status === "correct" || st.status === "wrong" || st.revealed)) n += 1;
   161→   161→    }
   162→   162→    return n > 0 ? n : null;
   163→   163→  }, [draft]);
   164→   164→
   165→   165→  const [idx, setIdx] = useState(() => Math.min(Math.max(draft?.idx ?? 0, 0), qs.length - 1));
   166→   166→  const [states, setStates] = useState<Record<string, QState>>(() => {
   167→   167→    const fresh: Record<string, QState> = Object.fromEntries(
   168→   168→      bank.questions.map((q) => [q.id, { status: "unattempted" as QStatus }])
   169→   169→    );
   170→   170→    if (draft) {
   171→   171→      for (const q of bank.questions) {
   172→   172→        const d = draft.states[q.id] as QState | undefined;
   173→   173→        if (d && typeof d.status === "string") fresh[q.id] = d;
   174→   174→      }
   175→   175→    }
   176→   176→    return fresh;
   177→   177→  });
   178→   178→  const [finished, setFinished] = useState(false);
   179→   179→  /* drift-proof clock: seconds = base + (now − startedAt), derived from the
   180→   180→     wall clock — never tick-counted, so background-tab throttling can't
   181→   181→     silently under-report the elapsed time. Survives refresh via the draft. */
   182→   182→  const [clock, setClock] = useState(() => ({
   183→   183→    base: draft?.base ?? 0,
   184→   184→    startedAt: draft?.startedAt ?? Date.now(),
   185→   185→  }));
   186→   186→  const [now, setNow] = useState(() => Date.now());
   187→   187→  const seconds = Math.max(0, Math.floor(clock.base + (now - clock.startedAt) / 1000));
   188→   188→  const bodyRef = useRef<HTMLDivElement>(null);
   189→   189→  /* one ledger row per attempt-cycle — "back to questions" then finishing
   190→   190→     again must not double-log; retake() re-arms it */
   191→   191→  const loggedRef = useRef(false);
   192→   192→
   193→   193→  /* scroll lock (the timer below stops ticking on the results view) */
   194→   194→  useEffect(() => {
   195→   195→    document.body.style.overflow = "hidden";
   196→   196→    return () => {
   197→   197→      document.body.style.overflow = "";
   198→   198→    };
   199→   199→  }, []);
   200→   200→
   201→   201→  useEffect(() => {
   202→   202→    if (finished) return;
   203→   203→    const t = window.setInterval(() => setNow(Date.now()), 1000);
   204→   204→    return () => window.clearInterval(t);
   205→   205→  }, [finished]);
   206→   206→
   207→   207→  /* draft autosave — debounced inside saveExamDraft, flushed on close/finish */
   208→   208→  useEffect(() => {
   209→   209→    if (finished) return;
   210→   210→    if (draftHasWork(states as Record<string, unknown>)) {
   211→   211→      saveExamDraft(DRAFT_KEY, {
   212→   212→        idx,
   213→   213→        states: states as Record<string, unknown>,
   214→   214→        base: clock.base,
   215→   215→        startedAt: clock.startedAt,
   216→   216→      });
   217→   217→    }
   218→   218→  }, [states, idx, clock, finished, DRAFT_KEY]);
   219→   219→
   220→   220→  const closeWindow = useCallback(() => {
   221→   221→    flushExamDraft(); // the attempt is kept — reopening this window restores it
   222→   222→    onClose();
   223→   223→  }, [onClose]);
   224→   224→
   225→   225→  /** freeze the clock, clear the draft, log the attempt, show the report */
   226→   226→  function finish() {
   227→   227→    flushExamDraft();
   228→   228→    clearExamDraft(DRAFT_KEY);
   229→   229→    setClock({ base: seconds, startedAt: Date.now() });
   230→   230→    setFinished(true);
   231→   231→    if (!loggedRef.current) {
   232→   232→      loggedRef.current = true;
   233→   233→      /* one row per subject: single-shelf windows log exactly one row
   234→   234→         (subject = theme.key, unchanged behaviour); mixed review sessions
   235→   235→         split into per-shelf rows so each shelf's accuracy stays true */
   236→   236→      const groups = new Map<string, { c: number; t: number }>();
   237→   237→      for (const qq of qs) {
   238→   238→        const s = states[qq.id]?.status;
   239→   239→        if (s !== "correct" && s !== "wrong") continue;
   240→   240→        const subj = qq.subject ?? theme.key;
   241→   241→        const g = groups.get(subj) ?? { c: 0, t: 0 };
   242→   242→        g.t += 1;
   243→   243→        if (s === "correct") g.c += 1;
   244→   244→        groups.set(subj, g);
   245→   245→      }
   246→   246→      const answered = groups.size > 0;
   247→   247→      const rows: { subject: string; correct: number; total: number }[] = answered
   248→   248→        ? [...groups.entries()].map(([subject, g]) => ({ subject, correct: g.c, total: g.t }))
   249→   249→        : [{ subject: theme.key, correct: score.correct, total: score.total }];
   250→   250→      for (const r of rows) {
   251→   251→        logAttempt({
   252→   252→          subject: r.subject,
   253→   253→          group: draftId,
   254→   254→          title: bank.title,
   255→   255→          correct: r.correct,
   256→   256→          total: r.total,
   257→   257→          pct: Math.round((r.correct / Math.max(1, r.total)) * 100),
   258→   258→          seconds,
   259→   259→        });
   260→   260→      }
   261→   261→    }
   262→   262→  }
   263→   263→
   264→   264→  useEffect(() => {
   265→   265→    const onKey = (e: KeyboardEvent) => {
   266→   266→      const inTextarea = e.target instanceof HTMLTextAreaElement;
   267→   267→      if (e.key === "Escape" && !inTextarea) closeWindow();
   268→   268→      if (!finished && !inTextarea) {
   269→   269→        if (e.key === "ArrowRight") setIdx((i) => Math.min(i + 1, qs.length - 1));
   270→   270→        if (e.key === "ArrowLeft") setIdx((i) => Math.max(i - 1, 0));
   271→   271→      }
   272→   272→    };
   273→   273→    document.addEventListener("keydown", onKey);
   274→   274→    return () => document.removeEventListener("keydown", onKey);
   275→   275→  }, [finished, closeWindow, qs.length]);
   276→   276→
   277→   277→  const q = qs[idx];
   278→   278→  const st = states[q.id];
   279→   279→
   280→   280→  const patch = (id: string, p: Partial<QState>) =>
   281→   281→    setStates((prev) => ({ ...prev, [id]: { ...prev[id], ...p } }));
   282→   282→
   283→   283→  /* scoring — byType rows follow the theme's declared type order */
   284→   284→  const typeKeys = useMemo(() => Object.keys(theme.typeMeta), [theme]);
   285→   285→  const score = useMemo(() => {
   286→   286→    let correct = 0;
   287→   287→    let wrong = 0;
   288→   288→    let answered = 0;
   289→   289→    const byType: Record<string, { c: number; t: number }> = {};
   290→   290→    for (const t of typeKeys) byType[t] = { c: 0, t: 0 };
   291→   291→    for (const qq of qs) {
   292→   292→      const row = (byType[qq.type] ??= { c: 0, t: 0 });
   293→   293→      row.t += 1;
   294→   294→      const s = states[qq.id]?.status ?? "unattempted";
   295→   295→      if (s === "correct" || s === "wrong") answered += 1;
   296→   296→      if (s === "correct") {
   297→   297→        correct += 1;
   298→   298→        row.c += 1;
   299→   299→      }
   300→   300→      if (s === "wrong") wrong += 1;
   301→   301→    }
   302→   302→    return { correct, wrong, answered, byType, total: qs.length };
   303→   303→  }, [states, qs, typeKeys]);
   304→   304→
   305→   305→  const pct = Math.round((score.correct / score.total) * 100);
   306→   306→  const verdict =
   307→   307→    pct >= 85 ? theme.tiers.topper : pct >= 60 ? theme.tiers.solid : pct >= 35 ? theme.tiers.getting : theme.tiers.restart;
   308→   308→
   309→   309→  function chooseMCQ(i: number) {
   310→   310→    if (st.status === "correct" || st.status === "wrong") return;
   311→   311→    const ok = i === q.answer;
   312→   312→    patch(q.id, {
   313→   313→      chosen: i,
   314→   314→      revealed: true,
   315→   315→      status: ok ? "correct" : "wrong",
   316→   316→    });
   317→   317→    // SM-2: first-try MCQ answers are trusted fully (5), misses reset the card (2)
   318→   318→    // — routed to the question's own shelf in mixed review sessions
   319→   319→    gradeQuestion(q.subject ?? theme.key, q.srcId ?? q.id, ok ? 5 : 2);
   320→   320→  }
   321→   321→
   322→   322→  function selfCheck(ok: boolean) {
   323→   323→    patch(q.id, { status: ok ? "correct" : "wrong" });
   324→   324→    // SM-2: the model answer was visible — grade honestly but trust less
   325→   325→    gradeQuestion(q.subject ?? theme.key, q.srcId ?? q.id, ok ? 3 : 1);
   326→   326→  }
   327→   327→
   328→   328→  function retake() {
   329→   329→    setStates(Object.fromEntries(qs.map((qq) => [qq.id, { status: "unattempted" as QStatus }])));
   330→   330→    setIdx(0);
   331→   331→    setClock({ base: 0, startedAt: Date.now() });
   332→   332→    setNow(Date.now());
   333→   333→    setFinished(false);
   334→   334→    loggedRef.current = false; // a retake is a new, honest attempt
   335→   335→    clearExamDraft(DRAFT_KEY);
   336→   336→    bodyRef.current?.scrollTo({ top: 0 });
   337→   337→  }
   338→   338→
   339→   339→  function go(i: number) {
   340→   340→    setIdx(i);
   341→   341→    bodyRef.current?.scrollTo({ top: 0, behavior: "smooth" });
   342→   342→  }
   343→   343→
   344→   344→  /* ---------------- results view ---------------- */
   345→   345→  if (finished) {
   346→   346→    const mcqMeta = theme.typeMeta.mcq ?? Object.values(theme.typeMeta)[0];
   347→   347→    return (
   348→   348→      <ExamShell
   349→   349→        theme={theme}
   350→   350→        bank={bank}
   351→   351→        score={score}
   352→   352→        seconds={seconds}
   353→   353→        onClose={closeWindow}
   354→   354→        restoredCount={restored}
   355→   355→        onExit={() => setFinished(false)}
   356→   356→        scrollRef={bodyRef}
   357→   357→      >
   358→   358→        <motion.div initial={{ opacity: 0, y: 18 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }}>
   359→   359→          <div className="exam-result-card">
   360→   360→            <div className="exam-score-ring" style={{ borderColor: pct >= 60 ? "#8fae5d" : "#e0a45e" }}>
   361→   361→              <span className="exam-score-num">{pct}%</span>
   362→   362→              <span className="exam-score-sub">
   363→   363→                {score.correct} / {score.total}
   364→   364→              </span>
   365→   365→            </div>
   366→   366→            <div>
   367→   367→              <h3 className="font-display text-2xl font-bold text-[#2b1d10]">Result report</h3>
   368→   368→              <p className="mt-1 text-[.95rem] font-semibold" style={{ color: verdict.c }} role="status">
   369→   369→                {verdict.t}
   370→   370→              </p>
   371→   371→              <div className="mt-3 flex flex-wrap gap-2 text-[.72rem] font-bold">
   372→   372→                <span className="exam-chip" style={{ background: "#eef4dd", color: "#44621d" }}>✓ {score.correct} correct</span>
   373→   373→                <span className="exam-chip" style={{ background: "#fdecec", color: "#8c2f2f" }}>✗ {score.wrong} wrong</span>
   374→   374→                <span className="exam-chip" style={{ background: "#f3e8d2", color: "#6f5438" }}>• {score.total - score.answered} unattempted</span>
   375→   375→                <span className="exam-chip" style={{ background: mcqMeta.bg, color: mcqMeta.color }}>
   376→   376→                  <Timer className="mr-1 inline h-3 w-3" />
   377→   377→                  {fmt(seconds)}
   378→   378→                </span>
   379→   379→              </div>
   380→   380→            </div>
   381→   381→          </div>
   382→   382→
   383→   383→          {/* per-type breakdown */}
   384→   384→          <div className="mt-5 grid gap-3 sm:grid-cols-2">
   385→   385→            {typeKeys.map((t) => {
   386→   386→              const meta = theme.typeMeta[t];
   387→   387→              const d = score.byType[t];
   388→   388→              if (!meta || !d) return null;
   389→   389→              return (
   390→   390→                <div key={t} className="flex items-center justify-between rounded-xl border border-[#e8dcc0] bg-[#fffdf9] px-4 py-3">
   391→   391→                  <span className="text-[.8rem] font-bold" style={{ color: meta.color }}>
   392→   392→                    {meta.label}
   393→   393→                  </span>
   394→   394→                  <span className="text-[.8rem] font-extrabold text-[#2b1d10]">
   395→   395→                    {d.c} / {d.t}
   396→   396→                  </span>
   397→   397→                </div>
   398→   398→              );
   399→   399→            })}
   400→   400→          </div>
   401→   401→
   402→   402→          {/* review the misses */}
   403→   403→          {score.correct < score.total ? (
   404→   404→            <div className="mt-6">
   405→   405→              <p className="exam-eyebrow">
   406→   406→                <Printer className="h-3.5 w-3.5" />
   407→   407→                Review &amp; rewrite these by hand
   408→   408→              </p>
   409→   409→              <div className="mt-3 grid gap-3">
   410→   410→                {qs.map((qq) => {
   411→   411→                  const s = states[qq.id]?.status ?? "unattempted";
   412→   412→                  if (s === "correct") return null;
   413→   413→                  const model =
   414→   414→                    qq.type === "mcq"
   415→   415→                      ? (qq.options?.[qq.answer ?? 0] ?? "")
   416→   416→                      : qq.answerText ?? qq.solution ?? "";
   417→   417→                  return (
   418→   418→                    <div key={qq.id} className="rounded-xl border border-[#e8dcc0] bg-[#fffdf9] p-4">
   419→   419→                      <p className="text-[.88rem] font-bold leading-snug text-[#2b1d10]">{qq.q}</p>
   420→   420→                      {qq.code ? <pre className="exam-code mt-2">{qq.code}</pre> : null}
   421→   421→                      <p className="mt-2 text-[.82rem] leading-relaxed" style={{ color: s === "wrong" ? "#8c2f2f" : "#6f5438" }}>
   422→   422→                        <strong>{s === "wrong" ? "Missed —" : "Unattempted —"}</strong>{" "}
   423→   423→                        <span className="whitespace-pre-wrap font-mono text-[.78rem] text-[#2f2013]">{model}</span>
   424→   424→                      </p>
   425→   425→                      {qq.explain ? <p className="mt-1 text-[.78rem] leading-relaxed text-[#6f5438]">{qq.explain}</p> : null}
   426→   426→                    </div>
   427→   427→                  );
   428→   428→                })}
   429→   429→              </div>
   430→   430→            </div>
   431→   431→          ) : null}
   432→   432→
   433→   433→          <div className="mt-7 flex flex-wrap justify-center gap-3">
   434→   434→            <button type="button" onClick={retake} className="exam-btn exam-btn-primary">
   435→   435→              <RotateCcw className="h-4 w-4" />
   436→   436→              Retake this test
   437→   437→            </button>
   438→   438→            <button type="button" onClick={() => setFinished(false)} className="exam-btn">
   439→   439→              <ListChecks className="h-4 w-4" />
   440→   440→              Back to questions
   441→   441→            </button>
   442→   442→          </div>
   443→   443→        </motion.div>
   444→   444→      </ExamShell>
   445→   445→    );
   446→   446→  }

   449→   449→  const meta = theme.typeMeta[q.type] ?? theme.typeMeta.mcq ?? Object.values(theme.typeMeta)[0];
   450→   450→  const mcqLocked = st.status === "correct" || st.status === "wrong";
   451→   451→
   452→   452→  return (
   453→   453→    <ExamShell
   454→   454→      theme={theme}
   455→   455→      bank={bank}
   456→   456→      score={score}
   457→   457→      seconds={seconds}
   458→   458→      onClose={closeWindow}
   459→   459→      restoredCount={restored}
   460→   460→      onExit={() => finish()}
   461→   461→      scrollRef={bodyRef}
   462→   462→    >
   463→   463→      <motion.div key={q.id} initial={{ opacity: 0, x: 24 }} animate={{ opacity: 1, x: 0 }} transition={{ duration: 0.22, ease: "easeOut" }}>
   464→   464→        {/* type chip + counter */}
   465→   465→        <div className="flex flex-wrap items-center gap-2">
   466→   466→          <span className="exam-type-chip" style={{ color: meta.color, background: meta.bg }}>
   467→   467→            {meta.label}
   468→   468→          </span>
   469→   469→          <span className="text-[.72rem] font-extrabold uppercase tracking-[.16em] text-[#8a6d4f]">
   470→   470→            Question {idx + 1} of {qs.length}
   471→   471→          </span>
   472→   472→        </div>
   473→   473→
   474→   474→        {/* question */}
   475→   475→        <h3 className="font-display mt-3 text-xl font-bold leading-snug text-[#2b1d10] md:text-[1.35rem]">{q.q}</h3>
   476→   476→
   477→   477→        {q.code ? <pre className="exam-code mt-4">{q.code}</pre> : null}
   478→   478→
   479→   479→        {/* answer area */}
   480→   480→        {q.type === "mcq" ? (
   481→   481→          <div className="mt-4 grid gap-2">
   482→   482→            {q.options?.map((opt, i) => {
   483→   483→              const isAnswer = i === q.answer;
   484→   484→              const chosen = st.chosen === i;
   485→   485→              let cls = "exam-opt";
   486→   486→              if (mcqLocked && isAnswer) cls += " exam-opt-correct";
   487→   487→              else if (mcqLocked && chosen) cls += " exam-opt-wrong";

   495→   495→            })}

   507→   507→              rows={q.type === "short" ? 4 : 6}
   508→   508→              placeholder={q.type === "output" ? theme.outputPlaceholder : q.type === "code" ? "# write your program here…" : "your answer…"}

   510→   510→            />
   511→   511→            {!st.revealed ? (
   512→   512→              <button type="button" onClick={() => patch(q.id, { revealed: true })} className="exam-btn mt-2">

   515→   515→              </button>
   516→   516→            ) : (

   518→   518→                initial={{ opacity: 0, y: 10 }}
   519→   519→                animate={{ opacity: 1, y: 0 }}

   521→   521→                role="status"
   522→   522→              >
   523→   523→                <div className="exam-answer">
   524→   524→                  <p className="exam-eyebrow">

   531→   531→                      {q.answerText}
   532→   532→                    </p>
   533→   533→                  ) : null}
   534→   534→                  {q.explain ? <p className="mt-2 text-[.82rem] leading-relaxed text-[#6f5438]">{q.explain}</p> : null}

   536→   536→                {st.status !== "correct" && st.status !== "wrong" ? (
   537→   537→                  <div className="mt-3 flex flex-wrap items-center gap-2">

   543→   543→                      <X className="h-4 w-4" /> Not yet
   544→   544→                    </button>
   545→   545→                  </div>
   546→   546→                ) : null}
   547→   547→              </motion.div>
   548→   548→            )}
   549→   549→          </div>
   550→   550→        )}
   551→   551→
   552→   552→        {/* MCQ explanation */}
   553→   553→        {q.type === "mcq" && mcqLocked && q.explain ? (
   554→   554→          <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} className="exam-answer mt-3" role="status">
   555→   555→            <p className="text-[.88rem] font-bold" style={{ color: st.status === "correct" ? "#44621d" : "#8c2f2f" }}>

   566→   566→            Jump to a question
   567→   567→          </p>
   568→   568→          <div className="mt-2 flex flex-wrap gap-1.5">
   569→   569→            {qs.map((qq, i) => {
   570→   570→              const s = states[qq.id]?.status ?? "unattempted";
   571→   571→              const cls =
   572→   572→                i === idx
   573→   573→                  ? "exam-dot exam-dot-current"
   574→   574→                  : s === "correct"
   575→   575→                  ? "exam-dot exam-dot-ok"
   576→   576→                  : s === "wrong"
   577→   577→                  ? "exam-dot exam-dot-bad"
   578→   578→                  : s === "revealed"
   579→   579→                  ? "exam-dot exam-dot-rev"
   580→   580→                  : "exam-dot";
   581→   581→              return (
   582→   582→                <button key={qq.id} type="button" onClick={() => go(i)} className={cls} aria-label={`Go to question ${i + 1}`}>
   583→   583→                  {i + 1}
   584→   584→                </button>
   585→   585→              );
   586→   586→            })}
   587→   587→          </div>
   588→   588→        </div>
   589→   589→
   590→   590→        {/* prev / next */}
   591→   591→        <div className="mt-6 flex items-center justify-between pb-2">
   592→   592→          <button type="button" onClick={() => go(Math.max(0, idx - 1))} disabled={idx === 0} className="exam-btn">
   593→   593→            <ChevronLeft className="h-4 w-4" />
   594→   594→            Prev
   595→   595→          </button>
   596→   596→          {idx < qs.length - 1 ? (
   597→   597→            <button type="button" onClick={() => go(idx + 1)} className="exam-btn exam-btn-primary">
   598→   598→              Next
   599→   599→              <ChevronRight className="h-4 w-4" />
   600→   600→            </button>
   601→   601→          ) : (
   602→   602→            <button type="button" onClick={() => finish()} className="exam-btn exam-btn-primary">
   603→   603→              <Trophy className="h-4 w-4" />
   604→   604→              Finish &amp; see score
   605→   605→            </button>
   606→   606→          )}
   607→   607→        </div>
   608→   608→      </motion.div>
   609→   609→    </ExamShell>
   610→   610→  );
   611→   611→}
