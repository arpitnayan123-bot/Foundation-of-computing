   190→/* ============================================================
   191→   useSeenDisplay — the read-checkmark data for every menu.
   192→
   193→   Marking is per-lecture (only the rendered chapters can be
   194→   swept — the Kindle-style furthest-read sweep must never
   195→   sweep chapters that aren't on the page), but DISPLAY spans
   196→   the whole shelf: whatever the reader marked in earlier
   197→   sessions/lectures stays visible via the persisted record.
   198→   ============================================================ */
   199→function useSeenDisplay(lecture: AtbLecture): ReadonlySet<string> {
   200→  const ids = useMemo(() => lecture.chapters.map((c) => c.id), [lecture]);
   201→  const live = useSectionProgress("atb", ids);
   202→  const persisted = useMemo(() => {
   203→    try {
   204→      return new Set(Object.keys(loadProgress().atb ?? {}));
   205→    } catch {
   206→      return new Set<string>();
   207→    }
   208→  }, []);
   209→  return useMemo(() => new Set<string>([...persisted, ...live]), [persisted, live]);
   210→}

   212→/* ============================================================
   213→   TopNav — structured chapter menu + a "Test my knowledge"
   214→   launcher inside every chapter group.
   215→
   216→   PERF (kept from FocApp): the progress bar is written straight
   217→   to the DOM through a ref inside a requestAnimationFrame, and
   218→   React state updates only when the active section CHANGES —
   219→   the nav never re-renders per scroll tick.
   220→
   221→   Chapters live inside the lecture reader, so a menu click on a
   222→   chapter of a DIFFERENT lecture first switches the lecture and
   223→   then scrolls (goToChapter in AtbApp) — a plain anchor would
   224→   aim at a section that isn't rendered.
   225→   ============================================================ */

   227→  onOpenExam,
   228→  onExitAtlas,
   229→  onJumpChapter,
   230→  seen,
   231→}: {
   232→  onOpenExam: (g: AtbExamGroup) => void;
   233→  onExitAtlas: () => void;
   234→  onJumpChapter: (chapterId: string) => void;
   235→  seen: ReadonlySet<string>;
   236→}) {
   237→  const [active, setActive] = useState<string>("");
   238→  const [openGroup, setOpenGroup] = useState<string | null>(null);
   239→  const [mobileOpen, setMobileOpen] = useState(false);
   240→  const mobileRef = useDialogA11y<HTMLDivElement>(mobileOpen);
   241→  const navRef = useRef<HTMLElement>(null);
   242→  const progressRef = useRef<HTMLDivElement>(null);
   243→  const activeRef = useRef("");
   244→
   245→  useEffect(() => {
   246→    let ticking = false;
   247→    const update = () => {
   248→      ticking = false;
   249→      const doc = document.documentElement;
   250→      const total = doc.scrollHeight - doc.clientHeight;
   251→      const pct = total > 0 ? (doc.scrollTop / total) * 100 : 0;
   252→      if (progressRef.current) progressRef.current.style.width = `${pct}%`;
   253→
   254→      // scroll-spy: only touch React state when the section changes.
   255→      // Sections of other lectures simply aren't in the DOM, so the
   256→      // loop naturally spies on the rendered lecture only.
   257→      const probe = window.innerHeight * 0.32;
   258→      let current = "";
   259→      for (const item of FLAT_ITEMS) {
   260→        const el = document.getElementById(item.id);
   261→        if (el && el.getBoundingClientRect().top <= probe) current = item.id;
   262→      }
   263→      if (current !== activeRef.current) {
   264→        activeRef.current = current;
   265→        setActive(current);
   266→      }
   267→    };
   268→    const onScroll = () => {
   269→      if (!ticking) {
   270→        ticking = true;
   271→        requestAnimationFrame(update);
   272→      }
   273→    };
   274→    update();
   275→    window.addEventListener("scroll", onScroll, { passive: true });
   276→    window.addEventListener("resize", onScroll);
   277→    return () => {
   278→      window.removeEventListener("scroll", onScroll);
   279→      window.removeEventListener("resize", onScroll);
   280→    };
   281→  }, []);
   282→
   283→  // close dropdown on outside click / Escape
   284→  useEffect(() => {
   285→    const onDown = (e: MouseEvent) => {
   286→      if (navRef.current && !navRef.current.contains(e.target as Node)) setOpenGroup(null);
   287→    };
   288→    const onKey = (e: KeyboardEvent) => {
   289→      if (e.key === "Escape") {
   290→        setOpenGroup(null);
   291→        setMobileOpen(false);
   292→      }
   293→    };
   294→    document.addEventListener("mousedown", onDown);
   295→    document.addEventListener("keydown", onKey);
   296→    return () => {
   297→      document.removeEventListener("mousedown", onDown);
   298→      document.removeEventListener("keydown", onKey);
   299→    };
   300→  }, []);
   301→
   302→  // lock body scroll while the mobile index is open
   303→  useEffect(() => {
   304→    document.body.style.overflow = mobileOpen ? "hidden" : "";
   305→    return () => {
   306→      document.body.style.overflow = "";
   307→    };
   308→  }, [mobileOpen]);
   309→
   310→  const activeGroup = active ? groupOf(active) : null;
   311→
   312→  return (
   313→    <nav ref={navRef} aria-label="Chapters" className="fixed inset-x-0 top-0 z-50">
   314→      <div className="nav-shell">
   315→        <div className="mx-auto flex h-14 max-w-7xl items-center gap-2 px-3 sm:px-5">
   316→          {/* back to The Atlas — the shelf this subject lives on */}
   317→          <button
   318→            type="button"
   319→            onClick={onExitAtlas}
   320→            title="Back to The Atlas — the subject shelf"
   321→            aria-label="Back to The Atlas"
   322→            className="flex h-9 shrink-0 items-center gap-1.5 rounded-full border border-white/70 bg-white/60 px-3 text-[.64rem] font-extrabold uppercase tracking-[.14em] text-[#8a5f30] shadow-[0_4px_16px_-8px_rgba(70,42,15,.35)] backdrop-blur-md transition-colors duration-200 hover:bg-white/85 hover:text-[#c2410c]"
   323→          >
   324→            <Map className="h-3.5 w-3.5" />
   325→            <span className="hidden sm:inline">Atlas</span>
   326→          </button>
   327→
   328→          {/* brand — quiet glass chip, Apple-quiet rather than loud gradient */}
   329→          <a
   330→            href="#top"
   331→            className="flex shrink-0 items-center gap-2 rounded-full border border-white/70 bg-white/60 py-1.5 pl-2 pr-3.5 shadow-[0_4px_16px_-8px_rgba(70,42,15,.35)] backdrop-blur-md transition-colors duration-200 hover:bg-white/80"
   332→          >
   333→            <span
   334→              className="flex h-6 w-6 items-center justify-center rounded-full text-white"
   335→              style={{ background: "linear-gradient(135deg,#ea580c,#c2410c)", boxShadow: "0 4px 10px -3px rgba(194,65,12,.55)" }}
   336→            >
   337→              <Route className="h-3.5 w-3.5" />
   338→            </span>
   339→            <span className="text-sm font-black tracking-tight text-[#3e2c1d]">ATB Notes</span>
   340→          </a>
   341→
   342→          {/* desktop — grouped chapter menu (all six groups). Opens at lg:
   343→              below that the bar would overflow once every group is shown,
   344→              so narrower screens get the scrolling chip strip + the
   345→              full-screen index instead. */}
   346→          <div className="ml-2 hidden flex-1 items-center gap-1 lg:flex">
   347→            {GROUPS.map((group) => {
   348→              const isActive = activeGroup === group.id;
   349→              const open = openGroup === group.id;
   350→              return (
   351→                <div
   352→                  key={group.id}
   353→                  className="relative"
   354→                  onMouseEnter={() => setOpenGroup(group.id)}
   355→                  onMouseLeave={() => setOpenGroup((g) => (g === group.id ? null : g))}
   356→                >
   357→                  <button
   358→                    type="button"
   359→                    aria-expanded={open}
   360→                    aria-haspopup="true"
   361→                    onClick={() => setOpenGroup(group.id)}
   362→                    onFocus={() => setOpenGroup(group.id)}
   363→                    className={`flex items-center gap-1.5 rounded-full px-3.5 py-1.5 text-[.78rem] font-semibold tracking-wide transition-all duration-200 ${
   364→                      isActive || open
   365→                        ? "bg-white/80 text-[#c2410c] shadow-[0_4px_16px_-8px_rgba(70,42,15,.4),inset_0_1px_0_rgba(255,255,255,.9)]"
   366→                        : "text-[#6b563d] hover:bg-white/55 hover:text-[#3e2c1d] hover:shadow-[inset_0_1px_0_rgba(255,255,255,.9)]"
   367→                    }`}
   368→                  >
   369→                    {group.label}
   370→                    <span className="text-[.58rem] font-extrabold opacity-45">{String(group.items.length).padStart(2, "0")}</span>
   371→                    <ChevronDown className={`h-3 w-3 opacity-50 transition-transform duration-200 ${open ? "rotate-180" : ""}`} />
   372→                  </button>
   373→
   374→                  <AnimatePresence>
   375→                    {open ? (

   378→                        initial={{ opacity: 0, y: 8, scale: 0.98 }}
   379→                        animate={{ opacity: 1, y: 0, scale: 1 }}
   380→                        exit={{ opacity: 0, y: 6, scale: 0.98 }}
   381→                        transition={{ duration: 0.16, ease: "easeOut" }}
   382→                        className="absolute left-1/2 top-full z-50 w-[min(26rem,calc(100vw-2rem))] -translate-x-1/2 pt-2"
   383→                      >
   384→                        <div
   385→                          className="overflow-hidden rounded-2xl border border-white/60 bg-white/70 p-1.5 backdrop-blur-2xl"
   386→                          style={{
   387→                            WebkitBackdropFilter: "blur(28px) saturate(170%)",
   388→                            backdropFilter: "blur(28px) saturate(170%)",
   389→                            boxShadow: "0 24px 70px -18px rgba(50,30,10,.4), inset 0 1px 0 rgba(255,255,255,.85)",
   390→                          }}
   391→                        >
   392→                          <p className="flex items-center gap-1.5 px-3 pb-1.5 pt-2 text-[.58rem] font-extrabold uppercase tracking-[.22em] text-[#b08b5c]">
   393→                            <BookOpenText className="h-3 w-3" />
   394→                            {group.label}
   395→                          <span className="ml-auto font-mono tracking-[.08em] opacity-90">
   396→                            {group.items.filter((it) => seen.has(it.id)).length}/{group.items.length} read

   398→                          </p>
   399→                          {group.items.map((item, i) => {

   412→                                }`}
   413→                              >
   414→                                <span
   415→                                  className={`mt-0.5 flex h-5 w-5 shrink-0 items-center justify-center rounded-md text-[.58rem] font-black ${
   416→                                    isActiveItem ? "text-white" : "border border-white/80 bg-white/70 text-[#b08b5c]"
   417→                                  }`}
   418→                                  style={isActiveItem ? { background: "linear-gradient(135deg,#ea580c,#c2410c)" } : undefined}
   419→                                >
   420→                                  {String(i + 1).padStart(2, "0")}
   421→                                </span>
   422→                                <span className="min-w-0">
   423→                                  <span className={`block text-[.82rem] font-bold leading-tight ${isActiveItem ? "text-[#c2410c]" : "text-[#3e2c1d]"}`}>

   426→                                  <span className="mt-0.5 block text-[.7rem] leading-snug text-[#a08568]">{item.desc}</span>
   427→                                </span>
   428→                                {isActiveItem ? (

   435→                          })}
   436→
   437→                          {/* Test my knowledge — opens the exam window for this group */}

   441→                              onClick={() => {
   442→                                setOpenGroup(null);
   443→                                onOpenExam(group.id);
   444→                              }}
   445→                              className="group/exam flex w-full items-center gap-2.5 rounded-xl px-3 py-2.5 text-left transition-colors duration-150 hover:bg-[#3e2c1d]/[.05]"
   446→                            >
   447→                              <span
   448→                                className="flex h-7 w-7 shrink-0 items-center justify-center rounded-lg text-white"
   449→                                style={{ background: "linear-gradient(135deg,#ea580c,#c2410c)" }}
   450→                              >
   451→                                <PenLine className="h-3.5 w-3.5" />
   452→                              </span>
   453→                              <span className="min-w-0">

   784→            onClick={() => onJump(prev.id)}
   785→            className="card-warm lift group flex items-center gap-3 p-4 text-left"
   786→            aria-label={`Previous: ${prev.code} — ${prev.title}`}
   787→          >
   788→            <ArrowLeft className="h-4 w-4 shrink-0 text-[#c2410c] transition-transform duration-200 group-hover:-translate-x-1" />
   789→            <span className="min-w-0">
   790→              <span className="block text-[.6rem] font-extrabold uppercase tracking-[.18em] text-[#a16207]">

   792→              </span>

   794→            </span>

   966→   AtbApp — the composed subject page: a lecture reader.

   968→export function AtbApp({ onExitAtlas }: { onExitAtlas: () => void }) {
   969→  const [lectureId, setLectureId] = useState<string>(ATB_DEFAULT_LECTURE);
   970→  const [examGroup, setExamGroup] = useState<AtbExamGroup | null>(null);
   971→  const [hubOpen, setHubOpen] = useState(false);
   972→  const lectureIdRef = useRef(lectureId);
   973→  useEffect(() => {
   974→    lectureIdRef.current = lectureId;
   975→  }, [lectureId]);
   976→
   977→  const lecture = lectureById(lectureId) ?? ATB_LECTURES[0];
   978→  const seen = useSeenDisplay(lecture);
   979→
   980→  /* pick a lecture from the hero sequence (or the pager) and bring
   981→     the notes into view — "the lecture number 1 notes will be
   982→     available below", literally */
   983→  const selectLecture = useCallback((id: string) => {
   984→    setLectureId(id);
   985→    requestAnimationFrame(() =>
   986→      requestAnimationFrame(() => {
   987→        const el = document.getElementById("notes-top");
   988→        if (!el) return;
   989→        const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   990→        el.scrollIntoView({ behavior: reduce ? "auto" : "smooth", block: "start" });
   991→      })
   992→    );
   993→  }, []);
   994→
   995→  /* chapter jumps — from the nav menus or the ⌘K palette. If the
   996→     chapter belongs to another lecture, switch first, wait for the
   997→     render, then run the standard flash-jump */
   998→  const goToChapter = useCallback((chapterId: string) => {
   999→    const target = lectureOfChapter(chapterId);
  1000→    if (!target) return;
  1001→    const scrollWhenReady = () => {
  1002→      let frames = 0;
  1003→      const tick = () => {
  1004→        const el = document.getElementById(chapterId);
  1005→        if (el) {
  1006→          jumpTo(el);
  1007→          return;
  1008→        }
  1009→        if (++frames < 240) requestAnimationFrame(tick);
  1010→      };
  1011→      requestAnimationFrame(tick);
  1012→    };
  1013→    if (lectureIdRef.current === target.id) {
  1014→      scrollWhenReady();
  1015→      return;
  1016→    }
  1017→    setLectureId(target.id);
  1018→    requestAnimationFrame(() => requestAnimationFrame(scrollWhenReady));
  1019→  }, []);
  1020→
  1021→  /* ⌘K handoffs (queued + live) route through the lecture-aware jumper */
  1022→  useOpenSection("atb", goToChapter);
  1023→
  1024→  const openExam = useCallback((g: AtbExamGroup) => setExamGroup(g), []);
  1025→  const openHub = useCallback(() => setHubOpen(true), []);
  1026→
  1027→  return (
  1028→    // id="top" — the brand chip links to #top; without a real element here the
  1029→    // client-side router swallows the fragment and the page never scrolls back
  1030→    <div id="top" className="page-wrap warm-canvas">
  1031→      <TopNav onOpenExam={openExam} onExitAtlas={onExitAtlas} onJumpChapter={goToChapter} seen={seen} />
  1032→      <main className="pt-14">
  1033→        <PartBoundary label="Hero">
  1034→          <AtbHero
  1035→            activeId={lectureId}
  1036→            onSelect={selectLecture}
  1037→            onPractice={openHub}
  1038→            seen={seen}
  1039→            totalQuestions={ATB_TOTAL_QUESTIONS}
  1040→          />
  1041→        </PartBoundary>
  1042→
  1043→        {/* the notes start here — selecting a lecture lands the reader on this line */}
  1044→        <div id="notes-top" aria-hidden className="relative scroll-mt-16" />
  1045→
  1046→        <PartBoundary key={lecture.id} label={`Lecture ${lecture.code} — ${lecture.title}`}>
  1047→          <motion.div
  1048→            initial={{ opacity: 0, y: 18 }}
  1049→            animate={{ opacity: 1, y: 0 }}
  1050→            transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
  1051→          >
  1052→            {lecture.chapters.length > 0 ? (
  1053→              lecture.chapters.map((c) => (
  1054→                <PartBoundary key={c.id} label={c.label}>
  1055→                  {SECTION_COMPONENTS[c.id]}
  1056→                </PartBoundary>
  1057→              ))
  1058→            ) : (
  1059→              <div className="mx-auto max-w-3xl px-4 py-16 sm:px-6">
  1060→                <div className="card-warm p-8 text-center">
  1061→                  <span className="chip mx-auto">
  1062→                    <Hourglass className="h-3.5 w-3.5" /> Slot reserved
  1063→                  </span>
  1064→                  <h2 className="mt-4 text-2xl font-extrabold text-[#3e2c1d]">
  1065→                    Lecture {lecture.n} hasn&rsquo;t been summarised yet
  1066→                  </h2>
  1067→                  <p className="mx-auto mt-3 max-w-md text-[.92rem] leading-relaxed text-muted-warm">
  1068→                    This lecture&rsquo;s notes haven&rsquo;t been uploaded, so the shelf keeps the
  1069→                    slot open. The moment its summary arrives, the chapters fill in right here —
  1070→                    diagrams, machines and all.
  1071→                  </p>
  1072→                </div>
  1073→              </div>
  1074→            )}
  1075→            <LecturePager current={lecture} onJump={selectLecture} />
  1076→          </motion.div>
  1077→        </PartBoundary>
  1078→      </main>
  1079→      <AtbFooter />
  1080→      <PracticeHub open={hubOpen} onClose={() => setHubOpen(false)} onOpenExam={openExam} />
  1081→      {examGroup ? <AtbExamWindow group={examGroup} onClose={() => setExamGroup(null)} /> : null}
  1082→    </div>
  1083→  );
  1084→}
