   146→   146→function useSeenDisplay(lecture: MktgLecture): ReadonlySet<string> {
   147→   147→  const ids = useMemo(() => lecture.chapters.map((c) => c.id), [lecture]);
   148→   148→  const live = useSectionProgress("mktg", ids);
   149→   149→  const persisted = useMemo(() => {
   150→   150→    try {
   151→   151→      return new Set(Object.keys(loadProgress().mktg ?? {}));
   152→   152→    } catch {
   153→   153→      return new Set<string>();
   154→   154→    }
   155→   155→  }, []);
   156→   156→  return useMemo(() => new Set<string>([...persisted, ...live]), [persisted, live]);
   157→   157→}

   179→   179→  const progressRef = useRef<HTMLDivElement>(null);
   180→   180→  const activeRef = useRef("");
   181→   181→
   182→   182→  useEffect(() => {
   183→   183→    let ticking = false;
   184→   184→    const update = () => {
   185→   185→      ticking = false;
   186→   186→      const doc = document.documentElement;
   187→   187→      const total = doc.scrollHeight - doc.clientHeight;

   189→   189→      if (progressRef.current) progressRef.current.style.width = `${pct}%`;
   190→   190→
   191→   191→      const probe = window.innerHeight * 0.32;
   192→   192→      let current = "";
   193→   193→      for (const item of FLAT_ITEMS) {
   194→   194→        const el = document.getElementById(item.id);
   195→   195→        if (el && el.getBoundingClientRect().top <= probe) current = item.id;
   196→   196→      }
   197→   197→      if (current !== activeRef.current) {
   198→   198→        activeRef.current = current;
   199→   199→        setActive(current);
   200→   200→      }
   201→   201→    };
   202→   202→    const onScroll = () => {
   203→   203→      if (!ticking) {
   204→   204→        ticking = true;
   205→   205→        requestAnimationFrame(update);
   206→   206→      }

   210→   210→    window.addEventListener("resize", onScroll);
   211→   211→    return () => {
   212→   212→      window.removeEventListener("scroll", onScroll);
   213→   213→      window.removeEventListener("resize", onScroll);

   247→   247→        <div className="mx-auto flex h-14 max-w-7xl items-center gap-2 px-3 sm:px-5">

   255→   255→          >
   256→   256→            <Map className="h-3.5 w-3.5" />

   261→   261→          <a
   262→   262→            href="#top"
   263→   263→            className="flex shrink-0 items-center gap-2 rounded-full border border-white/70 bg-white/60 py-1.5 pl-2 pr-3.5 shadow-[0_4px_16px_-8px_rgba(15,50,48,.35)] backdrop-blur-md transition-colors duration-200 hover:bg-white/80"
   264→   264→          >
   265→   265→            <span
   266→   266→              className="flex h-6 w-6 items-center justify-center rounded-full text-white"
   267→   267→              style={{ background: `linear-gradient(135deg,${TEAL_BRIGHT},${TEAL})`, boxShadow: "0 4px 10px -3px rgba(15,118,110,.55)" }}
   268→   268→            >

   879→   879→   MktgApp — the composed subject page: a lecture reader.
   880→   880→   ============================================================ */
   881→   881→export function MktgApp({ onExitAtlas }: { onExitAtlas: () => void }) {
   882→   882→  const [lectureId, setLectureId] = useState<string>(MKTG_DEFAULT_LECTURE);
   883→   883→  const [examGroup, setExamGroup] = useState<MktgExamGroup | null>(null);
   884→   884→  const [hubOpen, setHubOpen] = useState(false);
   885→   885→  const lectureIdRef = useRef(lectureId);
   886→   886→  useEffect(() => {
   887→   887→    lectureIdRef.current = lectureId;
   888→   888→  }, [lectureId]);
   889→   889→
   890→   890→  const lecture = lectureById(lectureId) ?? MKTG_LECTURES[0];
   891→   891→  const seen = useSeenDisplay(lecture);
   892→   892→
   893→   893→  const selectLecture = useCallback((id: string) => {
   894→   894→    setLectureId(id);
   895→   895→    requestAnimationFrame(() =>
   896→   896→      requestAnimationFrame(() => {
   897→   897→        const el = document.getElementById("notes-top");
   898→   898→        if (!el) return;
   899→   899→        const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   900→   900→        el.scrollIntoView({ behavior: reduce ? "auto" : "smooth", block: "start" });
   901→   901→      })
   902→   902→    );
   903→   903→  }, []);
   904→   904→
   905→   905→  const goToChapter = useCallback((chapterId: string) => {
   906→   906→    const target = lectureOfChapter(chapterId);
   907→   907→    if (!target) return;
   908→   908→    const scrollWhenReady = () => {
   909→   909→      let frames = 0;
   910→   910→      const tick = () => {
   911→   911→        const el = document.getElementById(chapterId);
   912→   912→        if (el) {
   913→   913→          jumpTo(el);
   914→   914→          return;
   915→   915→        }
   916→   916→        if (++frames < 240) requestAnimationFrame(tick);
   917→   917→      };
   918→   918→      requestAnimationFrame(tick);
   919→   919→    };
   920→   920→    if (lectureIdRef.current === target.id) {
   921→   921→      scrollWhenReady();
   922→   922→      return;
   923→   923→    }
   924→   924→    setLectureId(target.id);
   925→   925→    requestAnimationFrame(() => requestAnimationFrame(scrollWhenReady));
   926→   926→  }, []);
   927→   927→
   928→   928→  useOpenSection("mktg", goToChapter);
   929→   929→
   930→   930→  const openExam = useCallback((g: MktgExamGroup) => setExamGroup(g), []);
   931→   931→  const openHub = useCallback(() => setHubOpen(true), []);
   932→   932→
   933→   933→  return (
   934→   934→    <div id="top" className="page-wrap mktg-canvas">
   935→   935→      <TopNav onOpenExam={openExam} onExitAtlas={onExitAtlas} onJumpChapter={goToChapter} seen={seen} />
   936→   936→      <main className="pt-14">
   937→   937→        <PartBoundary label="Hero">
   938→   938→          <MktgHero
   939→   939→            activeId={lectureId}
   940→   940→            onSelect={selectLecture}
   941→   941→            onPractice={openHub}
   942→   942→            seen={seen}
   943→   943→            totalQuestions={MKTG_TOTAL_QUESTIONS}
   944→   944→          />
   945→   945→        </PartBoundary>
   946→   946→
   947→   947→        {/* the notes start here */}
   948→   948→        <div id="notes-top" aria-hidden className="relative scroll-mt-16" />
   949→   949→
   950→   950→        <PartBoundary key={lecture.id} label={`Lecture ${lecture.code} — ${lecture.title}`}>
   951→   951→          <motion.div
   952→   952→            initial={{ opacity: 0, y: 18 }}
   953→   953→            animate={{ opacity: 1, y: 0 }}
   954→   954→            transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
   955→   955→          >
   956→   956→            <MktgLectureHead lecture={lecture} />
   957→   957→            {lecture.chapters.length > 0 ? (
   958→   958→              lecture.chapters.map((c) => (
   959→   959→                <PartBoundary key={c.id} label={c.label}>
   960→   960→                  {SECTION_COMPONENTS[c.id]}
   961→   961→                </PartBoundary>
   962→   962→              ))
   963→   963→            ) : (
   964→   964→              <div className="mx-auto max-w-3xl px-4 py-16 sm:px-6">
   965→   965→                <div className="card-warm p-8 text-center">
   966→   966→                  <span className="chip mx-auto">
   967→   967→                    <BookOpen className="h-3.5 w-3.5" /> Slot reserved
   968→   968→                  </span>
   969→   969→                  <h2 className="mt-4 text-2xl font-extrabold text-[#1f2b2b]">
   970→   970→                    Lecture {lecture.n} hasn&rsquo;t been summarised yet
   971→   971→                  </h2>
   972→   972→                  <p className="mx-auto mt-3 max-w-md text-[.92rem] leading-relaxed text-[#5f7472]">
   973→   973→                    This lecture&rsquo;s notes haven&rsquo;t been uploaded, so the shelf keeps the
   974→   974→                    slot open. The moment its summary arrives, the chapters fill in right here.
   975→   975→                  </p>
   976→   976→                </div>
   977→   977→              </div>
   978→   978→            )}
   979→   979→            <LecturePager current={lecture} onJump={selectLecture} />
   980→   980→          </motion.div>
   981→   981→        </PartBoundary>
   982→   982→      </main>
   983→   983→      <MktgFooter />
   984→   984→      <PracticeHub open={hubOpen} onClose={() => setHubOpen(false)} onOpenExam={openExam} />
   985→   985→      {examGroup ? <MktgExamWindow group={examGroup} onClose={() => setExamGroup(null)} /> : null}
   986→   986→    </div>
   987→   987→  );
   988→   988→}
