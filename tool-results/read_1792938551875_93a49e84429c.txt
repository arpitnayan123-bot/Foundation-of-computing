   191→   191→   191→const PY_KEYWORDS = new Set([
   192→   192→   192→  "False","None","True","and","as","assert","break","class","continue","def",
   193→   193→   193→  "del","elif","else","except","finally","for","from","global","if","import",
   194→   194→   194→  "in","is","lambda","nonlocal","not","or","pass","raise","return","try",
   195→   195→   195→  "while","with","yield",
   196→   196→   196→]);
   197→   197→   197→const PY_BUILTINS = new Set([
   198→   198→   198→  "print","len","range","list","sorted","min","max","sum","int","str","float",
   199→   199→   199→  "input","append","insert","pop","remove","sort","reverse","title","upper",
   200→   200→   200→  "lower","strip","lstrip","rstrip","keys","values","items","get","set",
   201→   201→   201→  "dict","tuple","enumerate","zip","abs","round","type","clear","copy","count","index",
   202→   202→   202→]);
   203→   203→   203→
   204→   204→   204→type Tok = { text: string; cls: string };
   205→   205→   205→
   206→   206→   206→function tokenizePython(code: string): Tok[][] {
   207→   207→   207→  const pattern =
   208→   208→   208→    /(#.*$)|("""[\s\S]*?"""|'''[\s\S]*?'''|"(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*')|\b(\d[\d_]*\.?\d*)\b|([A-Za-z_][A-Za-z0-9_]*)|(\s+)|([^\sA-Za-z0-9_]+)/gm;
   209→   209→   209→  const lines: Tok[][] = [[]];
   210→   210→   210→  let m: RegExpExecArray | null;
   211→   211→   211→  while ((m = pattern.exec(code)) !== null) {
   212→   212→   212→    const [full, comment, str, num, word, ws, sym] = m;
   213→   213→   213→    if (comment !== undefined) push(comment, "tk-c");
   214→   214→   214→    else if (str !== undefined) push(str, "tk-s");
   215→   215→   215→    else if (num !== undefined) push(num, "tk-n");
   216→   216→   216→    else if (word !== undefined) {
   217→   217→   217→      if (PY_KEYWORDS.has(word)) push(word, "tk-k");
   218→   218→   218→      else if (PY_BUILTINS.has(word)) push(word, "tk-b");
   219→   219→   219→      else push(word, "");
   220→   220→   220→    } else if (ws !== undefined) push(ws, "ws");
   221→   221→   221→    else if (sym !== undefined) push(sym, "tk-d");
   222→   222→   222→  }
   223→   223→   223→  function push(text: string, cls: string) {
   224→   224→   224→    const parts = text.split("\n");
   225→   225→   225→    parts.forEach((p, i) => {
   226→   226→   226→      if (i > 0) lines.push([]);
   227→   227→   227→      if (p) lines[lines.length - 1].push({ text: p, cls: cls === "ws" ? "" : cls });
   228→   228→   228→    });
   229→   229→   229→  }
   230→   230→   230→  return lines;
   231→   231→   231→}

   261→   261→   261→   CodeBlock — syntax-highlighted snippet with an optional LIVE

   267→   267→   267→type RunPhase = "idle" | "running" | "done" | "error";
   268→   268→   268→
   269→   269→   269→export function CodeBlock({
   270→   270→   270→  title = "main.py",
   271→   271→   271→  code,
   272→   272→   272→  output,
   273→   273→   273→  note,
   274→   274→   274→  className = "",
   275→   275→   275→  runnable = true,
   276→   276→   276→  pre,
   277→   277→   277→  files,
   278→   278→   278→  dark = false,
   279→   279→   279→}: {
   280→   280→   280→  title?: string;
   281→   281→   281→  code: string;
   282→   282→   282→  output?: string;
   283→   283→   283→  note?: React.ReactNode;
   284→   284→   284→  className?: string;
   285→   285→   285→  runnable?: boolean;
   286→   286→   286→  pre?: string;
   287→   287→   287→  files?: Record<string, string>;
   288→   288→   288→  dark?: boolean;
   289→   289→   289→}) {
   290→   290→   290→  const [current, setCurrent] = useState(code);
   291→   291→   291→  const [editing, setEditing] = useState(false);
   292→   292→   292→  const [phase, setPhase] = useState<RunPhase>("idle");
   293→   293→   293→  const [bootMsg, setBootMsg] = useState("");
   294→   294→   294→  const [out, setOut] = useState("");
   295→   295→   295→  const [err, setErr] = useState("");
   296→   296→   296→  const [elapsed, setElapsed] = useState(0);
   297→   297→   297→  const [copied, setCopied] = useState(false);
   298→   298→   298→  const termRef = useRef<HTMLPreElement>(null);
   299→   299→   299→  const dirty = current !== code;
   300→   300→   300→  const live = phase !== "idle";
   301→   301→   301→  const busy = phase === "running";
   302→   302→   302→
   303→   303→   303→  useEffect(() => {
   304→   304→   304→    if (termRef.current) termRef.current.scrollTop = termRef.current.scrollHeight;
   305→   305→   305→  }, [out, err]);
   306→   306→   306→
   307→   307→   307→  async function run() {
   308→   308→   308→    if (busy) return;
   309→   309→   309→    setEditing(false);
   310→   310→   310→    setPhase("running");
   311→   311→   311→    setOut("");
   312→   312→   312→    setErr("");
   313→   313→   313→    setBootMsg("");
   314→   314→   314→    setElapsed(0);
   315→   315→   315→    const t0 = performance.now();
   316→   316→   316→    try {
   317→   317→   317→      // Global run lock: Pyodide exposes a single stdout/stderr/stdin handler
   318→   318→   318→      // set, so two simultaneous runners would steal each other's output
   319→   319→   319→      // (and the first finisher's cleanup would disconnect the second's
   320→   320→   320→      // capture mid-run). Queue behind whatever else is executing.
   321→   321→   321→      await withRunLock(async () => {
   322→   322→   322→        const pyr = await getPyodide((m) => setBootMsg(m));
   323→   323→   323→        setBootMsg("");
   324→   324→   324→        let lastOut = "";
   325→   325→   325→        // capped push: a print flood can't grow the buffer unbounded
   326→   326→   326→        const push = makeCappedPush((full) => {
   327→   327→   327→          lastOut = full;
   328→   328→   328→          setOut(full);
   329→   329→   329→        });
   330→   330→   330→        pyr.setStdout({ batched: push });
   331→   331→   331→        pyr.setStderr({ batched: push });
   332→   332→   332→        pyr.setStdin({
   333→   333→   333→          stdin: () => {
   334→   334→   334→            try {
   335→   335→   335→              return window.prompt("Python input() is asking for a value — cancel = stop the program:");
   336→   336→   336→            } catch {
   337→   337→   337→              return null; // prompt blocked (e.g. sandboxed iframe) → treat as EOF
   338→   338→   338→            }
   339→   339→   339→          },
   340→   340→   340→        });
   341→   341→   341→        try {
   342→   342→   342→          // Fresh module cache + virtual files for import demos.
   343→   343→   343→          await pyr.runPythonAsync(PURGE_USER_MODULES);
   344→   344→   344→          if (files) {
   345→   345→   345→            for (const [name, content] of Object.entries(files)) pyr.FS.writeFile(name, content);
   346→   346→   346→          }
   347→   347→   347→          const wf = current.match(/^%%writefile\s+([^\s#]+)/);
   348→   348→   348→          if (wf) {
   349→   349→   349→            // Notebook magic: save the cell as a real file inside the runtime.
   350→   350→   350→            const name = wf[1];
   351→   351→   351→            const nl = current.indexOf("\n");
   352→   352→   352→            const body = (nl === -1 ? "" : current.slice(nl + 1)).replace(/^\n+/, "");
   353→   353→   353→            pyr.FS.writeFile(name, body);
   354→   354→   354→            push(`✓ saved ${name} (${body.split("\n").length} lines) — import blocks can now use it`);
   355→   355→   355→          } else {
   356→   356→   356→            const ns = pyr.globals.get("dict")() as { destroy?: () => void };
   357→   357→   357→            try {
   358→   358→   358→              // step-guard: cooperative runaway-loop killer (see lib/pyguard.ts)
   359→   359→   359→              await withStepGuard(pyr, async () => {
   360→   360→   360→                if (pre) await pyr.runPythonAsync(pre, { globals: ns });
   361→   361→   361→                await pyr.runPythonAsync(current, { globals: ns });
   362→   362→   362→              });
   363→   363→   363→            } finally {
   364→   364→   364→              ns.destroy?.();
   365→   365→   365→            }
   366→   366→   366→          }
   367→   367→   367→        } finally {
   368→   368→   368→          // handlers live and die with THIS run — a queued runner can install
   369→   369→   369→          // its own the moment we yield the lock
   370→   370→   370→          try {
   371→   371→   371→            pyr.setStdout();
   372→   372→   372→            pyr.setStderr();
   373→   373→   373→            pyr.setStdin();
   374→   374→   374→          } catch {
   375→   375→   375→            /* already reset */
   376→   376→   376→          }
   377→   377→   377→        }
   378→   378→   378→      });
   379→   379→   379→      setElapsed((performance.now() - t0) / 1000);
   380→   380→   380→      setPhase("done");
   381→   381→   381→    } catch (e) {
   382→   382→   382→      const msg = e instanceof Error ? e.message : String(e);
   383→   383→   383→      setErr(cleanTraceback(msg) || msg);
   384→   384→   384→      setElapsed((performance.now() - t0) / 1000);
   385→   385→   385→      setPhase("error");
   386→   386→   386→    }
   387→   387→   387→  }
