    61→    61→const STACK_CAP = 6;

    63→    63→type Piece = { id: number; v: number };
    64→    64→
    65→    65→function StackWidget() {
    66→    66→  const [items, setItems] = useState<Piece[]>(STACK_SEQ.map((v, i) => ({ id: i + 1, v })));
    67→    67→  const [log, setLog] = useState("Push 64 → 32 → 53 → 66 → 54. Top = 54.");
    68→    68→  const nid = useRef(STACK_SEQ.length);
    69→    69→  const nseq = useRef(0);
    70→    70→
    71→    71→  function push() {
    72→    72→    if (items.length >= STACK_CAP) return setLog("Push() ✗ STACK OVERFLOW — the stack is FULL (6 of 6).");
    73→    73→    const v = STACK_SEQ[nseq.current % STACK_SEQ.length];
    74→    74→    nseq.current += 1; nid.current += 1;
    75→    75→    setItems((xs) => [...xs, { id: nid.current, v }]);
    76→    76→    setLog(`Push(${v}) → ${v} lands ON TOP. Top = ${v}.`);
    77→    77→  }
    78→    78→  function pop() {
    79→    79→    if (items.length === 0) return setLog("Pop() ✗ STACK UNDERFLOW — the stack is EMPTY, nothing to remove.");
    80→    80→    const t = items[items.length - 1];
    81→    81→    setItems((xs) => xs.slice(0, -1));
    82→    82→    setLog(`Pop() → ${t.v} lifts off. ${items.length - 1 > 0 ? `New top = ${items[items.length - 2].v}.` : "The stack is now empty."}`);
    83→    83→  }
    84→    84→  const peek = () => setLog(items.length ? `Peek() → ${items[items.length - 1].v} (reads the top, removes nothing)` : "Peek() → the stack is empty.");
    85→    85→  const size = () => setLog(`Size() → ${items.length}`);
    86→    86→  const reset = () => { setItems(STACK_SEQ.map((v, i) => ({ id: i + 1, v }))); nseq.current = 0; setLog("Reset — Push 64 → 32 → 53 → 66 → 54. Top = 54."); };
    87→    87→
    88→    88→  return (
    89→    89→    <div className="card-warm p-6 md:p-8">
    90→    90→      <div className="flex flex-wrap items-center gap-3">

   100→   100→            </div>

   102→   102→              <AnimatePresence initial={false}>

   172→   172→  const [log, setLog] = useState("Enqueue 64 → 32 → 53 → 66 → 54. Front = 64, Rear = 54.");
   173→   173→  const nid = useRef(Q_SEQ.length);
   174→   174→  const nseq = useRef(0);
   175→   175→
   176→   176→  function enqueue() {
   177→   177→    if (items.length >= Q_CAP) return setLog("Enqueue() ✗ QUEUE OVERFLOW — the queue is FULL (6 of 6).");
   178→   178→    const v = Q_SEQ[nseq.current % Q_SEQ.length];
   179→   179→    nseq.current += 1; nid.current += 1;
   180→   180→    setItems((xs) => [...xs, { id: nid.current, v }]);
   181→   181→    setLog(`Enqueue(${v}) → ${v} joins at the REAR.`);
   182→   182→  }
   183→   183→  function dequeue() {
   184→   184→    if (items.length === 0) return setLog("Dequeue() ✗ QUEUE UNDERFLOW — the queue is EMPTY, nobody to serve.");

   186→   186→    setItems((xs) => xs.slice(1));
   187→   187→    setLog(`Dequeue() → ${f.v} leaves from the FRONT. ${items.length - 1 > 0 ? `New front = ${items[1].v}.` : "The queue is now empty."}`);
   188→   188→  }
   189→   189→  const peek = () => setLog(items.length ? `Peek()/Front() → ${items[0].v} · Rear() → ${items[items.length - 1].v}` : "Peek() → the queue is empty.");
   190→   190→  const size = () => setLog(`Size() → ${items.length}`);
   191→   191→  const reset = () => { setItems(Q_SEQ.map((v, i) => ({ id: i + 1, v }))); nseq.current = 0; setLog("Reset — Enqueue 64 → 32 → 53 → 66 → 54. Front = 64, Rear = 54."); };
   192→   192→
   193→   193→  const front = items[0]?.v, rear = items[items.length - 1]?.v;
   194→   194→  return (
   195→   195→    <div className="card-warm p-6 md:p-8">

   204→   204→        <span className="rounded-full bg-[#fde8cf] px-2.5 py-1 font-mono-warm text-[.74rem] font-bold text-[#7c4a12]">Front → {front ?? "—"} · Rear → {rear ?? "—"} · Size {items.length}</span>
