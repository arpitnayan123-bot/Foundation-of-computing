     1→"use client";
     2→
     3→/* ============================================================
     4→   TEST MY KNOWLEDGE — THE ONE EXAM ENGINE.
     5→
     6→   All five shelves (FoC / ATB / FA / EBH / STM) used to carry a
     7→   550-line clone of this window each. Every difference between
     8→   them was pure presentation — colors, labels, copy — so the
     9→   engine now takes an `ExamTheme` and a plain bank object, and
    10→   each shelf ships a ~70-line wrapper that only declares its
    11→   theme. Logic (drafts, drift-proof clock, focus trap, scoring)
    12→   lives here and is shared.
    13→
    14→   Question shape is structural, not nominal: any bank whose
    15→   questions carry { id, type, q, options?, answer?, code?,
    16→   answerText?, solution?, explain? } plugs straight in.
    17→   ============================================================ */
    18→
    19→import React, { useCallback, useEffect, useMemo, useRef, useState } from "react";
    20→import { motion } from "framer-motion";
    21→import {
    22→  X,
    23→  ChevronLeft,
    24→  ChevronRight,
    25→  PenLine,
    26→  ListChecks,
    27→  RotateCcw,
    28→  Trophy,
    29→  Timer,
    30→  GraduationCap,
    31→  Check,
    32→  Eye,
    33→  Printer,
    34→} from "lucide-react";
    35→import {
    36→  clearExamDraft,
    37→  draftHasWork,
    38→  examDraftKey,
    39→  flushExamDraft,
    40→  loadExamDraft,
    41→  saveExamDraft,
    42→  useDialogA11y,
    43→} from "./exam-runtime";
    44→import { gradeQuestion } from "@/lib/sm2";
    45→import { logAttempt } from "@/lib/ledger";
    46→
    47→/* ---------------- theme contract ---------------- */
    48→
    49→export interface ExamTypeMeta {
    50→  label: string;
    51→  color: string;
    52→  bg: string;
    53→}
    54→
    55→export interface ExamTier {
    56→  t: string;

    58→}

    61→  /** draft namespace, e.g. "foc" — must stay stable per subject */

    63→  /** extra class on the shell for shelf-specific CSS, e.g. "stm-exam" */
    64→  rootClass?: string;
    65→  /** full-screen backdrop gradient */
    66→  pageBg: string;
    67→  /** header band gradient */
    68→  headerBg: string;
    69→  /** pen-line sigil gradient */
    70→  sigil: string;
    71→  /** bright text on the dark header */

    73→  /** muted subtitle on the dark header */

    76→  chipOnDark: string;
    77→  /** bottom bar border color */
    78→  footerBorder: string;
    79→  /** bottom bar background (alpha allowed) */
    80→  footerBg: string;
    81→  /** bottom-bar hint sentence */
    82→  footerNote: string;
    83→  /** label above the textarea */
    84→  paperLabel: string;
    85→  /** placeholder for write-the-output questions */
    86→  outputPlaceholder: string;
    87→  /** per-question-type chip colors + labels, in display order */
    88→  typeMeta: Record<string, ExamTypeMeta>;
    89→  /** result-report verdict tiers */
    90→  tiers: {
    91→    topper: ExamTier;
    92→    solid: ExamTier;
    93→    getting: ExamTier;
    94→    restart: ExamTier;
    95→  };
    96→}

    98→/* ---------------- structural bank contract ---------------- */
    99→
   100→export interface EngineQuestion {
   101→  id: string;
   102→  type: string;
   103→  q: string;
   104→  code?: string;
   105→  options?: string[];
   106→  answer?: number;
   107→  answerText?: string;
   108→  solution?: string;
   109→  explain?: string;
   110→}
   111→
   112→export interface EngineBank {
   113→  title: string;
   114→  subtitle: string;
   115→  questions: EngineQuestion[];
   116→}
   117→
   118→type QStatus = "unattempted" | "correct" | "wrong" | "revealed";
   119→interface QState {
   120→  chosen?: number;
   121→  typed?: string;
   122→  revealed?: boolean;
   123→  status: QStatus;
   124→}
   125→
   126→function fmt(sec: number) {
   127→  const m = Math.floor(sec / 60);
   128→  const s = sec % 60;
   129→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
   130→}
   131→
   132→export function ExamEngine({
   133→  theme,
   134→  bank,
   135→  draftId,
   136→  onClose,
   137→}: {
   138→  theme: ExamTheme;
   139→  bank: EngineBank;
   140→  /** stable per-window id (the exam group) — namespaces the draft */
   141→  draftId: string;
   142→  onClose: () => void;
   143→}) {
   144→  const qs = bank.questions;
   145→
   146→  /* ---- draft restore: a refresh or an accidental Escape keeps your work ---- */
   147→  const DRAFT_KEY = examDraftKey(theme.key, draftId);
   148→  const draft = useMemo(() => loadExamDraft(DRAFT_KEY), [DRAFT_KEY]);
   149→  const restored = useMemo(() => {
   150→    if (!draft) return null;
   151→    let n = 0;
   152→    for (const s of Object.values(draft.states)) {
   153→      const st = s as QState | undefined;
   154→      if (st && (st.status === "correct" || st.status === "wrong" || st.revealed)) n += 1;
   155→    }
   156→    return n > 0 ? n : null;
   157→  }, [draft]);
   158→
   159→  const [idx, setIdx] = useState(() => Math.min(Math.max(draft?.idx ?? 0, 0), qs.length - 1));
   160→  const [states, setStates] = useState<Record<string, QState>>(() => {
   161→    const fresh: Record<string, QState> = Object.fromEntries(
   162→      bank.questions.map((q) => [q.id, { status: "unattempted" as QStatus }])
   163→    );
   164→    if (draft) {
   165→      for (const q of bank.questions) {
   166→        const d = draft.states[q.id] as QState | undefined;
   167→        if (d && typeof d.status === "string") fresh[q.id] = d;
   168→      }
   169→    }
   170→    return fresh;
   171→  });
   172→  const [finished, setFinished] = useState(false);
   173→  /* drift-proof clock: seconds = base + (now − startedAt), derived from the
   174→     wall clock — never tick-counted, so background-tab throttling can't
   175→     silently under-report the elapsed time. Survives refresh via the draft. */
   176→  const [clock, setClock] = useState(() => ({
   177→    base: draft?.base ?? 0,
   178→    startedAt: draft?.startedAt ?? Date.now(),
   179→  }));
   180→  const [now, setNow] = useState(() => Date.now());
   181→  const seconds = Math.max(0, Math.floor(clock.base + (now - clock.startedAt) / 1000));
   182→  const bodyRef = useRef<HTMLDivElement>(null);
   183→  /* one ledger row per attempt-cycle — "back to questions" then finishing
   184→     again must not double-log; retake() re-arms it */
   185→  const loggedRef = useRef(false);
   186→
   187→  /* scroll lock (the timer below stops ticking on the results view) */
   188→  useEffect(() => {
   189→    document.body.style.overflow = "hidden";
   190→    return () => {
   191→      document.body.style.overflow = "";
   192→    };
   193→  }, []);
   194→
   195→  useEffect(() => {
   196→    if (finished) return;
   197→    const t = window.setInterval(() => setNow(Date.now()), 1000);
   198→    return () => window.clearInterval(t);
   199→  }, [finished]);
   200→
   201→  /* draft autosave — debounced inside saveExamDraft, flushed on close/finish */
   202→  useEffect(() => {
   203→    if (finished) return;
   204→    if (draftHasWork(states as Record<string, unknown>)) {
   205→      saveExamDraft(DRAFT_KEY, {
   206→        idx,
   207→        states: states as Record<string, unknown>,
   208→        base: clock.base,
   209→        startedAt: clock.startedAt,
   210→      });
   211→    }
   212→  }, [states, idx, clock, finished, DRAFT_KEY]);
   213→
   214→  const closeWindow = useCallback(() => {
   215→    flushExamDraft(); // the attempt is kept — reopening this window restores it
   216→    onClose();
   217→  }, [onClose]);
   218→
   219→  /** freeze the clock, clear the draft, log the attempt, show the report */
   220→  function finish() {
   221→    flushExamDraft();
   222→    clearExamDraft(DRAFT_KEY);
   223→    setClock({ base: seconds, startedAt: Date.now() });
   224→    setFinished(true);
   225→    if (!loggedRef.current) {
   226→      loggedRef.current = true;
   227→      const pct = Math.round((score.correct / score.total) * 100);
   228→      logAttempt({
   229→        subject: theme.key,
   230→        group: draftId,
   231→        title: bank.title,
   232→        correct: score.correct,
   233→        total: score.total,
   234→        pct,
   235→        seconds,
   236→      });
   237→    }
   238→  }
   239→
   240→  useEffect(() => {
   241→    const onKey = (e: KeyboardEvent) => {
   242→      const inTextarea = e.target instanceof HTMLTextAreaElement;
   243→      if (e.key === "Escape" && !inTextarea) closeWindow();
   244→      if (!finished && !inTextarea) {
   245→        if (e.key === "ArrowRight") setIdx((i) => Math.min(i + 1, qs.length - 1));
   246→        if (e.key === "ArrowLeft") setIdx((i) => Math.max(i - 1, 0));
   247→      }
   248→    };
   249→    document.addEventListener("keydown", onKey);
   250→    return () => document.removeEventListener("keydown", onKey);
   251→  }, [finished, closeWindow, qs.length]);
   252→
   253→  const q = qs[idx];
   254→  const st = states[q.id];
   255→
   256→  const patch = (id: string, p: Partial<QState>) =>
   257→    setStates((prev) => ({ ...prev, [id]: { ...prev[id], ...p } }));
   258→
   259→  /* scoring — byType rows follow the theme's declared type order */
   260→  const typeKeys = useMemo(() => Object.keys(theme.typeMeta), [theme]);
   261→  const score = useMemo(() => {
   262→    let correct = 0;
   263→    let wrong = 0;
   264→    let answered = 0;
   265→    const byType: Record<string, { c: number; t: number }> = {};
   266→    for (const t of typeKeys) byType[t] = { c: 0, t: 0 };
   267→    for (const qq of qs) {
   268→      const row = (byType[qq.type] ??= { c: 0, t: 0 });
   269→      row.t += 1;
   270→      const s = states[qq.id]?.status ?? "unattempted";
   271→      if (s === "correct" || s === "wrong") answered += 1;
   272→      if (s === "correct") {
   273→        correct += 1;
   274→        row.c += 1;
   275→      }
   276→      if (s === "wrong") wrong += 1;
   277→    }
   278→    return { correct, wrong, answered, byType, total: qs.length };
   279→  }, [states, qs, typeKeys]);
   280→
   281→  const pct = Math.round((score.correct / score.total) * 100);
   282→  const verdict =
   283→    pct >= 85 ? theme.tiers.topper : pct >= 60 ? theme.tiers.solid : pct >= 35 ? theme.tiers.getting : theme.tiers.restart;
   284→
   285→  function chooseMCQ(i: number) {
   286→    if (st.status === "correct" || st.status === "wrong") return;
   287→    const ok = i === q.answer;
   288→    patch(q.id, {
   289→      chosen: i,
   290→      revealed: true,
   291→      status: ok ? "correct" : "wrong",
   292→    });
   293→    // SM-2: first-try MCQ answers are trusted fully (5), misses reset the card (2)
   294→    gradeQuestion(theme.key, q.id, ok ? 5 : 2);
   295→  }
   296→
   297→  function selfCheck(ok: boolean) {
   298→    patch(q.id, { status: ok ? "correct" : "wrong" });
   299→    // SM-2: the model answer was visible — grade honestly but trust less
   300→    gradeQuestion(theme.key, q.id, ok ? 3 : 1);
   301→  }
   302→
   303→  function retake() {
   304→    setStates(Object.fromEntries(qs.map((qq) => [qq.id, { status: "unattempted" as QStatus }])));
   305→    setIdx(0);
   306→    setClock({ base: 0, startedAt: Date.now() });
   307→    setNow(Date.now());
   308→    setFinished(false);
   309→    loggedRef.current = false; // a retake is a new, honest attempt
   310→    clearExamDraft(DRAFT_KEY);
   311→    bodyRef.current?.scrollTo({ top: 0 });
   312→  }
   313→
   314→  function go(i: number) {
   315→    setIdx(i);
   316→    bodyRef.current?.scrollTo({ top: 0, behavior: "smooth" });
   317→  }
   318→
   319→  /* ---------------- results view ---------------- */
   320→  if (finished) {
   321→    const mcqMeta = theme.typeMeta.mcq ?? Object.values(theme.typeMeta)[0];
   322→    return (
   323→      <ExamShell
   324→        theme={theme}
   325→        bank={bank}
   326→        score={score}
   327→        seconds={seconds}
   328→        onClose={closeWindow}
   329→        restoredCount={restored}
   330→        onExit={() => setFinished(false)}
   331→        scrollRef={bodyRef}
   332→      >
   333→        <motion.div initial={{ opacity: 0, y: 18 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }}>
   334→          <div className="exam-result-card">
   335→            <div className="exam-score-ring" style={{ borderColor: pct >= 60 ? "#8fae5d" : "#e0a45e" }}>
   336→              <span className="exam-score-num">{pct}%</span>
   337→              <span className="exam-score-sub">
   338→                {score.correct} / {score.total}
   339→              </span>
   340→            </div>
   341→            <div>
   342→              <h3 className="font-display text-2xl font-bold text-[#2b1d10]">Result report</h3>
   343→              <p className="mt-1 text-[.95rem] font-semibold" style={{ color: verdict.c }} role="status">
   344→                {verdict.t}
   345→              </p>
   346→              <div className="mt-3 flex flex-wrap gap-2 text-[.72rem] font-bold">
   347→                <span className="exam-chip" style={{ background: "#eef4dd", color: "#44621d" }}>✓ {score.correct} correct</span>
   348→                <span className="exam-chip" style={{ background: "#fdecec", color: "#8c2f2f" }}>✗ {score.wrong} wrong</span>
   349→                <span className="exam-chip" style={{ background: "#f3e8d2", color: "#6f5438" }}>• {score.total - score.answered} unattempted</span>
   350→                <span className="exam-chip" style={{ background: mcqMeta.bg, color: mcqMeta.color }}>
   351→                  <Timer className="mr-1 inline h-3 w-3" />
   352→                  {fmt(seconds)}
   353→                </span>
   354→              </div>
   355→            </div>
   356→          </div>
   357→
   358→          {/* per-type breakdown */}
   359→          <div className="mt-5 grid gap-3 sm:grid-cols-2">
   360→            {typeKeys.map((t) => {
   361→              const meta = theme.typeMeta[t];
   362→              const d = score.byType[t];
   363→              if (!meta || !d) return null;
   364→              return (
   365→                <div key={t} className="flex items-center justify-between rounded-xl border border-[#e8dcc0] bg-[#fffdf9] px-4 py-3">
   366→                  <span className="text-[.8rem] font-bold" style={{ color: meta.color }}>
   367→                    {meta.label}
   368→                  </span>
   369→                  <span className="text-[.8rem] font-extrabold text-[#2b1d10]">
   370→                    {d.c} / {d.t}
   371→                  </span>
   372→                </div>
   373→              );
   374→            })}
   375→          </div>
   376→
   377→          {/* review the misses */}
   378→          {score.correct < score.total ? (
   379→            <div className="mt-6">
   380→              <p className="exam-eyebrow">
   381→                <Printer className="h-3.5 w-3.5" />
   382→                Review &amp; rewrite these by hand
   383→              </p>
   384→              <div className="mt-3 grid gap-3">
   385→                {qs.map((qq) => {
   386→                  const s = states[qq.id]?.status ?? "unattempted";
   387→                  if (s === "correct") return null;
   388→                  const model =
   389→                    qq.type === "mcq"
   390→                      ? (qq.options?.[qq.answer ?? 0] ?? "")
   391→                      : qq.answerText ?? qq.solution ?? "";
   392→                  return (
   393→                    <div key={qq.id} className="rounded-xl border border-[#e8dcc0] bg-[#fffdf9] p-4">
   394→                      <p className="text-[.88rem] font-bold leading-snug text-[#2b1d10]">{qq.q}</p>
   395→                      {qq.code ? <pre className="exam-code mt-2">{qq.code}</pre> : null}
   396→                      <p className="mt-2 text-[.82rem] leading-relaxed" style={{ color: s === "wrong" ? "#8c2f2f" : "#6f5438" }}>
   397→                        <strong>{s === "wrong" ? "Missed —" : "Unattempted —"}</strong>{" "}
   398→                        <span className="whitespace-pre-wrap font-mono text-[.78rem] text-[#2f2013]">{model}</span>
   399→                      </p>
   400→                      {qq.explain ? <p className="mt-1 text-[.78rem] leading-relaxed text-[#6f5438]">{qq.explain}</p> : null}
   401→                    </div>
   402→                  );
   403→                })}
   404→              </div>
   405→            </div>
   406→          ) : null}
   407→
   408→          <div className="mt-7 flex flex-wrap justify-center gap-3">
   409→            <button type="button" onClick={retake} className="exam-btn exam-btn-primary">
   410→              <RotateCcw className="h-4 w-4" />
   411→              Retake this test
   412→            </button>
   413→            <button type="button" onClick={() => setFinished(false)} className="exam-btn">
   414→              <ListChecks className="h-4 w-4" />
   415→              Back to questions
   416→            </button>
   417→          </div>
   418→        </motion.div>
   419→      </ExamShell>
   420→    );
   421→  }
   422→
   423→  /* ---------------- question view ---------------- */
   424→  const meta = theme.typeMeta[q.type] ?? theme.typeMeta.mcq ?? Object.values(theme.typeMeta)[0];
   425→  const mcqLocked = st.status === "correct" || st.status === "wrong";
   426→
   427→  return (
   428→    <ExamShell
   429→      theme={theme}
   430→      bank={bank}
   431→      score={score}
   432→      seconds={seconds}
   433→      onClose={closeWindow}
   434→      restoredCount={restored}
   435→      onExit={() => finish()}
   436→      scrollRef={bodyRef}
   437→    >
   438→      <motion.div key={q.id} initial={{ opacity: 0, x: 24 }} animate={{ opacity: 1, x: 0 }} transition={{ duration: 0.22, ease: "easeOut" }}>
   439→        {/* type chip + counter */}
   440→        <div className="flex flex-wrap items-center gap-2">
   441→          <span className="exam-type-chip" style={{ color: meta.color, background: meta.bg }}>
   442→            {meta.label}
   443→          </span>
   444→          <span className="text-[.72rem] font-extrabold uppercase tracking-[.16em] text-[#8a6d4f]">
   445→            Question {idx + 1} of {qs.length}
   446→          </span>
   447→        </div>
   448→
   449→        {/* question */}
   450→        <h3 className="font-display mt-3 text-xl font-bold leading-snug text-[#2b1d10] md:text-[1.35rem]">{q.q}</h3>
   451→
   452→        {q.code ? <pre className="exam-code mt-4">{q.code}</pre> : null}
   453→
   454→        {/* answer area */}
   455→        {q.type === "mcq" ? (
   456→          <div className="mt-4 grid gap-2">
   457→            {q.options?.map((opt, i) => {
   458→              const isAnswer = i === q.answer;
   459→              const chosen = st.chosen === i;
   460→              let cls = "exam-opt";
   461→              if (mcqLocked && isAnswer) cls += " exam-opt-correct";
   462→              else if (mcqLocked && chosen) cls += " exam-opt-wrong";
   463→              return (
   464→                <button key={i} type="button" disabled={mcqLocked} onClick={() => chooseMCQ(i)} className={cls}>
   465→                  <span className="exam-opt-letter">{String.fromCharCode(65 + i)}</span>
   466→                  <span className="font-mono text-[.86rem]">{opt}</span>
   467→                  {mcqLocked && isAnswer ? <Check className="ml-auto h-4 w-4" /> : null}
   468→                </button>
   469→              );
   470→            })}
   471→          </div>
   472→        ) : (
   473→          <div className="mt-4">
   474→            <label htmlFor={`ta-${q.id}`} className="text-[.74rem] font-bold uppercase tracking-[.14em] text-[#8a6d4f]">
   475→              {theme.paperLabel}
   476→            </label>
   477→            <textarea
   478→              id={`ta-${q.id}`}
   479→              value={st.typed ?? ""}
   480→              onChange={(e) => patch(q.id, { typed: e.target.value })}
   481→              spellCheck={false}
   482→              rows={q.type === "short" ? 4 : 6}
   483→              placeholder={q.type === "output" ? theme.outputPlaceholder : q.type === "code" ? "# write your program here…" : "your answer…"}
   484→              className="exam-ta"
   485→            />
   486→            {!st.revealed ? (
   487→              <button type="button" onClick={() => patch(q.id, { revealed: true })} className="exam-btn mt-2">
   488→                <Eye className="h-4 w-4" />
   489→                Reveal the model answer
   490→              </button>
   491→            ) : (
   492→              <motion.div
   493→                initial={{ opacity: 0, y: 10 }}
   494→                animate={{ opacity: 1, y: 0 }}
   495→                className="mt-3"
   496→                role="status"
   497→              >
   498→                <div className="exam-answer">
   499→                  <p className="exam-eyebrow">
   500→                    <GraduationCap className="h-3.5 w-3.5" />
   501→                    Model answer
   502→                  </p>
   503→                  {q.solution ? <pre className="exam-code mt-2">{q.solution}</pre> : null}
   504→                  {q.answerText ? (
   505→                    <p className={`mt-2 whitespace-pre-wrap font-mono text-[.82rem] leading-relaxed text-[#2f2013] ${q.solution ? "text-[.78rem]" : ""}`}>
   506→                      {q.answerText}
   507→                    </p>
   508→                  ) : null}
   509→                  {q.explain ? <p className="mt-2 text-[.82rem] leading-relaxed text-[#6f5438]">{q.explain}</p> : null}
   510→                </div>
   511→                {st.status !== "correct" && st.status !== "wrong" ? (
   512→                  <div className="mt-3 flex flex-wrap items-center gap-2">
   513→                    <span className="text-[.74rem] font-bold uppercase tracking-[.12em] text-[#8a6d4f]">Self-check:</span>
   514→                    <button type="button" onClick={() => selfCheck(true)} className="exam-btn" style={{ color: "#44621d" }}>
   515→                      <Check className="h-4 w-4" /> I got it
   516→                    </button>
   517→                    <button type="button" onClick={() => selfCheck(false)} className="exam-btn" style={{ color: "#8c2f2f" }}>
   518→                      <X className="h-4 w-4" /> Not yet
   519→                    </button>
   520→                  </div>
   521→                ) : null}
   522→              </motion.div>
   523→            )}
   524→          </div>
   525→        )}
   526→
   527→        {/* MCQ explanation */}
   528→        {q.type === "mcq" && mcqLocked && q.explain ? (
   529→          <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} className="exam-answer mt-3" role="status">
   530→            <p className="text-[.88rem] font-bold" style={{ color: st.status === "correct" ? "#44621d" : "#8c2f2f" }}>
   531→              {st.status === "correct" ? "Correct." : "Not quite."}
   532→            </p>
   533→            <p className="mt-1 text-[.85rem] leading-relaxed text-[#2f2013]">{q.explain}</p>
   534→          </motion.div>
   535→        ) : null}
   536→
   537→        {/* navigator */}
   538→        <div className="exam-nav">
   539→          <p className="exam-eyebrow">
   540→            <ListChecks className="h-3.5 w-3.5" />
   541→            Jump to a question
   542→          </p>
   543→          <div className="mt-2 flex flex-wrap gap-1.5">
   544→            {qs.map((qq, i) => {
   545→              const s = states[qq.id]?.status ?? "unattempted";
   546→              const cls =
   547→                i === idx
   548→                  ? "exam-dot exam-dot-current"
   549→                  : s === "correct"
   550→                  ? "exam-dot exam-dot-ok"
   551→                  : s === "wrong"
   552→                  ? "exam-dot exam-dot-bad"
   553→                  : s === "revealed"
   554→                  ? "exam-dot exam-dot-rev"
   555→                  : "exam-dot";
   556→              return (
   557→                <button key={qq.id} type="button" onClick={() => go(i)} className={cls} aria-label={`Go to question ${i + 1}`}>
   558→                  {i + 1}
   559→                </button>
   560→              );
   561→            })}
   562→          </div>
   563→        </div>
   564→
   565→        {/* prev / next */}
   566→        <div className="mt-6 flex items-center justify-between pb-2">
   567→          <button type="button" onClick={() => go(Math.max(0, idx - 1))} disabled={idx === 0} className="exam-btn">
   568→            <ChevronLeft className="h-4 w-4" />
   569→            Prev
   570→          </button>
   571→          {idx < qs.length - 1 ? (
   572→            <button type="button" onClick={() => go(idx + 1)} className="exam-btn exam-btn-primary">
   573→              Next
   574→              <ChevronRight className="h-4 w-4" />
   575→            </button>
   576→          ) : (
   577→            <button type="button" onClick={() => finish()} className="exam-btn exam-btn-primary">
   578→              <Trophy className="h-4 w-4" />
   579→              Finish &amp; see score
   580→            </button>
   581→          )}
   582→        </div>
   583→      </motion.div>
   584→    </ExamShell>
   585→  );
   586→}
