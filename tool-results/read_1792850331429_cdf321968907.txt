    76→function fmtSeconds(sec: number): string {
    77→  const m = Math.floor(sec / 60);
    78→  const s = sec % 60;
    79→  return `${m}m ${String(s).padStart(2, "0")}s`;
    80→}
    81→
    82→function relDay(at: number): string {
    83→  const now = new Date();
    84→  const then = new Date(at);
    85→  const days = Math.floor((now.setHours(0, 0, 0, 0) - new Date(then).setHours(0, 0, 0, 0)) / 86_400_000);
    86→  if (days <= 0) return "today";
    87→  if (days === 1) return "yesterday";
    88→  if (days < 7) return `${days}d ago`;
    89→  return then.toLocaleDateString(undefined, { month: "short", day: "numeric" });
    90→}

    92→export function LedgerView({ onExitAtlas }: { onExitAtlas: () => void }) {
    93→  const [store, setStore] = useState<LedgerStore | null>(null);
    94→  const [rows, setRows] = useState<ShelfRow[]>([]);
    95→  const [streak, setStreak] = useState(0);
    96→  /* the weakest cards in the schedule — shown as "trouble spots";
    97→     their question wording arrives from the banks a moment later */
    98→  const [trouble, setTrouble] = useState<TroubleSpot[]>([]);
    99→  const [troubleTexts, setTroubleTexts] = useState<Map<string, string> | null>(null);
   100→  /* the ⌘K palette may have queued a review handoff ("review <subject>")
   101→     before this view mounted — honor it once at init (client-only chunk) */
   102→  const [reviewSubject, setReviewSubject] = useState<string | null>(() => {
   103→    try {
   104→      const queued = sessionStorage.getItem("atlas:review-subject");
   105→      if (queued) {
   106→        sessionStorage.removeItem("atlas:review-subject");
   107→        return queued;
   108→      }
   109→    } catch {
   110→      /* private mode — no handoff */
   111→    }
   112→    return null;
   113→  });
   114→  /* trouble-spot reviews pull their specific card forward even when
   115→     it is not due until tomorrow — everything else reviews by due date */

   117→  const openReview = useCallback((subject: string, force: string[] = []) => {
   118→    setReviewForce(force);
   119→    setReviewSubject(subject);
   120→  }, []);
   121→
   122→  const refresh = useCallback(() => {
   123→    const ledger = loadLedger();

   127→    setStreak(currentStreak(ledger.days));

   131→        subject,
   132→        stats: statsFor(ledger, subject.id),
   133→        mastery: masteryFor(subject.id, QUESTION_COUNTS[subject.id as keyof typeof QUESTION_COUNTS] ?? 0),
   134→        due: due[subject.id] ?? 0,
   135→        read: reads[subject.id] ?? 0,
   136→        readTotal: SECTION_COUNTS[subject.id] ?? subject.chapters ?? 0,
   137→      }))
   138→    );
   139→  }, []);
   140→
   141→  useEffect(() => {
   142→    /* rAF-deferred first read — avoids sync setState-in-effect */
   143→    const raf = requestAnimationFrame(() => refresh());
   144→    const onVisible = () => {
   145→      if (document.visibilityState === "visible") refresh();
   146→    };
   147→    /* live handoff — the palette fires this when the ledger is already open */
   148→    const onReview = (e: Event) => {
   149→      const id = (e as CustomEvent<string>).detail;
   150→      if (typeof id === "string") {
   151→        setReviewForce([]);
   152→        setReviewSubject(id);
   153→        refresh();
   154→      }
   155→    };
   156→    document.addEventListener("visibilitychange", onVisible);
   157→    window.addEventListener("atlas:review-subject", onReview);
   158→    return () => {
   159→      cancelAnimationFrame(raf);
   160→      document.removeEventListener("visibilitychange", onVisible);
   161→      window.removeEventListener("atlas:review-subject", onReview);
   162→    };
   163→  }, [refresh]);
   164→
   165→  const totalAttempts = store?.attempts.length ?? 0;
   166→  const totalAnswered = store?.attempts.reduce((n, a) => n + a.total, 0) ?? 0;
   167→  const daysPracticed = store?.days.length ?? 0;
   168→  const recent = (store?.attempts ?? []).slice(-10).reverse();
   169→  const dueEverywhere = rows.reduce((n, r) => n + r.due, 0);
   170→  const readEverywhere = rows.reduce((n, r) => n + r.read, 0);
   171→
   172→  /* per-day attempt counts behind the practice heatmap */
   173→  const perDay = useMemo(() => {
   174→    const m = new Map<string, DayHit>();
   175→    for (const a of store?.attempts ?? []) {
   176→      const k = dayKeyFor(a.at);
   177→      const cur = m.get(k) ?? { a: 0, q: 0 };
   178→      cur.a += 1;
   179→      cur.q += a.total;
   180→      m.set(k, cur);
   181→    }
   182→    return m;
   183→  }, [store]);
   184→
   185→  /* weak cards exist → pull their wording from the banks once,
   186→     lazily (the ~250KB of bank chunks stay out of the ledger path
   187→     until there is something to say) */
   188→  useEffect(() => {
   189→    if (trouble.length === 0 || troubleTexts) return;
   190→    let alive = true;
   191→    void loadQuestionIndex().then((idx) => {
   192→      if (alive) setTroubleTexts(idx);
   193→    });
   194→    return () => {
   195→      alive = false;
   196→    };
   197→  }, [trouble, troubleTexts]);
   198→
   199→  /* ---------- your-data panel state ---------- */
   200→  const [imp, setImp] = useState<ImportState>({ kind: "idle" });
   201→  const [dataFlash, setDataFlash] = useState<string | null>(null);
   202→  const [eraseArm, setEraseArm] = useState(false);
   203→  const fileRef = useRef<HTMLInputElement>(null);
   204→
   205→  const onExport = () => {
   206→    try {
   207→      const blob = new Blob([exportBackup()], { type: "application/json" });
   208→      const url = URL.createObjectURL(blob);
   209→      const a = document.createElement("a");
   210→      a.href = url;
   211→      a.download = backupFilename();
   212→      a.click();
   213→      URL.revokeObjectURL(url);
   214→      setDataFlash(`Backup downloaded — ${backupFilename()}`);
   215→    } catch {
   216→      setDataFlash("The download failed — your browser blocked it. Try again.");
   217→    }
   218→  };
   219→
   220→  const onFile = async (file: File | undefined) => {
   221→    if (!file) return;
   222→    try {
   223→      const text = await file.text();

   225→      if (res.ok) {
   226→        setImp({ kind: "ready", name: file.name, summary: res.summary, bundle: res.bundle });

   228→        setImp({ kind: "error", message: res.error });
   229→      }
   230→    } catch {
   231→      setImp({ kind: "error", message: "That file could not be read." });
   232→    }
   233→  };

   235→  const onImportApply = () => {
   236→    if (imp.kind !== "ready" || !imp.bundle || !imp.summary) return;

   239→    setImp({ kind: "idle" });

   243→  };
   244→
   245→  const onErase = () => {

   247→      setEraseArm(true);
   248→      window.setTimeout(() => setEraseArm(false), 4000);
   249→      return;
   250→    }
   251→    eraseAllStores();
   252→    setEraseArm(false);
   253→    setImp({ kind: "idle" });
   254→    if (fileRef.current) fileRef.current.value = "";
   255→    refresh();
   256→    setDataFlash("Every store erased — a fresh start. Export a backup first if you want this memory back.");
   257→  };
   258→
   259→  return (
   260→    <div className="min-h-screen bg-[#070509] pb-20 text-[#f7ecdd]" id="top">

   263→        <div className="flex h-12 w-full max-w-3xl items-center gap-3 rounded-full border border-white/10 bg-[#0a0710]/60 py-2 pl-3 pr-4 shadow-[0_18px_50px_-22px_rgba(0,0,0,.85)] backdrop-blur-xl">
   264→          <button
   265→            type="button"
   266→            onClick={onExitAtlas}
   267→            aria-label="Back to the atlas"
   268→            className="flex h-8 items-center gap-1.5 rounded-full border border-white/10 bg-white/[.05] px-3 text-[.68rem] font-black uppercase tracking-[.16em] text-[#e9d7bf] transition hover:bg-white/[.12]"
   269→          >
   270→            <ArrowLeft className="h-3.5 w-3.5" />
   271→            Atlas
   272→          </button>
   273→          <span className="font-display text-[1rem] font-black tracking-tight text-[#f7ecdd]">The Ledger</span>
   274→          {dueEverywhere > 0 ? (
   275→            <button
   276→              type="button"
   277→              onClick={() => openReview("all")}
   278→              aria-label={`Review all ${dueEverywhere} due cards across every shelf`}
   279→              className="ml-auto flex cursor-pointer items-center gap-1.5 rounded-full border border-[#f59e0b40] bg-[#f59e0b14] px-3 py-1 font-mono text-[.58rem] font-bold uppercase tracking-[.16em] text-[#f5c889] transition hover:bg-[#f59e0b2a]"
   280→            >
   281→              <Sparkles className="h-3 w-3" />
   282→              {dueEverywhere} due
   283→            </button>
   284→          ) : null}
   285→        </div>
   286→      </nav>
   287→
   288→      <main className="mx-auto max-w-3xl px-4 pt-24 sm:px-6">
   289→        {/* ------- hero ------- */}
   290→        <motion.header initial={{ opacity: 0, y: 14 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.35 }}>
   291→          <p className="font-mono text-[.6rem] font-bold uppercase tracking-[.24em] text-[#8d7b63]">

   295→            Every answer leaves a <span className="atlas-sheen">mark</span>.
   296→          </h1>
   297→          <p className="mt-3 max-w-xl text-[.92rem] leading-relaxed text-[#c9b99f]">
   298→            The ledger is local-first: exam windows grade your cards on an SM-2 schedule, and this page reports the

   305→          <StatCard icon={Flame} label="streak" value={streak === 1 ? "1 day" : `${streak} days`} tone="#e4674a" />

   322→        {/* ------- per-shelf mastery ------- */}

   325→            <h2 className="font-display text-xl font-black">Shelf by shelf</h2>

   327→              <strong className="text-[#c9b99f]">{readEverywhere}</strong>/{TOTAL_SECTIONS} chapters read
   328→            </span>
   329→            {dueEverywhere > 0 ? (
   330→              <button

   332→                onClick={() => openReview("all")}

   335→                <Layers className="h-3.5 w-3.5" />
   336→                review everything
