    55→const colorCache = new Map<string, number[]>()
    56→function parseColor(c: string): number[] {
    57→    const key = c || ""
    58→    const hit = colorCache.get(key)
    59→    if (hit) return hit
    60→    let out = [0, 0, 0, 1]
    61→    const s = key.trim()
    62→    if (s[0] === "#") {
    63→        let h = s.slice(1)
    64→        if (h.length === 3 || h.length === 4)
    65→            h = h
    66→                .split("")
    67→                .map((x) => x + x)
    68→                .join("")
    69→        const n = parseInt(h.slice(0, 6), 16)
    70→        const a = h.length === 8 ? parseInt(h.slice(6, 8), 16) / 255 : 1
    71→        if (!isNaN(n)) out = [(n >> 16) & 255, (n >> 8) & 255, n & 255, a]
    72→    } else {
    73→        const m = s.match(/rgba?\(([^)]+)\)/i)
    74→        if (m) {
    75→            const p = m[1]
    76→                .split(/[\s,\/]+/)
    77→                .filter(Boolean)
    78→                .map(parseFloat)
    79→            out = [p[0] || 0, p[1] || 0, p[2] || 0, p[3] === undefined || isNaN(p[3]) ? 1 : p[3]]
    80→        }
    81→    }
    82→    if (colorCache.size > 256) colorCache.clear()
    83→    colorCache.set(key, out)
    84→    return out
    85→}
    86→function rgba(c: string, a: number) {
    87→    const p = parseColor(c)
    88→    return `rgba(${p[0] | 0},${p[1] | 0},${p[2] | 0},${Math.max(0, Math.min(1, p[3] * a))})`
    89→}
    90→function mix(a: string, b: string, t: number) {
    91→    const p = parseColor(a)
    92→    const q = parseColor(b)
    93→    const r = p[0] + (q[0] - p[0]) * t
    94→    const g = p[1] + (q[1] - p[1]) * t
    95→    const bl = p[2] + (q[2] - p[2]) * t
    96→    return `rgb(${r | 0},${g | 0},${bl | 0})`
    97→}

   101→function mulberry32(a: number) {
   102→    return function () {
   103→        a |= 0
   104→        a = (a + 0x6d2b79f5) | 0
   105→        let t = Math.imul(a ^ (a >>> 15), 1 | a)
   106→        t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
   107→        return ((t ^ (t >>> 14)) >>> 0) / 4294967296
   108→    }
   109→}

   129→    const rm = mulberry32(seed * 103 + 2)

   132→        mid.push({ x: ((i + 0.15 + rm() * 0.7) / midN) * 900, h: 70 + rm() * 60, base: 271 + rm() * 5, phase: rm() * TAU })
   133→    const rn = mulberry32(seed * 107 + 3)

   180→function each(x0: number, off: number, P: number, L: number, R: number, m: number, fn: (x: number) => void) {
   181→    if (!(P > 0)) return
   182→    const b = (((x0 + off) % P) + P) % P
   183→    let x = b + Math.ceil((L - m - b) / P) * P
   184→    for (let g = 0; x <= R + m && g < 64; x += P, g++) fn(x)
   185→}

   187→function drawTurbine(
   188→    ctx: CanvasRenderingContext2D,
   189→    x: number,
   190→    base: number,
   191→    h: number,
   192→    blade: number,
   193→    ang: number,
   194→    tw: number
   195→) {
   196→    const hy = base - h
   197→    ctx.beginPath()
   198→    ctx.moveTo(x - tw * 0.5, base)
   199→    ctx.lineTo(x - tw * 0.3, hy)
   200→    ctx.lineTo(x + tw * 0.3, hy)
   201→    ctx.lineTo(x + tw * 0.5, base)
   202→    ctx.closePath()
   203→    ctx.fill()
   204→    ctx.beginPath()
   205→    ctx.ellipse(x + tw * 0.5, hy, tw * 1.1, tw * 0.55, 0, 0, TAU)
   206→    ctx.fill()
   207→    const rw = Math.max(0.7, blade * 0.035)
   208→    ctx.beginPath()
   209→    for (let k = 0; k < 3; k++) {
   210→        const a = ang + (k * TAU) / 3
   211→        const dx = Math.sin(a)
   212→        const dy = -Math.cos(a)
   213→        const nx = Math.cos(a)
   214→        const ny = Math.sin(a)
   215→        ctx.moveTo(x + nx * rw * 0.45, hy + ny * rw * 0.45)
   216→        ctx.lineTo(x + dx * blade * 0.2 + nx * rw, hy + dy * blade * 0.2 + ny * rw)
   217→        ctx.lineTo(x + dx * blade, hy + dy * blade)
   218→        ctx.lineTo(x + dx * blade * 0.25 - nx * rw * 0.35, hy + dy * blade * 0.25 - ny * rw * 0.35)
   219→        ctx.lineTo(x - nx * rw * 0.45, hy - ny * rw * 0.45)
   220→        ctx.closePath()
   221→    }
   222→    ctx.fill()
   223→    ctx.beginPath()
   224→    ctx.arc(x, hy, Math.max(0.6, tw * 0.75), 0, TAU)
   225→    ctx.fill()
   226→}

   228→function bakeSky(
   229→    c: HTMLCanvasElement,
   230→    cw: number,
   231→    ch: number,
   232→    dpr: number,
   233→    s: number,
   234→    W: number,
   235→    H: number,
   236→    p: any,
   237→    adv: typeof ADVANCED_DEFAULTS,
   238→    scene: Scene
   239→) {
   240→    c.width = cw
   241→    c.height = ch
   242→    const g = c.getContext("2d")
   243→    if (!g) return
   244→    g.setTransform(1, 0, 0, 1, 0, 0)
   245→    g.clearRect(0, 0, cw, ch)
   246→    g.setTransform(dpr * s, 0, 0, dpr * s, dpr * (W / 2 - CX * s), dpr * (H / 2 - CY * s))
   247→    const L = CX - W / (2 * s) - 4
   248→    const R = CX + W / (2 * s) + 4
   249→    const T = CY - H / (2 * s) - 4
   250→    const top = p.skyTop || "#451922"
   251→    const hor = p.skyHorizon || "#EF6759"
   252→
   253→    const sky = g.createLinearGradient(0, 0, 0, HORIZON)
   254→    // ATLAS retune: the ramp derives from the two sky props (instead of
   255→    // hardcoded crimson stops) so the scene can be re-poured in any dusk.
   256→    sky.addColorStop(0, top)
   257→    sky.addColorStop(0.12, mix(top, hor, 0.22))
   258→    sky.addColorStop(0.3, mix(top, hor, 0.5))
   259→    sky.addColorStop(0.5, mix(hor, top, 0.16))
   260→    sky.addColorStop(0.7, hor)
   261→    sky.addColorStop(0.86, mix(hor, top, 0.34))
   262→    sky.addColorStop(1, mix(hor, top, 0.56))
   263→    g.fillStyle = sky
   264→    g.fillRect(L, T, R - L, HORIZON + 4 - T)
   265→
   266→    const sx = (clamp(adv.sunX, 0, 100) / 100) * REF_W
   267→    const sy = (clamp(adv.sunY, 0, 100) / 100) * REF_H
   268→    const sr = Math.max(4, adv.sunSize / 2)
   269→
   270→    const warm = g.createRadialGradient(sx, sy, 0, sx, sy, 300)
   271→    warm.addColorStop(0, rgba(hor, 0.55))
   272→    warm.addColorStop(1, rgba(hor, 0))
   273→    g.fillStyle = warm
   274→    g.fillRect(L, T, R - L, HORIZON + 4 - T)
   275→
   276→    const co = clamp(adv.cloudOpacity, 0, 100) / 100
   277→    const cloudC = adv.cloudColor
   278→    g.filter = `blur(${Math.max(1, 2.4 * s * dpr)}px)`
   279→    for (const cl of scene.clouds) {
   280→        const col = cl.tone === 1 ? mix(cloudC, "#3C1420", 0.55) : cl.tone === 2 ? mix(cloudC, hor, 0.55) : cloudC
   281→        g.fillStyle = rgba(col, cl.a * co)
   282→        g.beginPath()
   283→        g.ellipse(cl.x, cl.y, cl.w / 2, Math.max(0.5, cl.h / 2), 0, 0, TAU)
   284→        g.fill()
   285→    }
   286→    g.filter = `blur(${Math.max(0.5, 0.6 * s * dpr)}px)`
   287→    for (const cl of scene.clouds) {
   288→        if (cl.tone === 2) continue
   289→        g.fillStyle = rgba(adv.rimColor, 0.45 * co)
   290→        g.beginPath()
   291→        g.ellipse(cl.x - cl.w * 0.05, cl.y + cl.h * 0.45, cl.w * 0.38, Math.max(0.4, cl.h * 0.07 + 0.4), 0, 0, TAU)
   292→        g.fill()
   293→    }
   294→
   295→    g.filter = `blur(${Math.max(0.5, 1.2 * s * dpr)}px)`
   296→    g.fillStyle = rgba("#844E75", 0.6)
   297→    g.beginPath()
   298→    g.moveTo(L - 60, HORIZON + 2)
   299→    g.lineTo(L - 60, 238)
   300→    g.quadraticCurveTo(-30, 226, 20, 230)
   301→    g.quadraticCurveTo(70, 234, 120, 222)
   302→    g.quadraticCurveTo(170, 230, 205, 248)
   303→    g.quadraticCurveTo(225, 260, 250, HORIZON + 2)
   304→    g.closePath()
   305→    g.fill()
   306→    g.beginPath()
   307→    g.moveTo(630, HORIZON + 2)
   308→    g.quadraticCurveTo(680, 250, 725, 246)
   309→    g.quadraticCurveTo(780, 248, R + 60, 254)
   310→    g.lineTo(R + 60, HORIZON + 2)
   311→    g.closePath()
   312→    g.fill()
   313→    g.filter = "none"
   314→
   315→    const glow = clamp(adv.sunGlow, 0, 100) / 100
   316→    const sg = g.createRadialGradient(sx, sy, sr * 0.9, sx, sy, sr + 46)
   317→    sg.addColorStop(0, rgba("#F8C4B1", glow * 0.9))
   318→    sg.addColorStop(0.35, rgba(hor, glow))
   319→    sg.addColorStop(1, rgba(hor, 0))
   320→    g.fillStyle = sg
   321→    g.beginPath()
   322→    g.arc(sx, sy, sr + 46, 0, TAU)
   323→    g.fill()
   324→    const sun = p.sunColor || "#F4F7F6"
   325→    const disc = g.createRadialGradient(sx, sy, 0, sx, sy, sr)
   326→    disc.addColorStop(0, sun)
   327→    disc.addColorStop(0.7, sun)
   328→    disc.addColorStop(1, mix(sun, "#F8C4B1", 0.85))
   329→    g.fillStyle = disc
   330→    g.beginPath()
   331→    g.arc(sx, sy, sr, 0, TAU)
   332→    g.fill()
   333→
   334→    const k = sr / 65
   335→    g.save()
   336→    g.beginPath()
   337→    g.arc(sx, sy, sr + 0.5, 0, TAU)
   338→    g.clip()
   339→    g.filter = `blur(${Math.max(0.4, 0.8 * s * dpr)}px)`
   340→    g.fillStyle = rgba(mix(hor, "#CA424A", 0.45), 0.95)
   341→    const bars = [
   342→        [-32, 6, 0.75],
   343→        [5, 4, -0.85],
   344→        [16, 8, -0.75],
   345→    ]
   346→    for (const b of bars) {
   347→        g.beginPath()
   348→        g.ellipse(sx + b[2] * sr, sy + b[0] * k, sr * 1.0, Math.max(0.5, (b[1] * k) / 2), 0, 0, TAU)
   349→        g.fill()
   350→    }
   351→    g.restore()
   352→    g.filter = "none"
   353→}

   355→function drawWheel(ctx: CanvasRenderingContext2D, cx: number, cy: number, ang: number, blur: number, rimCol: string) {
   356→    ctx.fillStyle = "#0C0507"

   363→    ctx.arc(cx, cy, 15.5, 0, TAU)
   364→    ctx.stroke()

   366→    rg.addColorStop(0, "#D6DAE2")

   369→    ctx.beginPath()

   371→    ctx.fill()

   373→    ctx.fillStyle = "#3E4250"
   374→    for (let j = 0; j < ghosts; j++) {
   375→        ctx.globalAlpha = j === 0 ? 1 - blur * 0.55 : blur * 0.35
   376→        const a0 = ang + j * 0.16 * blur
   377→        for (let q = 0; q < 5; q++) {
   378→            const a = a0 + (q * TAU) / 5

   380→            ctx.ellipse(cx + Math.cos(a) * 6.6, cy + Math.sin(a) * 6.6, 2.6, 1.6, a, 0, TAU)

   382→        }
   383→    }
   384→    ctx.globalAlpha = 1
   385→    if (blur > 0.02) {
   386→        ctx.fillStyle = `rgba(170,175,190,${0.22 * blur})`
   387→        ctx.beginPath()
   388→        ctx.arc(cx, cy, 11, 0, TAU)
   389→        ctx.fill()
   390→    }
   391→    ctx.fillStyle = "#C0C4CC"
   392→    ctx.beginPath()
   393→    ctx.arc(cx, cy, 3, 0, TAU)
   394→    ctx.fill()
   395→    ctx.fillStyle = "#5A5F6C"
   396→    ctx.beginPath()
   397→    ctx.arc(cx, cy, 1, 0, TAU)
   398→    ctx.fill()
   399→    ctx.strokeStyle = "rgba(240,242,248,0.55)"
   400→    ctx.lineWidth = 0.7
   401→    ctx.beginPath()
   402→    ctx.arc(cx, cy, 11.3, 0, TAU)
   403→    ctx.stroke()
   404→    ctx.strokeStyle = rgba(rimCol, 0.45)
   405→    ctx.lineWidth = 1
   406→    ctx.beginPath()
   407→    ctx.arc(cx, cy, 18, -2.5, -0.6)
   408→    ctx.stroke()
   409→}
