   125→   125→interface QState {

   132→   132→function fmt(sec: number) {
   133→   133→  const m = Math.floor(sec / 60);
   134→   134→  const s = sec % 60;
   135→   135→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;

   138→   138→export function ExamEngine({
   139→   139→  theme,
   140→   140→  bank,
   141→   141→  draftId,
   142→   142→  onClose,
   143→   143→}: {
   144→   144→  theme: ExamTheme;
   145→   145→  bank: EngineBank;
   146→   146→  /** stable per-window id (the exam group) — namespaces the draft */
   147→   147→  draftId: string;
   148→   148→  onClose: () => void;
   149→   149→}) {
   150→   150→  const qs = bank.questions;
   151→   151→
   152→   152→  /* ---- draft restore: a refresh or an accidental Escape keeps your work ---- */
   153→   153→  const DRAFT_KEY = examDraftKey(theme.key, draftId);
   154→   154→  const draft = useMemo(() => loadExamDraft(DRAFT_KEY), [DRAFT_KEY]);
   155→   155→  const restored = useMemo(() => {
   156→   156→    if (!draft) return null;
   157→   157→    let n = 0;
   158→   158→    for (const s of Object.values(draft.states)) {
   159→   159→      const st = s as QState | undefined;
   160→   160→      if (st && (st.status === "correct" || st.status === "wrong" || st.revealed)) n += 1;
   161→   161→    }
   162→   162→    return n > 0 ? n : null;
   163→   163→  }, [draft]);
   164→   164→
   165→   165→  const [idx, setIdx] = useState(() => Math.min(Math.max(draft?.idx ?? 0, 0), qs.length - 1));
   166→   166→  const [states, setStates] = useState<Record<string, QState>>(() => {
   167→   167→    const fresh: Record<string, QState> = Object.fromEntries(
   168→   168→      bank.questions.map((q) => [q.id, { status: "unattempted" as QStatus }])
   169→   169→    );
   170→   170→    if (draft) {
   171→   171→      for (const q of bank.questions) {
   172→   172→        const d = draft.states[q.id] as QState | undefined;
   173→   173→        if (d && typeof d.status === "string") fresh[q.id] = d;
   174→   174→      }
   175→   175→    }
   176→   176→    return fresh;
   177→   177→  });
   178→   178→  const [finished, setFinished] = useState(false);
   179→   179→  /* drift-proof clock: seconds = base + (now − startedAt), derived from the
   180→   180→     wall clock — never tick-counted, so background-tab throttling can't
   181→   181→     silently under-report the elapsed time. Survives refresh via the draft. */
   182→   182→  const [clock, setClock] = useState(() => ({
   183→   183→    base: draft?.base ?? 0,
   184→   184→    startedAt: draft?.startedAt ?? Date.now(),
   185→   185→  }));
   186→   186→  const [now, setNow] = useState(() => Date.now());
   187→   187→  const seconds = Math.max(0, Math.floor(clock.base + (now - clock.startedAt) / 1000));
   188→   188→  const bodyRef = useRef<HTMLDivElement>(null);
   189→   189→  /* one ledger row per attempt-cycle — "back to questions" then finishing
   190→   190→     again must not double-log; retake() re-arms it */
   191→   191→  const loggedRef = useRef(false);
   192→   192→
   193→   193→  /* scroll lock (the timer below stops ticking on the results view) */
   194→   194→  useEffect(() => {
   195→   195→    document.body.style.overflow = "hidden";
   196→   196→    return () => {
   197→   197→      document.body.style.overflow = "";
   198→   198→    };
   199→   199→  }, []);
   200→   200→
   201→   201→  useEffect(() => {
   202→   202→    if (finished) return;
   203→   203→    const t = window.setInterval(() => setNow(Date.now()), 1000);
   204→   204→    return () => window.clearInterval(t);
   205→   205→  }, [finished]);
   206→   206→
   207→   207→  /* draft autosave — debounced inside saveExamDraft, flushed on close/finish */
   208→   208→  useEffect(() => {
   209→   209→    if (finished) return;
   210→   210→    if (draftHasWork(states as Record<string, unknown>)) {
   211→   211→      saveExamDraft(DRAFT_KEY, {
   212→   212→        idx,
   213→   213→        states: states as Record<string, unknown>,
   214→   214→        base: clock.base,
   215→   215→        startedAt: clock.startedAt,
   216→   216→      });
   217→   217→    }
   218→   218→  }, [states, idx, clock, finished, DRAFT_KEY]);
   219→   219→
   220→   220→  const closeWindow = useCallback(() => {
   221→   221→    flushExamDraft(); // the attempt is kept — reopening this window restores it
   222→   222→    onClose();
   223→   223→  }, [onClose]);
   224→   224→
   225→   225→  /** freeze the clock, clear the draft, log the attempt, show the report */
   226→   226→  function finish() {
   227→   227→    flushExamDraft();
   228→   228→    clearExamDraft(DRAFT_KEY);
   229→   229→    setClock({ base: seconds, startedAt: Date.now() });
   230→   230→    setFinished(true);
   231→   231→    if (!loggedRef.current) {
   232→   232→      loggedRef.current = true;
   233→   233→      /* one row per subject: single-shelf windows log exactly one row
   234→   234→         (subject = theme.key, unchanged behaviour); mixed review sessions
   235→   235→         split into per-shelf rows so each shelf's accuracy stays true */
   236→   236→      const groups = new Map<string, { c: number; t: number }>();
   237→   237→      for (const qq of qs) {
   238→   238→        const s = states[qq.id]?.status;
   239→   239→        if (s !== "correct" && s !== "wrong") continue;
   240→   240→        const subj = qq.subject ?? theme.key;
   241→   241→        const g = groups.get(subj) ?? { c: 0, t: 0 };
   242→   242→        g.t += 1;
   243→   243→        if (s === "correct") g.c += 1;
   244→   244→        groups.set(subj, g);
   245→   245→      }
   246→   246→      const answered = groups.size > 0;
   247→   247→      const rows: { subject: string; correct: number; total: number }[] = answered
   248→   248→        ? [...groups.entries()].map(([subject, g]) => ({ subject, correct: g.c, total: g.t }))
   249→   249→        : [{ subject: theme.key, correct: score.correct, total: score.total }];
   250→   250→      for (const r of rows) {
   251→   251→        logAttempt({
   252→   252→          subject: r.subject,
   253→   253→          group: draftId,
   254→   254→          title: bank.title,
   255→   255→          correct: r.correct,
   256→   256→          total: r.total,
   257→   257→          pct: Math.round((r.correct / Math.max(1, r.total)) * 100),
   258→   258→          seconds,
   259→   259→        });
   260→   260→      }
   261→   261→    }
   262→   262→  }
   263→   263→
   264→   264→  const q = qs[idx];
   265→   265→  const st = states[q.id];
   266→   266→
   267→   267→  /* keyboard flow — ←/→ move, keys 1–4 answer the MCQ on stage,
   268→   268→     ↵ advances (next question, or the report on the last one).
   269→   269→     Typing surfaces keep their keys: textareas, the ⌘K search
   270→   270→     input, and focused buttons (where ↵ already means "press"). */
   271→   271→  useEffect(() => {
   272→   272→    const onKey = (e: KeyboardEvent) => {
   273→   273→      const t = e.target;
   274→   274→      const inTextarea = t instanceof HTMLTextAreaElement;
   275→   275→      if (e.key === "Escape" && !inTextarea) closeWindow();
   276→   276→      if (finished || inTextarea || e.isComposing || e.metaKey || e.ctrlKey || e.altKey) return;
   277→   277→      if (t instanceof HTMLInputElement) return; // palette search etc.
   278→   278→      if (e.key === "ArrowRight") {
   279→   279→        setIdx((i) => Math.min(i + 1, qs.length - 1));
   280→   280→      } else if (e.key === "ArrowLeft") {
   281→   281→        setIdx((i) => Math.max(i - 1, 0));
   282→   282→      } else if (e.key === "Enter") {
   283→   283→        if (e.repeat) return;
   284→   284→        if (t instanceof HTMLButtonElement || t instanceof HTMLAnchorElement || t instanceof HTMLSelectElement) return;
   285→   285→        if (idx < qs.length - 1) go(idx + 1);
   286→   286→        else finish();
   287→   287→      } else if (!e.repeat) {
   288→   288→        // keys 1–4 pick an option — same contract as clicking the row
   289→   289→        const n = Number(e.key);
   290→   290→        if (!Number.isInteger(n) || n < 1 || n > 4) return;
   291→   291→        if (q.type !== "mcq" || st.status === "correct" || st.status === "wrong") return;
   292→   292→        if (n <= (q.options?.length ?? 0)) chooseMCQ(n - 1);
   293→   293→      }
   294→   294→    };
   295→   295→    document.addEventListener("keydown", onKey);
   296→   296→    return () => document.removeEventListener("keydown", onKey);
   297→   297→  }, [finished, closeWindow, qs.length, idx, q, st]);
   298→   298→
   299→   299→  const patch = (id: string, p: Partial<QState>) =>
   300→   300→    setStates((prev) => ({ ...prev, [id]: { ...prev[id], ...p } }));
   301→   301→
   302→   302→  /* scoring — byType rows follow the theme's declared type order */
   303→   303→  const typeKeys = useMemo(() => Object.keys(theme.typeMeta), [theme]);
   304→   304→  const score = useMemo(() => {
   305→   305→    let correct = 0;
   306→   306→    let wrong = 0;
   307→   307→    let answered = 0;
   308→   308→    const byType: Record<string, { c: number; t: number }> = {};
   309→   309→    for (const t of typeKeys) byType[t] = { c: 0, t: 0 };
   310→   310→    for (const qq of qs) {
   311→   311→      const row = (byType[qq.type] ??= { c: 0, t: 0 });
   312→   312→      row.t += 1;
   313→   313→      const s = states[qq.id]?.status ?? "unattempted";
   314→   314→      if (s === "correct" || s === "wrong") answered += 1;
   315→   315→      if (s === "correct") {
   316→   316→        correct += 1;
   317→   317→        row.c += 1;
   318→   318→      }
   319→   319→      if (s === "wrong") wrong += 1;
   320→   320→    }
   321→   321→    return { correct, wrong, answered, byType, total: qs.length };
   322→   322→  }, [states, qs, typeKeys]);
   323→   323→
   324→   324→  const pct = Math.round((score.correct / score.total) * 100);
   325→   325→  const verdict =
   326→   326→    pct >= 85 ? theme.tiers.topper : pct >= 60 ? theme.tiers.solid : pct >= 35 ? theme.tiers.getting : theme.tiers.restart;
   327→   327→
   328→   328→  function chooseMCQ(i: number) {
   329→   329→    if (st.status === "correct" || st.status === "wrong") return;
   330→   330→    const ok = i === q.answer;
   331→   331→    patch(q.id, {
   332→   332→      chosen: i,
   333→   333→      revealed: true,
   334→   334→      status: ok ? "correct" : "wrong",
   335→   335→    });
   336→   336→    // SM-2: first-try MCQ answers are trusted fully (5), misses reset the card (2)
   337→   337→    // — routed to the question's own shelf in mixed review sessions
   338→   338→    gradeQuestion(q.subject ?? theme.key, q.srcId ?? q.id, ok ? 5 : 2);
   339→   339→  }
   340→   340→
   341→   341→  function selfCheck(ok: boolean) {
   342→   342→    patch(q.id, { status: ok ? "correct" : "wrong" });
   343→   343→    // SM-2: the model answer was visible — grade honestly but trust less
   344→   344→    gradeQuestion(q.subject ?? theme.key, q.srcId ?? q.id, ok ? 3 : 1);
   345→   345→  }
   346→   346→
   347→   347→  function retake() {
   348→   348→    setStates(Object.fromEntries(qs.map((qq) => [qq.id, { status: "unattempted" as QStatus }])));
   349→   349→    setIdx(0);
   350→   350→    setClock({ base: 0, startedAt: Date.now() });
   351→   351→    setNow(Date.now());
   352→   352→    setFinished(false);
   353→   353→    loggedRef.current = false; // a retake is a new, honest attempt
   354→   354→    clearExamDraft(DRAFT_KEY);
   355→   355→    bodyRef.current?.scrollTo({ top: 0 });
   356→   356→  }
   357→   357→
   358→   358→  function go(i: number) {
   359→   359→    setIdx(i);
   360→   360→    bodyRef.current?.scrollTo({ top: 0, behavior: "smooth" });
   361→   361→  }
   362→   362→
   363→   363→  /* ---------------- results view ---------------- */
   364→   364→  if (finished) {
   365→   365→    const mcqMeta = theme.typeMeta.mcq ?? Object.values(theme.typeMeta)[0];
   366→   366→    return (
   367→   367→      <ExamShell
   368→   368→        theme={theme}
   369→   369→        bank={bank}
   370→   370→        score={score}
   371→   371→        seconds={seconds}
   372→   372→        onClose={closeWindow}
   373→   373→        restoredCount={restored}
   374→   374→        onExit={() => setFinished(false)}
   375→   375→        scrollRef={bodyRef}
   376→   376→      >
   377→   377→        <motion.div initial={{ opacity: 0, y: 18 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }}>
   378→   378→          <div className="exam-result-card">
   379→   379→            <div className="exam-score-ring" style={{ borderColor: pct >= 60 ? "#8fae5d" : "#e0a45e" }}>
   380→   380→              <span className="exam-score-num">{pct}%</span>
   381→   381→              <span className="exam-score-sub">
   382→   382→                {score.correct} / {score.total}
   383→   383→              </span>
   384→   384→            </div>
   385→   385→            <div>
   386→   386→              <h3 className="font-display text-2xl font-bold text-[#2b1d10]">Result report</h3>
   387→   387→              <p className="mt-1 text-[.95rem] font-semibold" style={{ color: verdict.c }} role="status">
   388→   388→                {verdict.t}
   389→   389→              </p>
   390→   390→              <div className="mt-3 flex flex-wrap gap-2 text-[.72rem] font-bold">
   391→   391→                <span className="exam-chip" style={{ background: "#eef4dd", color: "#44621d" }}>✓ {score.correct} correct</span>
   392→   392→                <span className="exam-chip" style={{ background: "#fdecec", color: "#8c2f2f" }}>✗ {score.wrong} wrong</span>
   393→   393→                <span className="exam-chip" style={{ background: "#f3e8d2", color: "#6f5438" }}>• {score.total - score.answered} unattempted</span>
   394→   394→                <span className="exam-chip" style={{ background: mcqMeta.bg, color: mcqMeta.color }}>
   395→   395→                  <Timer className="mr-1 inline h-3 w-3" />
   396→   396→                  {fmt(seconds)}
   397→   397→                </span>
   398→   398→              </div>
   399→   399→            </div>
   400→   400→          </div>
   401→   401→
   402→   402→          {/* per-type breakdown */}
   403→   403→          <div className="mt-5 grid gap-3 sm:grid-cols-2">
   404→   404→            {typeKeys.map((t) => {
   405→   405→              const meta = theme.typeMeta[t];
   406→   406→              const d = score.byType[t];
   407→   407→              if (!meta || !d) return null;
   408→   408→              return (
   409→   409→                <div key={t} className="flex items-center justify-between rounded-xl border border-[#e8dcc0] bg-[#fffdf9] px-4 py-3">
   410→   410→                  <span className="text-[.8rem] font-bold" style={{ color: meta.color }}>
   411→   411→                    {meta.label}
   412→   412→                  </span>
   413→   413→                  <span className="text-[.8rem] font-extrabold text-[#2b1d10]">
   414→   414→                    {d.c} / {d.t}
   415→   415→                  </span>
   416→   416→                </div>
   417→   417→              );
   418→   418→            })}
   419→   419→          </div>
   420→   420→
   421→   421→          {/* review the misses */}
   422→   422→          {score.correct < score.total ? (
   423→   423→            <div className="mt-6">
   424→   424→              <p className="exam-eyebrow">
   425→   425→                <Printer className="h-3.5 w-3.5" />
   426→   426→                Review &amp; rewrite these by hand
   427→   427→              </p>
   428→   428→              <div className="mt-3 grid gap-3">
   429→   429→                {qs.map((qq) => {
   430→   430→                  const s = states[qq.id]?.status ?? "unattempted";
   431→   431→                  if (s === "correct") return null;
   432→   432→                  const model =
   433→   433→                    qq.type === "mcq"
   434→   434→                      ? (qq.options?.[qq.answer ?? 0] ?? "")
   435→   435→                      : qq.answerText ?? qq.solution ?? "";
   436→   436→                  return (
   437→   437→                    <div key={qq.id} className="rounded-xl border border-[#e8dcc0] bg-[#fffdf9] p-4">
   438→   438→                      <p className="text-[.88rem] font-bold leading-snug text-[#2b1d10]">{qq.q}</p>
   439→   439→                      {qq.code ? <pre className="exam-code mt-2">{qq.code}</pre> : null}
