   142→/* ============================================================
   143→   useSeenDisplay — the read-checkmark data for every menu.
   144→   Marking is per-lecture; display spans the whole shelf.
   145→   ============================================================ */
   146→function useSeenDisplay(lecture: MktgLecture): ReadonlySet<string> {
   147→  const ids = useMemo(() => lecture.chapters.map((c) => c.id), [lecture]);
   148→  const live = useSectionProgress("mktg", ids);
   149→  const persisted = useMemo(() => {
   150→    try {
   151→      return new Set(Object.keys(loadProgress().mktg ?? {}));
   152→    } catch {
   153→      return new Set<string>();
   154→    }
   155→  }, []);
   156→  return useMemo(() => new Set<string>([...persisted, ...live]), [persisted, live]);
   157→}

   262→            href="#top"

   264→          >
   265→            <span
   266→              className="flex h-6 w-6 items-center justify-center rounded-full text-white"
