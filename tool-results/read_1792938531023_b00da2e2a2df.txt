    30→/* ============================================================
    31→   Reveal — scroll-triggered fade/slide
    32→   ============================================================ */
    33→export function Reveal({
    34→  children,
    35→  delay = 0,
    36→  className = "",
    37→}: {
    38→  children: React.ReactNode;
    39→  delay?: number;
    40→  className?: string;
    41→}) {
    42→  const ref = useRef<HTMLDivElement>(null);
    43→  const [seen, setSeen] = useState(false);
    44→  useEffect(() => {
    45→    const el = ref.current;
    46→    if (!el) return;
    47→    const io = new IntersectionObserver(
    48→      ([e]) => {
    49→        if (e.isIntersecting) {
    50→          setSeen(true);
    51→          io.disconnect();
    52→        }
    53→      },
    54→      { threshold: 0.1, rootMargin: "0px 0px -36px 0px" }
    55→    );
    56→    io.observe(el);
    57→    return () => io.disconnect();
    58→  }, []);
    59→  return (
    60→    <div
    61→      ref={ref}
    62→      className={`reveal ${seen ? "reveal-in" : ""} ${className}`}
    63→      style={{ transitionDelay: `${delay}ms` }}
    64→    >
    65→      {children}
    66→    </div>
    67→  );
    68→}

   188→/* ============================================================
   189→   Python syntax highlighter (tiny, warm palette)
   190→   ============================================================ */
   191→const PY_KEYWORDS = new Set([
   192→  "False","None","True","and","as","assert","break","class","continue","def",
   193→  "del","elif","else","except","finally","for","from","global","if","import",
   194→  "in","is","lambda","nonlocal","not","or","pass","raise","return","try",
   195→  "while","with","yield",
   196→]);
   197→const PY_BUILTINS = new Set([
   198→  "print","len","range","list","sorted","min","max","sum","int","str","float",
   199→  "input","append","insert","pop","remove","sort","reverse","title","upper",
   200→  "lower","strip","lstrip","rstrip","keys","values","items","get","set",
   201→  "dict","tuple","enumerate","zip","abs","round","type","clear","copy","count","index",
   202→]);
   203→
   204→type Tok = { text: string; cls: string };
   205→
   206→function tokenizePython(code: string): Tok[][] {
   207→  const pattern =
   208→    /(#.*$)|("""[\s\S]*?"""|'''[\s\S]*?'''|"(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*')|\b(\d[\d_]*\.?\d*)\b|([A-Za-z_][A-Za-z0-9_]*)|(\s+)|([^\sA-Za-z0-9_]+)/gm;
   209→  const lines: Tok[][] = [[]];
   210→  let m: RegExpExecArray | null;
   211→  while ((m = pattern.exec(code)) !== null) {
   212→    const [full, comment, str, num, word, ws, sym] = m;
   213→    if (comment !== undefined) push(comment, "tk-c");
   214→    else if (str !== undefined) push(str, "tk-s");
   215→    else if (num !== undefined) push(num, "tk-n");
   216→    else if (word !== undefined) {
   217→      if (PY_KEYWORDS.has(word)) push(word, "tk-k");
   218→      else if (PY_BUILTINS.has(word)) push(word, "tk-b");
   219→      else push(word, "");
   220→    } else if (ws !== undefined) push(ws, "ws");
   221→    else if (sym !== undefined) push(sym, "tk-d");
   222→  }
   223→  function push(text: string, cls: string) {
   224→    const parts = text.split("\n");
   225→    parts.forEach((p, i) => {
   226→      if (i > 0) lines.push([]);
   227→      if (p) lines[lines.length - 1].push({ text: p, cls: cls === "ws" ? "" : cls });
   228→    });
   229→  }
   230→  return lines;
   231→}
   232→
   233→function HighlightedCode({ code }: { code: string }) {
   234→  const lines = tokenizePython(code);
   235→  return (
   236→    <pre className="code-body">
   237→      {lines.map((toks, i) => (
   238→        <div key={i} className="flex">
   239→          <span className="mr-4 w-6 shrink-0 select-none text-right text-[#6d5236]">
   240→            {i + 1}
   241→          </span>
   242→          <span className="flex-1">
   243→            {toks.length === 0 ? " " : null}
   244→            {toks.map((t, j) =>
   245→              t.cls ? (
   246→                <span key={j} className={t.cls}>
   247→                  {t.text}
   248→                </span>
   249→              ) : (
   250→                <React.Fragment key={j}>{t.text}</React.Fragment>
   251→              )
   252→            )}
   253→          </span>
   254→        </div>
   255→      ))}
   256→    </pre>
   257→  );
   258→}

   260→/* ============================================================
   261→   CodeBlock — syntax-highlighted snippet with an optional LIVE
   262→   Python runtime: ▶ Run executes the code in the browser (Pyodide),
   263→   ✎ lets the learner edit the code, ⟲ restores the original.
   264→   `pre`   → hidden Python executed before the visible code
   265→   `files` → virtual .py modules written before the run
   266→   ============================================================ */
   267→type RunPhase = "idle" | "running" | "done" | "error";
   268→
   269→export function CodeBlock({
   270→  title = "main.py",
   271→  code,
   272→  output,
   273→  note,
   274→  className = "",
   275→  runnable = true,
   276→  pre,
   277→  files,
   278→  dark = false,
   279→}: {
   280→  title?: string;
   281→  code: string;
   282→  output?: string;
   283→  note?: React.ReactNode;
   284→  className?: string;
   285→  runnable?: boolean;
   286→  pre?: string;
   287→  files?: Record<string, string>;
   288→  dark?: boolean;
   289→}) {
   290→  const [current, setCurrent] = useState(code);
   291→  const [editing, setEditing] = useState(false);
   292→  const [phase, setPhase] = useState<RunPhase>("idle");
   293→  const [bootMsg, setBootMsg] = useState("");
   294→  const [out, setOut] = useState("");
   295→  const [err, setErr] = useState("");
   296→  const [elapsed, setElapsed] = useState(0);
   297→  const [copied, setCopied] = useState(false);
   298→  const termRef = useRef<HTMLPreElement>(null);
   299→  const dirty = current !== code;
   300→  const live = phase !== "idle";
   301→  const busy = phase === "running";
   302→
   303→  useEffect(() => {
   304→    if (termRef.current) termRef.current.scrollTop = termRef.current.scrollHeight;
   305→  }, [out, err]);
   306→
   307→  async function run() {
   308→    if (busy) return;
   309→    setEditing(false);
   310→    setPhase("running");
   311→    setOut("");
   312→    setErr("");
   313→    setBootMsg("");
   314→    setElapsed(0);
   315→    const t0 = performance.now();
   316→    try {
   317→      // Global run lock: Pyodide exposes a single stdout/stderr/stdin handler
   318→      // set, so two simultaneous runners would steal each other's output
   319→      // (and the first finisher's cleanup would disconnect the second's
   320→      // capture mid-run). Queue behind whatever else is executing.
   321→      await withRunLock(async () => {
   322→        const pyr = await getPyodide((m) => setBootMsg(m));
   323→        setBootMsg("");
   324→        let lastOut = "";
   325→        // capped push: a print flood can't grow the buffer unbounded
   326→        const push = makeCappedPush((full) => {
   327→          lastOut = full;
   328→          setOut(full);
   329→        });
   330→        pyr.setStdout({ batched: push });
   331→        pyr.setStderr({ batched: push });
   332→        pyr.setStdin({
   333→          stdin: () => {
   334→            try {
   335→              return window.prompt("Python input() is asking for a value — cancel = stop the program:");
   336→            } catch {
   337→              return null; // prompt blocked (e.g. sandboxed iframe) → treat as EOF
   338→            }
   339→          },
   340→        });
   341→        try {
   342→          // Fresh module cache + virtual files for import demos.
   343→          await pyr.runPythonAsync(PURGE_USER_MODULES);
   344→          if (files) {
   345→            for (const [name, content] of Object.entries(files)) pyr.FS.writeFile(name, content);
   346→          }
   347→          const wf = current.match(/^%%writefile\s+([^\s#]+)/);
   348→          if (wf) {
   349→            // Notebook magic: save the cell as a real file inside the runtime.
   350→            const name = wf[1];
   351→            const nl = current.indexOf("\n");
   352→            const body = (nl === -1 ? "" : current.slice(nl + 1)).replace(/^\n+/, "");
   353→            pyr.FS.writeFile(name, body);
   354→            push(`✓ saved ${name} (${body.split("\n").length} lines) — import blocks can now use it`);
   355→          } else {
   356→            const ns = pyr.globals.get("dict")() as { destroy?: () => void };
   357→            try {
   358→              // step-guard: cooperative runaway-loop killer (see lib/pyguard.ts)
   359→              await withStepGuard(pyr, async () => {
   360→                if (pre) await pyr.runPythonAsync(pre, { globals: ns });
   361→                await pyr.runPythonAsync(current, { globals: ns });
   362→              });
   363→            } finally {
   364→              ns.destroy?.();
   365→            }
   366→          }
   367→        } finally {
   368→          // handlers live and die with THIS run — a queued runner can install
   369→          // its own the moment we yield the lock
   370→          try {
   371→            pyr.setStdout();
   372→            pyr.setStderr();
   373→            pyr.setStdin();
   374→          } catch {
   375→            /* already reset */
   376→          }
   377→        }
   378→      });
   379→      setElapsed((performance.now() - t0) / 1000);
   380→      setPhase("done");
   381→    } catch (e) {
   382→      const msg = e instanceof Error ? e.message : String(e);
   383→      setErr(cleanTraceback(msg) || msg);
   384→      setElapsed((performance.now() - t0) / 1000);
   385→      setPhase("error");
   386→    }
   387→  }
   388→
   389→  function copy() {
   390→    try {
   391→      navigator.clipboard.writeText(current).then(
   392→        () => {
   393→          setCopied(true);
   394→          window.setTimeout(() => setCopied(false), 1600);
   395→        },
   396→        () => undefined
   397→      );
   398→    } catch {
   399→      /* clipboard unavailable */
   400→    }
   401→  }
   402→
   403→  return (
   404→    <div className={`code-shell ${className}`}>
   405→      <div className="code-bar">
   406→        <span className="code-dot" style={{ background: "#e4674a" }} />
   407→        <span className="code-dot" style={{ background: "#e8a13c" }} />
   408→        <span className="code-dot" style={{ background: "#8fae5d" }} />
   409→        <span className="code-title ml-2">{title}</span>
   410→        {runnable ? (
   411→          <div className="ml-auto flex items-center gap-1">
   412→            <button
   413→              type="button"
   414→              onClick={copy}
   415→              className="code-icon-btn"
   416→              title="Copy code"
   417→              aria-label="Copy code"
   418→            >
   419→              {copied ? <Check className="h-3.5 w-3.5" style={{ color: "#8fae5d" }} /> : <Copy className="h-3.5 w-3.5" />}
   420→            </button>
   421→            <button
   422→              type="button"
   423→              onClick={() => {
   424→                setCurrent(code);
   425→                setEditing((v) => !v);
   426→              }}
   427→              className={`code-icon-btn ${editing ? "code-icon-btn-on" : ""}`}
   428→              title={dirty ? "Editing — click ⟲ to restore the original" : editing ? "Done editing" : "Edit the code, then run it"}
   429→              aria-label="Edit code"
   430→            >
   431→              <Pencil className="h-3.5 w-3.5" />
   432→            </button>
   433→            {dirty ? (
   434→              <button
   435→                type="button"
   436→                onClick={() => setCurrent(code)}
   437→                className="code-icon-btn"
   438→                title="Restore the original code"
   439→                aria-label="Restore original code"
   440→              >
   441→                <RotateCcw className="h-3.5 w-3.5" />
   442→              </button>
   443→            ) : null}
   444→            <button
   445→              type="button"
   446→              onClick={run}
   447→              disabled={busy}
   448→              className="run-btn"
   449→              title="Run this code live — real Python 3.12 in your browser"
   450→            >
   451→              {busy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Play className="h-3.5 w-3.5" />}
   452→              {busy ? "running" : "Run"}
   453→            </button>
   454→          </div>
   455→        ) : (
   456→          <span className="readonly-chip" title="Illustrative snippet — not executable">
   457→            <EyeOff className="h-3 w-3" />
   458→            read-only
   459→          </span>
   460→        )}
   461→      </div>
   462→
   463→      {editing ? (
   464→        <textarea
   465→          value={current}
   466→          onChange={(e) => setCurrent(e.target.value)}
   467→          spellCheck={false}
   468→          aria-label="Editable code"
   469→          className="code-edit"
   470→          rows={Math.max(6, current.split("\n").length + 1)}
   471→        />
   472→      ) : (
   473→        <HighlightedCode code={current} />
   474→      )}
   475→
   476→      <AnimatePresence>
   477→        {live ? (
   478→          <motion.div
   479→            key="live-term"
   480→            initial={{ opacity: 0, height: 0 }}
   481→            animate={{ opacity: 1, height: "auto" }}
   482→            exit={{ opacity: 0, height: 0 }}
   483→            transition={{ duration: 0.22 }}
   484→            style={{ overflow: "hidden" }}
   485→          >
   486→            <div className="live-term">
   487→              <div className="live-term-bar">
   488→                <Terminal className="h-3 w-3" />
   489→                LIVE RUN
   490→                <span className="ml-auto normal-case tracking-normal opacity-70">
   491→                  {bootMsg ? "python runtime" : "python 3.12 · webassembly"}
   492→                </span>
   493→              </div>
   494→              <pre ref={termRef} className="live-term-body">
   495→                {bootMsg ? (
   496→                  <span className="inline-flex items-center gap-2 text-[#f5c889]">
   497→                    <Loader2 className="h-3.5 w-3.5 animate-spin" /> {bootMsg}
   498→                  </span>
   499→                ) : null}
   500→                {out ? out : !bootMsg && busy ? "\u2026" : ""}
   501→                {!bootMsg && busy ? <span className="term-caret" aria-hidden /> : null}
   502→                {err ? <span className="tb-err">{err}\n</span> : null}
   503→              </pre>
   504→              {phase === "done" || phase === "error" ? (
   505→                <div className="live-term-foot">
   506→                  <span className={phase === "done" ? "foot-ok" : "foot-bad"}>
   507→                    {phase === "done"
   508→                      ? `✓ finished in ${elapsed.toFixed(2)}s`
   509→                      : "✗ the program raised an error"}
   510→                  </span>
   511→                  <span className="ml-auto opacity-60">CPython · Pyodide</span>
   512→                </div>
   513→              ) : null}
   514→            </div>
   515→          </motion.div>
   516→        ) : null}
   517→      </AnimatePresence>
   518→
   519→      {/* Pen-paper practice: runnable blocks HIDE the static output — the
   520→          learner predicts it first, then presses ▶ Run to see the real thing.
   521→          Read-only blocks keep their output (there is no Run button). */}
   522→      {output && !runnable ? (
   523→        <div className="code-out">
   524→          <span className="out-label">OUTPUT</span>
   525→          {output}
   526→        </div>
   527→      ) : null}
   528→      {note ? (
   529→        <div className="border-t border-[#4a3020] bg-[rgba(255,220,170,.04)] px-5 py-3 text-[.8rem] leading-relaxed text-[#d9b88c]">
   530→          {note}
   531→        </div>
   532→      ) : null}
   533→    </div>
   534→  );
   535→}

   537→/* ============================================================

   540→export function PyReadyToast() {
   541→  const [show, setShow] = useState(false);
   542→  useEffect(() => {
   543→    const onReady = () => setShow(true);
   544→    window.addEventListener("foc-py-ready", onReady);
   545→    return () => window.removeEventListener("foc-py-ready", onReady);
   546→  }, []);
   547→  useEffect(() => {
   548→    if (!show) return;
   549→    const t = window.setTimeout(() => setShow(false), 6000);
   550→    return () => window.clearTimeout(t);
   551→  }, [show]);
   552→  return (
   553→    <AnimatePresence>
   554→      {show ? (
   555→        <motion.div
   556→          key="py-toast"
   557→          initial={{ opacity: 0, y: 18, scale: 0.96 }}
   558→          animate={{ opacity: 1, y: 0, scale: 1 }}
   559→          exit={{ opacity: 0, y: 10 }}
   560→          transition={{ type: "spring", stiffness: 320, damping: 26 }}
   561→          role="status"
   562→          className="fixed bottom-4 left-4 z-[70] flex max-w-[calc(100vw-2rem)] items-center gap-2.5 rounded-2xl border border-[#ecd9bd] bg-[#fffaf1]/95 px-4 py-3 shadow-2xl backdrop-blur-md"
   563→          style={{ boxShadow: "0 24px 50px -20px rgba(43,27,16,.5)" }}
   564→        >
   565→          <span className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full" style={{ background: "linear-gradient(135deg,#ea580c,#c2410c)" }}>
   566→            <Sparkles className="h-3.5 w-3.5 text-white" />

   630→/* ============================================================
   631→   Quiz — interactive single-choice
   632→   ============================================================ */
   633→export function Quiz({
   634→  q,
   635→  options,
   636→  answer,
   637→  explain,
   638→  source,
   639→}: {
   640→  q: string;
   641→  options: string[];
   642→  answer: number;
   643→  explain: string;
   644→  source?: string;
   645→}) {
   646→  const [chosen, setChosen] = useState<number | null>(null);
   647→  const revealed = chosen !== null;
   648→  return (
   649→    <div className="card-warm p-5">
   650→      <div className="flex items-start gap-2">
   651→        <GraduationCap className="mt-0.5 h-5 w-5 shrink-0" style={{ color: "#c2410c" }} />
   652→        <div>
   653→          <p className="quiz-q font-semibold leading-snug text-[#3e2c1d]">{q}</p>
   654→          {source ? (
   655→            <p className="quiz-src mt-1 text-[.7rem] font-bold uppercase tracking-[.16em] text-[#8a5a10]">
   656→              {source}
   657→            </p>
   658→          ) : null}
   659→        </div>
   660→      </div>
   661→      <div className="mt-3 grid gap-2">
   662→        {options.map((opt, i) => {
   663→          const isAnswer = i === answer;
   664→          const cls = !revealed
   665→            ? "quiz-opt"
   666→            : isAnswer
   667→            ? "quiz-opt correct"
   668→            : i === chosen
   669→            ? "quiz-opt wrong"
   670→            : "quiz-opt";
   671→          return (
   672→            <button
   673→              key={i}
   674→              className={cls}
   675→              disabled={revealed}
   676→              onClick={() => setChosen(i)}
   677→              aria-label={`Option ${i + 1}: ${opt}`}
   678→            >
   679→              <span className="mr-2 inline-flex h-5 w-5 items-center justify-center rounded-full border border-current text-[.65rem] font-bold">
   680→                {revealed && isAnswer ? (
   681→                  <Check className="h-3 w-3" />
   682→                ) : revealed && i === chosen ? (
   683→                  <X className="h-3 w-3" />
   684→                ) : (
   685→                  String.fromCharCode(65 + i)
   686→                )}
   687→              </span>
   688→              {opt}
   689→            </button>
   690→          );
   691→        })}
   692→      </div>
   693→      {revealed ? (
   694→        <div className="anim-pop mt-3 rounded-xl border border-[#e8dcc0] bg-[#fbf3e3] p-3 text-[.85rem] leading-relaxed text-[#2b2014]">
   695→          <strong style={{ color: "#c2410c" }}>
   696→            {chosen === answer ? "Correct. " : "Not quite. "}
   697→          </strong>
   698→          {explain}
   699→        </div>
   700→      ) : null}
   701→    </div>
   702→  );
   703→}
