    81→/* ---------------------------------------------- count-up stat */
    82→function Stat({
    83→  value,
    84→  render,
    85→  label,
    86→  delay = 0,
    87→}: {
    88→  value: number;
    89→  render?: (n: number) => React.ReactNode;
    90→  label: string;
    91→  delay?: number;
    92→}) {
    93→  const ref = useRef<HTMLDivElement>(null);
    94→  const inView = useInView(ref, { once: true, margin: "-40px" });
    95→  const [n, setN] = useState(0);
    96→
    97→  useEffect(() => {
    98→    if (!inView) return;
    99→    const controls = animate(0, value, {
   100→      duration: 1.5,
   101→      delay,
   102→      ease: EASE,
   103→      onUpdate: (v) => setN(Math.round(v)),
   104→    });
   105→    return () => controls.stop();
   106→  }, [inView, value, delay]);
   107→
   108→  return (
   109→    <div ref={ref} className="flex flex-col items-center gap-1">
   110→      <span className="font-display text-3xl font-black tracking-tight text-[#f7ecdd] md:text-4xl">
   111→        {render ? render(n) : n}
   112→      </span>
   113→      <span className="text-[.62rem] font-extrabold uppercase tracking-[.22em] text-[#8d7b63]">{label}</span>
   114→    </div>
   115→  );
   116→}

   230→/* cards due for spaced review on this shelf — read once after mount so the
   231→   first paint stays hydration-stable; renders nothing until the store says so */
   232→function DueBadge({ subjectId }: { subjectId: string }) {
   233→  const [n, setN] = useState(0);
   234→  useEffect(() => {
   235→    /* rAF-deferred: keeps the first paint hydration-stable */
   236→    const raf = requestAnimationFrame(() => setN(dueCounts()[subjectId] ?? 0));
   237→    return () => cancelAnimationFrame(raf);
   238→  }, [subjectId]);
   239→  if (n <= 0) return null;
   240→  return (
   241→    <span
   242→      className="glass-chip rounded-full px-2.5 py-1 font-mono text-[.6rem] font-bold uppercase tracking-[.14em] text-[#f5c889]"
   243→      title="Cards due for spaced review — open the ledger (⌘K → ledger)"
   244→    >
   245→      {n} due
   246→    </span>
   247→  );
   248→}

   250→/* chapters read on this shelf — same hydration-safe pattern as DueBadge;
   251→   silent until the reader has actually walked at least one chapter */
   252→function ChaptersBadge({ subjectId, total }: { subjectId: string; total: number }) {
   253→  const [seen, setSeen] = useState(0);
   254→  useEffect(() => {
   255→    const raf = requestAnimationFrame(() => setSeen(readProgressCounts()[subjectId] ?? 0));
   256→    return () => cancelAnimationFrame(raf);
   257→  }, [subjectId]);
   258→  if (seen <= 0 || total <= 0) return null;
   259→  const done = seen >= total;
   260→  return (
   261→    <span
   262→      className="glass-chip rounded-full px-2.5 py-1 font-mono text-[.6rem] font-bold uppercase tracking-[.14em]"
   263→      style={{ color: done ? "#a9c977" : "#c9b99f" }}
   264→      title="Chapters read — tracked as you scroll the notes"
   265→    >
   266→      {done ? "✓ " : ""}
   267→      {seen}/{total} read
   268→    </span>
   269→  );
   270→}

   447→/* today's daily-goal progress — a tiny hydration-safe ring that rides
   448→   the pill nav; one click lands on the ledger where the ring lives */
   449→function GoalChip() {
   450→  const [goal, setGoal] = useState<{ done: number; target: number; met: boolean } | null>(null);
   451→  useEffect(() => {
   452→    /* rAF-deferred: keeps the first paint hydration-stable */
   453→    const raf = requestAnimationFrame(() => {
   454→      const g = goalStatus();
   455→      setGoal({ done: g.done, target: g.target, met: g.met });
   456→    });
   457→    return () => cancelAnimationFrame(raf);
   458→  }, []);
   459→  if (!goal) return null;
   460→  const R = 7;
   461→  const C = 2 * Math.PI * R;
   462→  const p = Math.max(0, Math.min(1, goal.target > 0 ? goal.done / goal.target : 0));
   463→  const tone = goal.met ? "#8fae5d" : "#f5c889";
   464→  return (
   465→    <button
   466→      type="button"
   467→      onClick={() => {
   468→        window.location.hash = "#/ledger";
   469→        window.scrollTo(0, 0);
   470→      }}
   471→      aria-label={`Daily goal: ${goal.done} of ${goal.target} answers today — open the ledger`}
   472→      title={`Daily goal — ${goal.done}/${goal.target} answers today`}
   473→      className="flex items-center gap-1.5 rounded-full border border-white/10 bg-white/[.05] px-2.5 py-1 font-mono text-[.6rem] font-bold tracking-wide text-[#cbb89a] transition hover:bg-white/[.12] hover:text-[#f5c889]"
   474→    >
   475→      <svg viewBox="0 0 18 18" className="h-3.5 w-3.5 -rotate-90" aria-hidden>
   476→        <circle cx="9" cy="9" r={R} fill="none" stroke="#ffffff1c" strokeWidth="2.4" />
   477→        <circle
   478→          cx="9"
   479→          cy="9"
   480→          r={R}
   481→          fill="none"
   482→          stroke={tone}
   483→          strokeWidth="2.4"
   484→          strokeLinecap="round"
   485→          strokeDasharray={C}
   486→          strokeDashoffset={C * (1 - p)}
   487→        />
   488→      </svg>
   489→      {goal.done}/{goal.target}
   490→    </button>
   491→  );
   492→}
