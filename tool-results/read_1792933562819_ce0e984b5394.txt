   132→    let ticking = false;
   133→    const update = () => {
   134→      ticking = false;

   137→      const pct = total > 0 ? (doc.scrollTop / total) * 100 : 0;

   146→      }
   147→      if (current !== activeRef.current) {
   148→        activeRef.current = current;
   149→        setActive(current);
   150→      }

   153→      if (!ticking) {
   154→        ticking = true;
   155→        requestAnimationFrame(update);
   156→      }
