   381→/* curved pointer path between two boxes, trimmed so arrowheads stay visible */
   382→function hopPath(
   383→  ax: number, ay: number, hwa: number, hha: number,
   384→  bx: number, by: number, hwb: number, hhb: number,
   385→  bend: number
   386→): string {
   387→  const dx = bx - ax, dy = by - ay;
   388→  const len = Math.hypot(dx, dy) || 1;
   389→  const ux = dx / len, uy = dy / len;
   390→  const exit = (hw: number, hh: number) =>
   391→    Math.min(Math.abs(ux) < 1e-6 ? Infinity : hw / Math.abs(ux), Math.abs(uy) < 1e-6 ? Infinity : hh / Math.abs(uy));
   392→  const sx = ax + ux * (exit(hwa, hha) + 6);
   393→  const sy = ay + uy * (exit(hwa, hha) + 6);
   394→  const ex = bx - ux * (exit(hwb, hhb) + 4);
   395→  const ey = by - uy * (exit(hwb, hhb) + 4);
   396→  const mx = (sx + ex) / 2 - uy * bend;
   397→  const my = (sy + ey) / 2 + ux * bend;
   398→  const f = (n: number) => n.toFixed(1);
   399→  return `M ${f(sx)} ${f(sy)} Q ${f(mx)} ${f(my)} ${f(ex)} ${f(ey)}`;
   400→}

   747→function shuffledOrder(prev: number[]): number[] {
   748→  const p = [1, 2, 3, 4];
   749→  do {
   750→    for (let i = p.length - 1; i > 0; i--) {
   751→      const j = Math.floor(Math.random() * (i + 1));
   752→      [p[i], p[j]] = [p[j], p[i]];
   753→    }
   754→  } while (p.every((v, i) => v === prev[i]));
   755→  return p;
   756→}
