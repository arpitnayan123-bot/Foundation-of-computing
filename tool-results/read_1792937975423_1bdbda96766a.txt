   156→/* ============================================================
   157→   useSeenDisplay — the read-checkmark data for every menu.
   158→
   159→   Marking is per-lecture (only the rendered chapters can be
   160→   swept — the Kindle-style furthest-read sweep must never
   161→   sweep chapters that aren't on the page), but DISPLAY spans
   162→   the whole shelf: whatever the reader marked in earlier
   163→   sessions/lectures stays visible via the persisted record.
   164→   ============================================================ */
   165→function useSeenDisplay(lecture: AtbLecture): ReadonlySet<string> {
   166→  const ids = useMemo(() => lecture.chapters.map((c) => c.id), [lecture]);
   167→  const live = useSectionProgress("atb", ids);
   168→  const persisted = useMemo(() => {
   169→    try {
   170→      return new Set(Object.keys(loadProgress().atb ?? {}));
   171→    } catch {
   172→      return new Set<string>();
   173→    }
   174→  }, []);
   175→  return useMemo(() => new Set<string>([...persisted, ...live]), [persisted, live]);
   176→}

   178→/* ============================================================
   179→   TopNav — structured chapter menu + a "Test my knowledge"
   180→   launcher inside every chapter group.
   181→
   182→   PERF (kept from FocApp): the progress bar is written straight
   183→   to the DOM through a ref inside a requestAnimationFrame, and
   184→   React state updates only when the active section CHANGES —
   185→   the nav never re-renders per scroll tick.
   186→
   187→   Chapters live inside the lecture reader, so a menu click on a
   188→   chapter of a DIFFERENT lecture first switches the lecture and
   189→   then scrolls (goToChapter in AtbApp) — a plain anchor would
   190→   aim at a section that isn't rendered.
   191→   ============================================================ */

   205→  const [mobileOpen, setMobileOpen] = useState(false);
   206→  const mobileRef = useDialogA11y<HTMLDivElement>(mobileOpen);
   207→  const navRef = useRef<HTMLElement>(null);
   208→  const progressRef = useRef<HTMLDivElement>(null);
   209→  const activeRef = useRef("");
   210→
   211→  useEffect(() => {
   212→    let ticking = false;
   213→    const update = () => {
   214→      ticking = false;
   215→      const doc = document.documentElement;
   216→      const total = doc.scrollHeight - doc.clientHeight;
   217→      const pct = total > 0 ? (doc.scrollTop / total) * 100 : 0;
   218→      if (progressRef.current) progressRef.current.style.width = `${pct}%`;
   219→
   220→      // scroll-spy: only touch React state when the section changes.
   221→      // Sections of other lectures simply aren't in the DOM, so the
   222→      // loop naturally spies on the rendered lecture only.
   223→      const probe = window.innerHeight * 0.32;
   224→      let current = "";
   225→      for (const item of FLAT_ITEMS) {
   226→        const el = document.getElementById(item.id);
   227→        if (el && el.getBoundingClientRect().top <= probe) current = item.id;
   228→      }
   229→      if (current !== activeRef.current) {
   230→        activeRef.current = current;
   231→        setActive(current);
   232→      }
   233→    };
   234→    const onScroll = () => {
   235→      if (!ticking) {
   236→        ticking = true;
   237→        requestAnimationFrame(update);
   238→      }
   239→    };
   240→    update();
   241→    window.addEventListener("scroll", onScroll, { passive: true });
   242→    window.addEventListener("resize", onScroll);
   243→    return () => {
   244→      window.removeEventListener("scroll", onScroll);
   245→      window.removeEventListener("resize", onScroll);
   246→    };
   247→  }, []);

   255→      if (e.key === "Escape") {
   256→        setOpenGroup(null);
   257→        setMobileOpen(false);
   258→      }
   259→    };
   260→    document.addEventListener("mousedown", onDown);
   261→    document.addEventListener("keydown", onKey);
   262→    return () => {
   263→      document.removeEventListener("mousedown", onDown);
   264→      document.removeEventListener("keydown", onKey);
   265→    };
   266→  }, []);
   267→
   268→  // lock body scroll while the mobile index is open
   269→  useEffect(() => {
   270→    document.body.style.overflow = mobileOpen ? "hidden" : "";
   271→    return () => {
   272→      document.body.style.overflow = "";
   273→    };
   274→  }, [mobileOpen]);
   275→
   276→  const activeGroup = active ? groupOf(active) : null;
   277→
   278→  return (
   279→    <nav ref={navRef} aria-label="Chapters" className="fixed inset-x-0 top-0 z-50">
   280→      <div className="nav-shell">
   281→        <div className="mx-auto flex h-14 max-w-7xl items-center gap-2 px-3 sm:px-5">
   282→          {/* back to The Atlas — the shelf this subject lives on */}
   283→          <button
   284→            type="button"
   285→            onClick={onExitAtlas}
   286→            title="Back to The Atlas — the subject shelf"
   287→            aria-label="Back to The Atlas"
   288→            className="flex h-9 shrink-0 items-center gap-1.5 rounded-full border border-white/70 bg-white/60 px-3 text-[.64rem] font-extrabold uppercase tracking-[.14em] text-[#8a5f30] shadow-[0_4px_16px_-8px_rgba(70,42,15,.35)] backdrop-blur-md transition-colors duration-200 hover:bg-white/85 hover:text-[#c2410c]"
   289→          >
   290→            <Map className="h-3.5 w-3.5" />
   291→            <span className="hidden sm:inline">Atlas</span>
   292→          </button>
   293→
   294→          {/* brand — quiet glass chip, Apple-quiet rather than loud gradient */}
   295→          <a
   296→            href="#top"
   297→            className="flex shrink-0 items-center gap-2 rounded-full border border-white/70 bg-white/60 py-1.5 pl-2 pr-3.5 shadow-[0_4px_16px_-8px_rgba(70,42,15,.35)] backdrop-blur-md transition-colors duration-200 hover:bg-white/80"
   298→          >
   299→            <span
   300→              className="flex h-6 w-6 items-center justify-center rounded-full text-white"
   301→              style={{ background: "linear-gradient(135deg,#ea580c,#c2410c)", boxShadow: "0 4px 10px -3px rgba(194,65,12,.55)" }}
   302→            >
   303→              <Route className="h-3.5 w-3.5" />
   304→            </span>
   305→            <span className="text-sm font-black tracking-tight text-[#3e2c1d]">ATB Notes</span>
   306→          </a>
   307→
   308→          {/* desktop — grouped chapter menu (all six groups). Opens at lg:
   309→              below that the bar would overflow once every group is shown,
   310→              so narrower screens get the scrolling chip strip + the
   311→              full-screen index instead. */}
   312→          <div className="ml-2 hidden flex-1 items-center gap-1 lg:flex">
   313→            {GROUPS.map((group) => {
   314→              const isActive = activeGroup === group.id;
   315→              const open = openGroup === group.id;
   316→              return (
   317→                <div
   318→                  key={group.id}
   319→                  className="relative"
   320→                  onMouseEnter={() => setOpenGroup(group.id)}
   321→                  onMouseLeave={() => setOpenGroup((g) => (g === group.id ? null : g))}
   322→                >
   323→                  <button
   324→                    type="button"
   325→                    aria-expanded={open}
   326→                    aria-haspopup="true"

   332→                        : "text-[#6b563d] hover:bg-white/55 hover:text-[#3e2c1d] hover:shadow-[inset_0_1px_0_rgba(255,255,255,.9)]"

   340→                  <AnimatePresence>

   943→  const lecture = lectureById(lectureId) ?? ATB_LECTURES[0];
   944→  const seen = useSeenDisplay(lecture);
   945→
   946→  /* pick a lecture from the hero sequence (or the pager) and bring
   947→     the notes into view — "the lecture number 1 notes will be
   948→     available below", literally */
   949→  const selectLecture = useCallback((id: string) => {
   950→    setLectureId(id);
   951→    requestAnimationFrame(() =>
   952→      requestAnimationFrame(() => {
   953→        const el = document.getElementById("notes-top");
   954→        if (!el) return;
   955→        const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   956→        el.scrollIntoView({ behavior: reduce ? "auto" : "smooth", block: "start" });
   957→      })
   958→    );
   959→  }, []);
   960→
   961→  /* chapter jumps — from the nav menus or the ⌘K palette. If the
   962→     chapter belongs to another lecture, switch first, wait for the
   963→     render, then run the standard flash-jump */
   964→  const goToChapter = useCallback((chapterId: string) => {
   965→    const target = lectureOfChapter(chapterId);
   966→    if (!target) return;
   967→    const scrollWhenReady = () => {
   968→      let frames = 0;
   969→      const tick = () => {
   970→        const el = document.getElementById(chapterId);
   971→        if (el) {
   972→          jumpTo(el);
   973→          return;
   974→        }
   975→        if (++frames < 240) requestAnimationFrame(tick);
   976→      };
   977→      requestAnimationFrame(tick);
   978→    };
   979→    if (lectureIdRef.current === target.id) {
   980→      scrollWhenReady();
   981→      return;
   982→    }
   983→    setLectureId(target.id);
   984→    requestAnimationFrame(() => requestAnimationFrame(scrollWhenReady));
   985→  }, []);
   986→
   987→  /* ⌘K handoffs (queued + live) route through the lecture-aware jumper */
   988→  useOpenSection("atb", goToChapter);
   989→
   990→  const openExam = useCallback((g: AtbExamGroup) => setExamGroup(g), []);
   991→  const openHub = useCallback(() => setHubOpen(true), []);
   992→
   993→  return (
   994→    // id="top" — the brand chip links to #top; without a real element here the
   995→    // client-side router swallows the fragment and the page never scrolls back
   996→    <div id="top" className="page-wrap warm-canvas">
   997→      <TopNav onOpenExam={openExam} onExitAtlas={onExitAtlas} onJumpChapter={goToChapter} seen={seen} />
   998→      <main className="pt-14">
   999→        <PartBoundary label="Hero">
  1000→          <AtbHero
  1001→            activeId={lectureId}
  1002→            onSelect={selectLecture}
  1003→            onPractice={openHub}
  1004→            seen={seen}
  1005→            totalQuestions={ATB_TOTAL_QUESTIONS}
  1006→          />
  1007→        </PartBoundary>
  1008→
  1009→        {/* the notes start here — selecting a lecture lands the reader on this line */}
  1010→        <div id="notes-top" aria-hidden className="relative scroll-mt-16" />
  1011→
  1012→        <PartBoundary key={lecture.id} label={`Lecture ${lecture.code} — ${lecture.title}`}>
  1013→          <motion.div
  1014→            initial={{ opacity: 0, y: 18 }}
  1015→            animate={{ opacity: 1, y: 0 }}
  1016→            transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
  1017→          >
  1018→            {lecture.chapters.map((c) => (
  1019→              <PartBoundary key={c.id} label={c.label}>
  1020→                {SECTION_COMPONENTS[c.id]}
  1021→              </PartBoundary>
  1022→            ))}
  1023→            <LecturePager current={lecture} onJump={selectLecture} />
  1024→          </motion.div>
  1025→        </PartBoundary>
  1026→      </main>
  1027→      <AtbFooter />
  1028→      <PracticeHub open={hubOpen} onClose={() => setHubOpen(false)} onOpenExam={openExam} />
  1029→      {examGroup ? <AtbExamWindow group={examGroup} onClose={() => setExamGroup(null)} /> : null}
  1030→    </div>
  1031→  );
  1032→}
