   203→/* ============================================================
   204→   useSeenDisplay — the read-checkmark data for every menu.
   205→   Marking is per-lecture; display spans the whole shelf.
   206→   ============================================================ */
   207→function useSeenDisplay(lecture: FocLecture): ReadonlySet<string> {
   208→  const ids = useMemo(() => lecture.chapters.map((c) => c.id), [lecture]);
   209→  const live = useSectionProgress("foc", ids);
   210→  const persisted = useMemo(() => {
   211→    try {
   212→      return new Set(Object.keys(loadProgress().foc ?? {}));
   213→    } catch {
   214→      return new Set<string>();
   215→    }
   216→  }, []);
   217→  return useMemo(() => new Set<string>([...persisted, ...live]), [persisted, live]);
   218→}

   220→/* ============================================================
   221→   TopNav — structured chapter menu + Python Lab button +
   222→   a "Test my knowledge" launcher inside every chapter group.
   223→
   224→   PERF: the progress bar is written straight to the DOM
   225→   through a ref inside a requestAnimationFrame, and React state
   226→   updates only when the active section actually CHANGES.
   227→
   228→   Chapter links never jump as plain anchors — they call
   229→   onJumpChapter so a cross-lecture jump switches the lecture
   230→   first, then scrolls to the freshly-rendered section.
   231→   ============================================================ */
   232→function TopNav({
   233→  onOpenLab,
   234→  onOpenExam,
   235→  onExitAtlas,
   236→  onJumpChapter,
   237→  seen,
   238→}: {
   239→  onOpenLab: () => void;
   240→  onOpenExam: (g: ExamGroup) => void;
   241→  onExitAtlas: () => void;
   242→  onJumpChapter: (chapterId: string) => void;
   243→  seen: ReadonlySet<string>;
   244→}) {
   245→  const [active, setActive] = useState<string>("");
   246→  const [openGroup, setOpenGroup] = useState<string | null>(null);
   247→  const [mobileOpen, setMobileOpen] = useState(false);
   248→  const mobileRef = useDialogA11y<HTMLDivElement>(mobileOpen);
   249→  const [pyReady, setPyReady] = useState(false);
   250→  const navRef = useRef<HTMLElement>(null);
   251→  const progressRef = useRef<HTMLDivElement>(null);
   252→  const activeRef = useRef("");
   253→
   254→  useEffect(() => {
   255→    let ticking = false;
   256→    const update = () => {
   257→      ticking = false;
   258→      const doc = document.documentElement;
   259→      const total = doc.scrollHeight - doc.clientHeight;
   260→      const pct = total > 0 ? (doc.scrollTop / total) * 100 : 0;
   261→      if (progressRef.current) progressRef.current.style.width = `${pct}%`;
   262→
   263→      // scroll-spy: only touch React state when the section changes
   264→      const probe = window.innerHeight * 0.32;
   265→      let current = "";
   266→      for (const item of FLAT_ITEMS) {
   267→        const el = document.getElementById(item.id);
   268→        if (el && el.getBoundingClientRect().top <= probe) current = item.id;
   269→      }
   270→      if (current !== activeRef.current) {
   271→        activeRef.current = current;
   272→        setActive(current);
   273→      }
   274→    };
   275→    const onScroll = () => {
   276→      if (!ticking) {
   277→        ticking = true;
   278→        requestAnimationFrame(update);
   279→      }
   280→    };
   281→    update();
   282→    window.addEventListener("scroll", onScroll, { passive: true });
   283→    window.addEventListener("resize", onScroll);
   284→    return () => {
   285→      window.removeEventListener("scroll", onScroll);
   286→      window.removeEventListener("resize", onScroll);
   287→    };
   288→  }, []);
   289→
   290→  // shared Python runtime status
   291→  useEffect(() => {
   292→    const onReady = () => setPyReady(true);
   293→    window.addEventListener("foc-py-ready", onReady);
   294→    return () => window.removeEventListener("foc-py-ready", onReady);
   295→  }, []);
   296→
   297→  // close dropdown on outside click / Escape
   298→  useEffect(() => {
   299→    const onDown = (e: MouseEvent) => {
   300→      if (navRef.current && !navRef.current.contains(e.target as Node)) setOpenGroup(null);
   301→    };
   302→    const onKey = (e: KeyboardEvent) => {
   303→      if (e.key === "Escape") {
   304→        setOpenGroup(null);
   305→        setMobileOpen(false);
   306→      }
   307→    };
   308→    document.addEventListener("mousedown", onDown);
   309→    document.addEventListener("keydown", onKey);
   310→    return () => {
   311→      document.removeEventListener("mousedown", onDown);
   312→      document.removeEventListener("keydown", onKey);
   313→    };
   314→  }, []);
   315→
   316→  // lock body scroll while the mobile index is open
   317→  useEffect(() => {
   318→    document.body.style.overflow = mobileOpen ? "hidden" : "";
   319→    return () => {
   320→      document.body.style.overflow = "";
   321→    };
   322→  }, [mobileOpen]);
   323→
   324→  const activeGroup = active ? groupOf(active) : null;
   325→
   326→  return (
   327→    <nav ref={navRef} aria-label="Chapters" className="fixed inset-x-0 top-0 z-50">
   328→      <div className="nav-shell">
   329→        <div className="mx-auto flex h-14 max-w-7xl items-center gap-2 px-3 sm:px-5">
   330→          {/* back to The Atlas — the shelf this subject lives on */}
   331→          <button
   332→            type="button"
   333→            onClick={onExitAtlas}
   334→            title="Back to The Atlas — the subject shelf"
   335→            aria-label="Back to The Atlas"
   336→            className="relative flex h-10 shrink-0 items-center gap-1.5 rounded-full after:absolute after:-inset-y-0.5 after:-inset-x-1 after:rounded-full after:content-[''] border border-white/70 bg-white/60 px-3 text-[.64rem] font-extrabold uppercase tracking-[.14em] text-[#8a5f30] shadow-[0_4px_16px_-8px_rgba(70,42,15,.35)] backdrop-blur-md transition-colors duration-200 hover:bg-white/85 hover:text-[#c2410c]"
   337→          >
   338→            <Map className="h-3.5 w-3.5" />
   339→            <span className="hidden sm:inline">Atlas</span>
   340→          </button>
   341→
   342→          {/* brand — quiet glass chip, Apple-quiet rather than loud gradient */}
   343→          <button type="button" aria-label="Scroll back to the top of these notes" onClick={() => scrollToId("top")} className="relative flex h-10 shrink-0 items-center gap-2 rounded-full border border-white/70 bg-white/60 pl-2 pr-3.5 after:absolute after:-inset-y-0.5 after:-inset-x-1 after:rounded-full after:content-[''] shadow-[0_4px_16px_-8px_rgba(70,42,15,.35)] backdrop-blur-md transition-colors duration-200 hover:bg-white/80 cursor-pointer">
   344→            <span
   345→              className="flex h-6 w-6 items-center justify-center rounded-full text-white"
   346→              style={{ background: `linear-gradient(135deg,${EMBER_BRIGHT},${EMBER})`, boxShadow: "0 4px 10px -3px rgba(194,65,12,.55)" }}
   347→            >
   348→              <Compass className="h-3.5 w-3.5" />
   349→            </span>
   350→            <span className="hidden min-[380px]:inline text-sm font-black tracking-tight text-[#3e2c1d]">FoC Notes</span>
   351→          </button>
   352→
   353→          {/* desktop — grouped chapter menu (all four groups, Python Core
   354→              included). Opens at lg: below that the bar would overflow once
   355→              the Python Core group is shown, so narrower screens get the
   356→              scrolling chip strip + the full-screen index instead. */}
   357→          <div className="ml-2 hidden flex-1 items-center gap-1 lg:flex">
   358→            {GROUPS.map((group) => {
   359→              const isActive = activeGroup === group.id;
   360→              const open = openGroup === group.id;
   361→              return (
   362→                <div
   363→                  key={group.id}
   364→                  className="relative"
   365→                  onMouseEnter={() => setOpenGroup(group.id)}
   366→                  onMouseLeave={() => setOpenGroup((g) => (g === group.id ? null : g))}
   367→                >
   368→                  <button
   369→                    type="button"
   370→                    aria-expanded={open}
   371→                    aria-haspopup="true"
   372→                    onClick={() => setOpenGroup(group.id)}
   373→                    onFocus={() => setOpenGroup(group.id)}
   374→                    className={`flex items-center gap-1.5 rounded-full px-3.5 py-1.5 text-[.78rem] font-semibold tracking-wide transition-all duration-200 ${
   375→                      isActive || open
   376→                        ? "bg-white/80 text-[#c2410c] shadow-[0_4px_16px_-8px_rgba(70,42,15,.4),inset_0_1px_0_rgba(255,255,255,.9)]"
   377→                        : "text-[#6b563d] hover:bg-white/55 hover:text-[#3e2c1d] hover:shadow-[inset_0_1px_0_rgba(255,255,255,.9)]"
   378→                    }`}
   379→                  >
   380→                    {group.label}
   381→                    <span className="text-[.58rem] font-extrabold opacity-45">{String(group.items.length).padStart(2, "0")}</span>
   382→                    <ChevronDown className={`h-3 w-3 opacity-50 transition-transform duration-200 ${open ? "rotate-180" : ""}`} />
   383→                  </button>
   384→
   385→                  <AnimatePresence>
   386→                    {open ? (
   387→                      <motion.div
   388→                        key={`${group.id}-panel`}
   389→                        initial={{ opacity: 0, y: 8, scale: 0.98 }}
   390→                        animate={{ opacity: 1, y: 0, scale: 1 }}
   391→                        exit={{ opacity: 0, y: 6, scale: 0.98 }}
   392→                        transition={{ duration: 0.16, ease: "easeOut" }}
   393→                        className="absolute left-1/2 top-full z-50 w-[min(26rem,calc(100vw-2rem))] -translate-x-1/2 pt-2"
   394→                      >
   395→                        <div
   396→                          className="overflow-hidden rounded-2xl border border-white/60 bg-white/70 p-1.5 backdrop-blur-2xl"

   907→          <span className="rounded-2xl border border-dashed border-[#e8dcc0] p-4 text-center text-[.78rem] font-bold leading-snug text-[#a08568]">
   908→            End of the sequence — Lecture 28 closes the recorded deck, and Classes waits just beyond it. Revise, then retake the windows.
   909→          </span>
   910→        )}
   911→      </div>
   912→    </nav>
   913→  );

   916→/* ============================================================
   917→   FocApp — the composed subject page: a lecture reader.
   918→   ============================================================ */
   919→export function FocApp({ onExitAtlas }: { onExitAtlas: () => void }) {
   920→  const [lectureId, setLectureId] = useState<string>(FOC_DEFAULT_LECTURE);
   921→  const [examGroup, setExamGroup] = useState<ExamGroup | null>(null);
   922→  const [hubOpen, setHubOpen] = useState(false);
   923→  const lectureIdRef = useRef(lectureId);
   924→  useEffect(() => {
   925→    lectureIdRef.current = lectureId;
   926→  }, [lectureId]);
   927→
   928→  const lecture = lectureById(lectureId) ?? FOC_LECTURES[0];
   929→  const seen = useSeenDisplay(lecture);
   930→
   931→  const selectLecture = useCallback((id: string) => {
   932→    setLectureId(id);
   933→    requestAnimationFrame(() =>
   934→      requestAnimationFrame(() => {
   935→        const el = document.getElementById("notes-top");
   936→        if (!el) return;
   937→        const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   938→        el.scrollIntoView({ behavior: reduce ? "auto" : "smooth", block: "start" });
   939→      })
   940→    );
   941→  }, []);
   942→
   943→  const goToChapter = useCallback((chapterId: string) => {
   944→    const target = lectureOfChapter(chapterId);
   945→    if (!target) return;
   946→    const scrollWhenReady = () => {
   947→      let frames = 0;
   948→      const tick = () => {
   949→        const el = document.getElementById(chapterId);
   950→        if (el) {
   951→          jumpTo(el);
   952→          return;
   953→        }
   954→        if (++frames < 240) requestAnimationFrame(tick);
   955→      };
   956→      requestAnimationFrame(tick);
   957→    };
   958→    if (lectureIdRef.current === target.id) {
   959→      scrollWhenReady();
   960→      return;
   961→    }
   962→    setLectureId(target.id);
   963→    requestAnimationFrame(() => requestAnimationFrame(scrollWhenReady));
   964→  }, []);
   965→
   966→  useOpenSection("foc", goToChapter);
   967→
   968→  const openExam = useCallback((g: ExamGroup) => setExamGroup(g), []);
   969→  const openHub = useCallback(() => setHubOpen(true), []);
   970→
   971→  /* the ⌘K palette may have left a lab handoff flag before this
   972→     app mounted — read it once at init (client-only component) */
   973→  const [labOpen, setLabOpen] = useState(() => {
   974→    try {
   975→      if (sessionStorage.getItem("atlas:open-lab") === "1") {
   976→        sessionStorage.removeItem("atlas:open-lab");
   977→        return true;
   978→      }
   979→    } catch {
   980→      /* private mode — flag unreadable, start closed */
   981→    }
   982→    return false;
   983→  });
   984→  const openLab = useCallback(() => setLabOpen(true), []);
   985→
   986→  /* once FoC is on stage the palette just broadcasts this event */
   987→  useEffect(() => {
   988→    const onOpenLab = () => setLabOpen(true);
   989→    window.addEventListener("atlas:open-lab", onOpenLab);
   990→    return () => window.removeEventListener("atlas:open-lab", onOpenLab);
   991→  }, []);
   992→
   993→  return (
   994→    // id="top" — the brand chip links to #top; without a real element here the
   995→    // client-side router swallows the fragment and the page never scrolls back
   996→    <div id="top" className="page-wrap warm-canvas">
   997→      <TopNav onOpenLab={openLab} onOpenExam={openExam} onExitAtlas={onExitAtlas} onJumpChapter={goToChapter} seen={seen} />
   998→      <main className="pt-14">
   999→        <PartBoundary label="Hero">
  1000→          <FocHero
  1001→            activeId={lectureId}
  1002→            onSelect={selectLecture}
  1003→            onPractice={openHub}
  1004→            onOpenLab={openLab}
  1005→            seen={seen}
  1006→          />
  1007→        </PartBoundary>
  1008→
  1009→        {/* the notes start here */}
  1010→        <div id="notes-top" aria-hidden className="relative scroll-mt-16" />
  1011→
  1012→        <PartBoundary key={lecture.id} label={`Lecture ${lecture.code} — ${lecture.title}`}>
  1013→          <motion.div
  1014→            initial={{ opacity: 0, y: 18 }}
  1015→            animate={{ opacity: 1, y: 0 }}
  1016→            transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
  1017→          >
  1018→            <FocLectureHead lecture={lecture} />
  1019→            {lecture.chapters.map((c) => (
  1020→              <PartBoundary key={c.id} label={c.label}>
  1021→                {SECTION_COMPONENTS[c.id]}
  1022→              </PartBoundary>
  1023→            ))}
  1024→            <LecturePager current={lecture} onJump={selectLecture} />
  1025→          </motion.div>
  1026→        </PartBoundary>
  1027→      </main>
  1028→      <PyReadyToast />
  1029→      <PythonLab open={labOpen} onClose={() => setLabOpen(false)} />
  1030→      <PracticeHub open={hubOpen} onClose={() => setHubOpen(false)} onOpenExam={openExam} />
  1031→      {examGroup ? <ExamWindow group={examGroup} onClose={() => setExamGroup(null)} /> : null}
  1032→    </div>
  1033→  );
  1034→}
