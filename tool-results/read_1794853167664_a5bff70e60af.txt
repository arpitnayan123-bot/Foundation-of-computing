   193→function OdometerLab() {
   194→  const [reading, setReading] = useState(0);
   195→  const [log, setLog] = useState<string[]>(["__init__() → odometer_reading = 0"]);
   196→
   197→  const push = (msg: string) => setLog((l) => [msg, ...l].slice(0, 5));
   198→
   199→  const direct = () => {
   200→    setReading(23);
   201→    push("my_new_car.odometer_reading = 23   → read_odometer(): 'This car has 23 miles on it.'");
   202→  };
   203→  const update = (v: number) => {
   204→    if (v >= reading) {
   205→      setReading(v);
   206→      push(`update_odometer(${v}) → mileage >= reading → set → 'This car has ${v} miles on it.'`);
   207→    } else {
   208→      push(`update_odometer(${v}) → ⚠ "You can't roll back an odometer!"`);
   209→    }
   210→  };
   211→  const increment = (m: number) => {
   212→    setReading((r) => r + m);
   213→    push(`increment_odometer(${m}) → reading += ${m} → now ${reading + m} miles`);
   214→  };
   215→
   216→  const rejected = log[0]?.includes("roll back");
   217→
   218→  return (
   219→    <div className="card-warm overflow-hidden p-6 md:p-8">
   220→      <div className="flex items-center gap-2">
   221→        <Gauge className="h-5 w-5" style={{ color: "#c2410c" }} />
