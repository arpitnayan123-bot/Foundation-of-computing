    42→function Rise({
    43→  children,
    44→  delay = 0,
    45→  className = "",
    46→  y = 26,
    47→}: {
    48→  children: React.ReactNode;
    49→  delay?: number;

    51→  y?: number;
    52→}) {

    54→    <motion.div

    59→      transition={{ duration: 0.8, delay, ease: EASE }}
    60→    >

    64→}

    77→/* ---------------------------------------------- count-up stat */
    78→function Stat({
    79→  value,
    80→  render,
    81→  label,
    82→  delay = 0,
    83→}: {
    84→  value: number;
    85→  render?: (n: number) => React.ReactNode;
    86→  label: string;
    87→  delay?: number;
    88→}) {
    89→  const ref = useRef<HTMLDivElement>(null);
    90→  const inView = useInView(ref, { once: true, margin: "-40px" });
    91→  const [n, setN] = useState(0);
    92→
    93→  useEffect(() => {
    94→    if (!inView) return;
    95→    const controls = animate(0, value, {
    96→      duration: 1.5,
    97→      delay,
    98→      ease: EASE,
    99→      onUpdate: (v) => setN(Math.round(v)),
   100→    });
   101→    return () => controls.stop();
   102→  }, [inView, value, delay]);
   103→
   104→  return (
   105→    <div ref={ref} className="flex flex-col items-center gap-1">
   106→      <span className="font-display text-3xl font-black tracking-tight text-[#f7ecdd] md:text-4xl">
   107→        {render ? render(n) : n}
   108→      </span>
   109→      <span className="text-[.62rem] font-extrabold uppercase tracking-[.22em] text-[#8d7b63]">{label}</span>
   110→    </div>
   111→  );
   112→}
