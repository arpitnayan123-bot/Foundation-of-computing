   358→function TravTree({ seq, step, mini = false }: { seq: string[]; step: number; mini?: boolean }) {
   359→  const rank = (n: string) => seq.indexOf(n);
   360→  const seen = (n: string) => rank(n) > -1 && rank(n) < step;
   361→  return (
   362→    <svg
   363→      viewBox="0 0 440 292"
   364→      className="mx-auto h-auto w-full"
   365→      style={{ maxWidth: mini ? 180 : 380 }}
   366→      aria-hidden={mini || undefined}
   367→      role={mini ? undefined : "img"}
   368→      aria-label={mini ? undefined : "Tree A with children B and C; B has children D and E, showing the current traversal order"}
   369→    >
   370→      {T_EDGES.map(([a, b]) => (
   371→        <line
   372→          key={`${a}${b}`}
   373→          x1={T_POS[a].x}
   374→          y1={T_POS[a].y}
   375→          x2={T_POS[b].x}
   376→          y2={T_POS[b].y}
   377→          stroke="rgba(201,154,91,.65)"
   378→          strokeWidth="2.5"
   379→        />
   380→      ))}
   381→      {Object.entries(T_POS).map(([n, p]) => {
   382→        const on = seen(n);
   383→        const cur = rank(n) === step - 1;
   384→        return (
   385→          <g key={n}>
   386→            {cur ? <circle cx={p.x} cy={p.y} r="27" fill="none" stroke="#f59e0b" strokeWidth="3" /> : null}
   387→            <circle
   388→              cx={p.x}
   389→              cy={p.y}
   390→              r="20"
   391→              fill={on ? "#ea580c" : "rgba(255,246,232,.09)"}
   392→              stroke={on ? "#c2410c" : "rgba(245,158,11,.45)"}
   393→              strokeWidth="2"
   394→            />
   395→            <text
   396→              x={p.x}
   397→              y={p.y + 5}
   398→              textAnchor="middle"
   399→              fontSize="14"
   400→              fontWeight="800"
   401→              fill={on ? "#fff7ec" : "#f5e6cd"}
   402→            >
   403→              {n}
   404→            </text>
   405→            {on ? (
   406→              <g>
   407→                <circle cx={p.x + 24} cy={p.y - 24} r="11" fill="#2b1b10" stroke="#f5c889" strokeWidth="1.5" />
   408→                <text x={p.x + 24} y={p.y - 20} textAnchor="middle" fontSize="10" fontWeight="800" fill="#f5c889">
   409→                  {rank(n) + 1}
   410→                </text>
   411→              </g>
   412→            ) : null}
   413→          </g>
   414→        );
   415→      })}
   416→    </svg>
   417→  );
   418→}

   420→function TraversalLab() {
   421→  const [mode, setMode] = useState<TravKey>("pre");
   422→  const [step, setStep] = useState(5);
   423→  const seq = T_SEQ[mode];
   424→
