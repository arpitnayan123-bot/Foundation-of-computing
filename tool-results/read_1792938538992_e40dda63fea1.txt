    60→const STACK_CAP = 6;
    61→const STACK_SEQ = [64, 32, 53, 66, 54];
    62→type Piece = { id: number; v: number };
    63→
    64→function StackWidget() {
    65→  const [items, setItems] = useState<Piece[]>(STACK_SEQ.map((v, i) => ({ id: i + 1, v })));
    66→  const [log, setLog] = useState("Push 64 → 32 → 53 → 66 → 54. Top = 54.");
    67→  const nid = useRef(STACK_SEQ.length);
    68→  const nseq = useRef(0);
    69→
    70→  function push() {
    71→    if (items.length >= STACK_CAP) return setLog("Push() ✗ STACK OVERFLOW — the stack is FULL (6 of 6).");
    72→    const v = STACK_SEQ[nseq.current % STACK_SEQ.length];
    73→    nseq.current += 1; nid.current += 1;
    74→    setItems((xs) => [...xs, { id: nid.current, v }]);
    75→    setLog(`Push(${v}) → ${v} lands ON TOP. Top = ${v}.`);
    76→  }
    77→  function pop() {
    78→    if (items.length === 0) return setLog("Pop() ✗ STACK UNDERFLOW — the stack is EMPTY, nothing to remove.");
    79→    const t = items[items.length - 1];
    80→    setItems((xs) => xs.slice(0, -1));
    81→    setLog(`Pop() → ${t.v} lifts off. ${items.length - 1 > 0 ? `New top = ${items[items.length - 2].v}.` : "The stack is now empty."}`);
    82→  }
    83→  const peek = () => setLog(items.length ? `Peek() → ${items[items.length - 1].v} (reads the top, removes nothing)` : "Peek() → the stack is empty.");
    84→  const size = () => setLog(`Size() → ${items.length}`);
    85→  const reset = () => { setItems(STACK_SEQ.map((v, i) => ({ id: i + 1, v }))); nseq.current = 0; setLog("Reset — Push 64 → 32 → 53 → 66 → 54. Top = 54."); };
    86→
    87→  return (
    88→    <div className="card-warm p-6 md:p-8">
    89→      <div className="flex flex-wrap items-center gap-3">
    90→        <Layers className="h-6 w-6" style={{ color: "#c2410c" }} />
    91→        <h3 className="text-xl font-bold md:text-2xl">Last-In, First-Out — try the machine</h3>
    92→        <span className="chip">like a pile of plates — last placed, first picked</span>
    93→      </div>
    94→      <div className="mt-5 grid gap-6 lg:grid-cols-[auto,1fr]">
    95→        <div>
    96→          <div className="relative">
    97→            <div className="flex flex-col-reverse gap-1.5" aria-hidden>
    98→              {Array.from({ length: STACK_CAP }).map((_, i) => <div key={i} className="h-11 w-44 rounded-xl border-2 border-dashed sm:w-52" style={{ borderColor: "#e8d5b5" }} />)}
    99→            </div>
   100→            <div className="absolute inset-0 flex flex-col-reverse gap-1.5">
   101→              <AnimatePresence initial={false}>
   102→                {items.map((p, i) => (
   103→                  <motion.div key={p.id} initial={{ y: -44, opacity: 0 }} animate={{ y: 0, opacity: 1 }} exit={{ y: -46, opacity: 0, scale: 0.92 }} transition={{ type: "spring", stiffness: 380, damping: 28 }}
   104→                    className="flex h-11 w-44 items-center justify-center rounded-xl text-[1.05rem] font-extrabold text-white shadow-md sm:w-52"
   105→                    style={{ background: i === items.length - 1 ? "linear-gradient(135deg,#ea580c,#c2410c)" : "linear-gradient(135deg,#e8a13c,#d97706)" }}>
   106→                    {p.v}
   107→                    {i === items.length - 1 ? <span className="ml-2 rounded-md bg-black/25 px-1.5 py-0.5 text-[.62rem] font-bold">top</span> : null}
   108→                  </motion.div>
   109→                ))}
   110→              </AnimatePresence>
   111→            </div>
   112→          </div>
   113→          <p className="mt-2 text-center text-[.72rem] font-bold uppercase tracking-[.14em] text-[#8a5a06]">↑ Push enters here · Pop leaves from here ↑</p>
   114→        </div>
   115→        <div>
   116→          <div className="flex flex-wrap gap-2">
   117→            <OpBtn tone="warm" onClick={push}>Push()</OpBtn>
   118→            <OpBtn tone="warm" onClick={pop}>Pop()</OpBtn>
   119→            <OpBtn onClick={peek}>Peek()</OpBtn>
   120→            <OpBtn onClick={size}>Size()</OpBtn>
   121→            <OpBtn onClick={reset}><RotateCcw className="mr-1 inline h-3.5 w-3.5" />reset</OpBtn>
   122→          </div>
   123→          <LogLine log={log} />
   124→          <p className="mt-3 text-[.88rem] leading-relaxed text-[#2b2014]">
   125→            The lecture&apos;s run: push <span className="font-mono-warm font-bold">64, 32, 53, 66, 54</span> → top
   126→            is <strong>54</strong>. <span className="font-mono-warm font-bold">Pop()</span> → 54 leaves,{" "}
   127→            <span className="font-mono-warm font-bold">Pop()</span> → 66 leaves,{" "}
   128→            <span className="font-mono-warm font-bold">Peek()</span> → <strong>53</strong>,{" "}
   129→            <span className="font-mono-warm font-bold">Size()</span> → <strong>3</strong>. The last value in
   130→            is always the first value out — <strong>LIFO</strong>.

   165→const Q_CAP = 6;
   166→const Q_SEQ = [64, 32, 53, 66, 54];
   167→type QPiece = { id: number; v: number };
   168→
   169→function QueueWidget() {
   170→  const [items, setItems] = useState<QPiece[]>(Q_SEQ.map((v, i) => ({ id: i + 1, v })));
   171→  const [log, setLog] = useState("Enqueue 64 → 32 → 53 → 66 → 54. Front = 64, Rear = 54.");
   172→  const nid = useRef(Q_SEQ.length);
   173→  const nseq = useRef(0);
   174→
   175→  function enqueue() {
   176→    if (items.length >= Q_CAP) return setLog("Enqueue() ✗ QUEUE OVERFLOW — the queue is FULL (6 of 6).");
   177→    const v = Q_SEQ[nseq.current % Q_SEQ.length];
   178→    nseq.current += 1; nid.current += 1;
   179→    setItems((xs) => [...xs, { id: nid.current, v }]);
   180→    setLog(`Enqueue(${v}) → ${v} joins at the REAR.`);
   181→  }
   182→  function dequeue() {
   183→    if (items.length === 0) return setLog("Dequeue() ✗ QUEUE UNDERFLOW — the queue is EMPTY, nobody to serve.");
   184→    const f = items[0];
   185→    setItems((xs) => xs.slice(1));
   186→    setLog(`Dequeue() → ${f.v} leaves from the FRONT. ${items.length - 1 > 0 ? `New front = ${items[1].v}.` : "The queue is now empty."}`);
   187→  }
   188→  const peek = () => setLog(items.length ? `Peek()/Front() → ${items[0].v} · Rear() → ${items[items.length - 1].v}` : "Peek() → the queue is empty.");
   189→  const size = () => setLog(`Size() → ${items.length}`);
   190→  const reset = () => { setItems(Q_SEQ.map((v, i) => ({ id: i + 1, v }))); nseq.current = 0; setLog("Reset — Enqueue 64 → 32 → 53 → 66 → 54. Front = 64, Rear = 54."); };
   191→
   192→  const front = items[0]?.v, rear = items[items.length - 1]?.v;
   193→  return (
   194→    <div className="card-warm p-6 md:p-8">
   195→      <div className="flex flex-wrap items-center gap-3">
   196→        <ListMusic className="h-6 w-6" style={{ color: "#c2410c" }} />
   197→        <h3 className="text-xl font-bold md:text-2xl">Front and rear — try the line</h3>
   198→        <span className="chip">FCFS · First-Come-First-Served = FIFO</span>
   199→      </div>
   200→      <div className="mt-5 flex flex-wrap items-center gap-3 text-[.78rem] font-extrabold text-[#8a5a06]">
   201→        <span className="flex items-center gap-1"><ArrowLeft className="h-4 w-4" style={{ color: "#c2410c" }} aria-hidden /> Front — Dequeue leaves here</span>
   202→        <span className="flex items-center gap-1">Rear — Enqueue enters here <ArrowRight className="h-4 w-4" style={{ color: "#c2410c" }} aria-hidden /></span>
   203→        <span className="rounded-full bg-[#fde8cf] px-2.5 py-1 font-mono-warm text-[.74rem] font-bold text-[#7c4a12]">Front → {front ?? "—"} · Rear → {rear ?? "—"} · Size {items.length}</span>
   204→      </div>
   205→      <div className="relative mt-3">
   206→        <div className="grid grid-cols-6 gap-2" aria-hidden>
   207→          {Array.from({ length: Q_CAP }).map((_, i) => <div key={i} className="h-12 rounded-xl border-2 border-dashed" style={{ borderColor: "#e8d5b5" }} />)}
   208→        </div>
   209→        <div className="absolute inset-0 grid grid-cols-6 gap-2">
   210→          <AnimatePresence initial={false}>
   211→            {items.map((p, i) => (
   212→              <motion.div key={p.id} initial={{ x: 46, opacity: 0 }} animate={{ x: 0, opacity: 1 }} exit={{ x: -46, opacity: 0, scale: 0.92 }} transition={{ type: "spring", stiffness: 380, damping: 28 }}
   213→                className="flex h-12 items-center justify-center rounded-xl text-[1rem] font-extrabold text-white shadow-md"
   214→                style={{ background: i === 0 ? "linear-gradient(135deg,#ea580c,#c2410c)" : "linear-gradient(135deg,#e8a13c,#d97706)", gridColumnStart: i + 1 }}>
   215→                {p.v}
   216→              </motion.div>
   217→            ))}
   218→          </AnimatePresence>
   219→        </div>
   220→      </div>
   221→      <div className="mt-4 flex flex-wrap gap-2">
   222→        <OpBtn tone="warm" onClick={enqueue}>Enqueue()</OpBtn>
   223→        <OpBtn tone="warm" onClick={dequeue}>Dequeue()</OpBtn>
   224→        <OpBtn onClick={peek}>Peek / Front · Rear</OpBtn>
   225→        <OpBtn onClick={size}>Size()</OpBtn>
   226→        <OpBtn onClick={reset}><RotateCcw className="mr-1 inline h-3.5 w-3.5" />reset</OpBtn>
   227→      </div>
   228→      <LogLine log={log} />
   229→      <p className="mt-3 text-[.88rem] leading-relaxed text-[#2b2014]">
   230→        The lecture&apos;s run: Enqueue <span className="font-mono-warm font-bold">64, 32, 53, 66, 54</span> fills
   231→        left → right. <span className="font-mono-warm font-bold">Dequeue()</span> removes <strong>64</strong> (the
   232→        front — it arrived first), again removes <strong>32</strong>. Then{" "}
   233→        <span className="font-mono-warm font-bold">Peek</span> → <strong>53</strong>, Rear → <strong>54</strong>,
   234→        Size → <strong>3</strong>. Your <strong>order doesn&apos;t matter — your PLACE does</strong>.
   235→      </p>
   236→    </div>
   237→  );
