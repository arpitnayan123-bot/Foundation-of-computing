     3→/* ============================================================
     4→   TEST MY KNOWLEDGE — THE ONE EXAM ENGINE.
     5→
     6→   All five shelves (FoC / ATB / FA / EBH / STM) used to carry a
     7→   550-line clone of this window each. Every difference between
     8→   them was pure presentation — colors, labels, copy — so the
     9→   engine now takes an `ExamTheme` and a plain bank object, and
    10→   each shelf ships a ~70-line wrapper that only declares its
    11→   theme. Logic (drafts, drift-proof clock, focus trap, scoring)
    12→   lives here and is shared.
    13→
    14→   Question shape is structural, not nominal: any bank whose
    15→   questions carry { id, type, q, options?, answer?, code?,
    16→   answerText?, solution?, explain? } plugs straight in.
    17→   ============================================================ */

    99→
   100→export interface EngineQuestion {
   101→  id: string;
   102→  type: string;
   103→  q: string;
   104→  code?: string;
   105→  options?: string[];
   106→  answer?: number;
   107→  answerText?: string;
   108→  solution?: string;
   109→  explain?: string;
   110→  /** schedule this question grades into when it differs from theme.key
   111→   *  (cross-shelf review sessions mix subjects in one paper) */
   112→  subject?: string;
   113→  /** the question's id in its own subject's schedule, when `id` was
   114→   *  namespaced to stay unique inside a mixed bank */
   115→  srcId?: string;
   116→}
   117→
   118→export interface EngineBank {
   119→  title: string;
   120→  subtitle: string;
   121→  questions: EngineQuestion[];
   122→}
   123→
   124→type QStatus = "unattempted" | "correct" | "wrong" | "revealed";
   125→interface QState {
   126→  chosen?: number;
   127→  typed?: string;
   128→  revealed?: boolean;
   129→  status: QStatus;
   130→}
   131→
   132→function fmt(sec: number) {
   133→  const m = Math.floor(sec / 60);
   134→  const s = sec % 60;
   135→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
   136→}
   137→
   138→export function ExamEngine({
   139→  theme,
   140→  bank,
   141→  draftId,
   142→  onClose,
   143→}: {
   144→  theme: ExamTheme;
   145→  bank: EngineBank;
   146→  /** stable per-window id (the exam group) — namespaces the draft */
   147→  draftId: string;
   148→  onClose: () => void;
   149→}) {
   150→  const qs = bank.questions;
   151→
   152→  /* ---- draft restore: a refresh or an accidental Escape keeps your work ---- */
   153→  const DRAFT_KEY = examDraftKey(theme.key, draftId);
   154→  const draft = useMemo(() => loadExamDraft(DRAFT_KEY), [DRAFT_KEY]);
   155→  const restored = useMemo(() => {
   156→    if (!draft) return null;
   157→    let n = 0;
   158→    for (const s of Object.values(draft.states)) {
   159→      const st = s as QState | undefined;
   160→      if (st && (st.status === "correct" || st.status === "wrong" || st.revealed)) n += 1;
   161→    }
   162→    return n > 0 ? n : null;
   163→  }, [draft]);
   164→
   165→  const [idx, setIdx] = useState(() => Math.min(Math.max(draft?.idx ?? 0, 0), qs.length - 1));
   166→  const [states, setStates] = useState<Record<string, QState>>(() => {
   167→    const fresh: Record<string, QState> = Object.fromEntries(
   168→      bank.questions.map((q) => [q.id, { status: "unattempted" as QStatus }])
   169→    );
   170→    if (draft) {
   171→      for (const q of bank.questions) {
   172→        const d = draft.states[q.id] as QState | undefined;
   173→        if (d && typeof d.status === "string") fresh[q.id] = d;
   174→      }
   175→    }
   176→    return fresh;
   177→  });
   178→  const [finished, setFinished] = useState(false);
   179→  /* drift-proof clock: seconds = base + (now − startedAt), derived from the
   180→     wall clock — never tick-counted, so background-tab throttling can't
   181→     silently under-report the elapsed time. Survives refresh via the draft. */
   182→  const [clock, setClock] = useState(() => ({
   183→    base: draft?.base ?? 0,
   184→    startedAt: draft?.startedAt ?? Date.now(),
   185→  }));
   186→  const [now, setNow] = useState(() => Date.now());
   187→  const seconds = Math.max(0, Math.floor(clock.base + (now - clock.startedAt) / 1000));
   188→  const bodyRef = useRef<HTMLDivElement>(null);
   189→  /* one ledger row per attempt-cycle — "back to questions" then finishing
   190→     again must not double-log; retake() re-arms it */
   191→  const loggedRef = useRef(false);
   192→
   193→  /* scroll lock (the timer below stops ticking on the results view) */
   194→  useEffect(() => {
   195→    document.body.style.overflow = "hidden";
   196→    return () => {
   197→      document.body.style.overflow = "";
   198→    };
   199→  }, []);
   200→
   201→  useEffect(() => {
   202→    if (finished) return;
   203→    const t = window.setInterval(() => setNow(Date.now()), 1000);
   204→    return () => window.clearInterval(t);
   205→  }, [finished]);
   206→
   207→  /* draft autosave — debounced inside saveExamDraft, flushed on close/finish */
   208→  useEffect(() => {
   209→    if (finished) return;
   210→    if (draftHasWork(states as Record<string, unknown>)) {
   211→      saveExamDraft(DRAFT_KEY, {
   212→        idx,
   213→        states: states as Record<string, unknown>,
   214→        base: clock.base,
   215→        startedAt: clock.startedAt,
   216→      });
   217→    }
   218→  }, [states, idx, clock, finished, DRAFT_KEY]);
   219→
   220→  const closeWindow = useCallback(() => {
   221→    flushExamDraft(); // the attempt is kept — reopening this window restores it
   222→    onClose();
   223→  }, [onClose]);
   224→
   225→  /** freeze the clock, clear the draft, log the attempt, show the report */
   226→  function finish() {
   227→    flushExamDraft();
   228→    clearExamDraft(DRAFT_KEY);
   229→    setClock({ base: seconds, startedAt: Date.now() });
   230→    setFinished(true);
   231→    if (!loggedRef.current) {
   232→      loggedRef.current = true;
   233→      /* one row per subject: single-shelf windows log exactly one row
   234→         (subject = theme.key, unchanged behaviour); mixed review sessions
   235→         split into per-shelf rows so each shelf's accuracy stays true */
   236→      const groups = new Map<string, { c: number; t: number }>();
   237→      for (const qq of qs) {
   238→        const s = states[qq.id]?.status;
   239→        if (s !== "correct" && s !== "wrong") continue;
   240→        const subj = qq.subject ?? theme.key;
   241→        const g = groups.get(subj) ?? { c: 0, t: 0 };
   242→        g.t += 1;
   243→        if (s === "correct") g.c += 1;
   244→        groups.set(subj, g);
   245→      }
   246→      const answered = groups.size > 0;
   247→      const rows: { subject: string; correct: number; total: number }[] = answered
   248→        ? [...groups.entries()].map(([subject, g]) => ({ subject, correct: g.c, total: g.t }))
   249→        : [{ subject: theme.key, correct: score.correct, total: score.total }];
   250→      for (const r of rows) {
   251→        logAttempt({
   252→          subject: r.subject,
   253→          group: draftId,
   254→          title: bank.title,
   255→          correct: r.correct,
   256→          total: r.total,
   257→          pct: Math.round((r.correct / Math.max(1, r.total)) * 100),
   258→          seconds,
   259→        });
   260→      }
   261→    }
   262→  }
   263→
   264→  const q = qs[idx];
   265→  const st = states[q.id];
   266→
   267→  /* keyboard flow — ←/→ move, keys 1–4 answer the MCQ on stage,
   268→     ↵ advances (next question, or the report on the last one).
   269→     Typing surfaces keep their keys: textareas, the ⌘K search
   270→     input, and focused buttons (where ↵ already means "press"). */
   271→  useEffect(() => {
   272→    const onKey = (e: KeyboardEvent) => {
   273→      const t = e.target;
   274→      const inTextarea = t instanceof HTMLTextAreaElement;
   275→      if (e.key === "Escape" && !inTextarea) closeWindow();
   276→      if (finished || inTextarea || e.isComposing || e.metaKey || e.ctrlKey || e.altKey) return;
   277→      if (t instanceof HTMLInputElement) return; // palette search etc.
   278→      if (e.key === "ArrowRight") {
   279→        setIdx((i) => Math.min(i + 1, qs.length - 1));
   280→      } else if (e.key === "ArrowLeft") {
   281→        setIdx((i) => Math.max(i - 1, 0));
   282→      } else if (e.key === "Enter") {
   283→        if (e.repeat) return;
   284→        if (t instanceof HTMLButtonElement || t instanceof HTMLAnchorElement || t instanceof HTMLSelectElement) return;
   285→        if (idx < qs.length - 1) go(idx + 1);
   286→        else finish();
   287→      } else if (!e.repeat) {
   288→        // keys 1–4 pick an option — same contract as clicking the row
   289→        const n = Number(e.key);
   290→        if (!Number.isInteger(n) || n < 1 || n > 4) return;
   291→        if (q.type !== "mcq" || st.status === "correct" || st.status === "wrong") return;
   292→        if (n <= (q.options?.length ?? 0)) chooseMCQ(n - 1);
   293→      }
   294→    };
   295→    document.addEventListener("keydown", onKey);
   296→    return () => document.removeEventListener("keydown", onKey);
   297→  }, [finished, closeWindow, qs.length, idx, q, st]);
   298→
   299→  const patch = (id: string, p: Partial<QState>) =>
   300→    setStates((prev) => ({ ...prev, [id]: { ...prev[id], ...p } }));
   301→
   302→  /* scoring — byType rows follow the theme's declared type order */
   303→  const typeKeys = useMemo(() => Object.keys(theme.typeMeta), [theme]);
   304→  const score = useMemo(() => {
   305→    let correct = 0;
   306→    let wrong = 0;
   307→    let answered = 0;
   308→    const byType: Record<string, { c: number; t: number }> = {};
   309→    for (const t of typeKeys) byType[t] = { c: 0, t: 0 };
   310→    for (const qq of qs) {
   311→      const row = (byType[qq.type] ??= { c: 0, t: 0 });
   312→      row.t += 1;
   313→      const s = states[qq.id]?.status ?? "unattempted";
   314→      if (s === "correct" || s === "wrong") answered += 1;
   315→      if (s === "correct") {
   316→        correct += 1;
   317→        row.c += 1;
   318→      }
   319→      if (s === "wrong") wrong += 1;
   320→    }
   321→    return { correct, wrong, answered, byType, total: qs.length };
   322→  }, [states, qs, typeKeys]);
   323→
   324→  const pct = Math.round((score.correct / score.total) * 100);
   325→  const verdict =
   326→    pct >= 85 ? theme.tiers.topper : pct >= 60 ? theme.tiers.solid : pct >= 35 ? theme.tiers.getting : theme.tiers.restart;
   327→
   328→  function chooseMCQ(i: number) {
   329→    if (st.status === "correct" || st.status === "wrong") return;
   330→    const ok = i === q.answer;
   331→    patch(q.id, {
   332→      chosen: i,
   333→      revealed: true,
   334→      status: ok ? "correct" : "wrong",
   335→    });
   336→    // SM-2: first-try MCQ answers are trusted fully (5), misses reset the card (2)
   337→    // — routed to the question's own shelf in mixed review sessions
   338→    gradeQuestion(q.subject ?? theme.key, q.srcId ?? q.id, ok ? 5 : 2);
   339→  }
   340→
   341→  function selfCheck(ok: boolean) {
   342→    patch(q.id, { status: ok ? "correct" : "wrong" });
   343→    // SM-2: the model answer was visible — grade honestly but trust less
   344→    gradeQuestion(q.subject ?? theme.key, q.srcId ?? q.id, ok ? 3 : 1);
   345→  }
   346→
   347→  function retake() {
   348→    setStates(Object.fromEntries(qs.map((qq) => [qq.id, { status: "unattempted" as QStatus }])));
   349→    setIdx(0);
   350→    setClock({ base: 0, startedAt: Date.now() });
   351→    setNow(Date.now());
   352→    setFinished(false);
   353→    loggedRef.current = false; // a retake is a new, honest attempt
   354→    clearExamDraft(DRAFT_KEY);
   355→    bodyRef.current?.scrollTo({ top: 0 });
   356→  }
   357→
   358→  function go(i: number) {
   359→    setIdx(i);
   360→    bodyRef.current?.scrollTo({ top: 0, behavior: "smooth" });
   361→  }
   362→
   363→  /* ---------------- results view ---------------- */
   364→  if (finished) {
   365→    const mcqMeta = theme.typeMeta.mcq ?? Object.values(theme.typeMeta)[0];
   366→    return (
   367→      <ExamShell
   368→        theme={theme}
   369→        bank={bank}
   370→        score={score}
   371→        seconds={seconds}
   372→        onClose={closeWindow}
   373→        restoredCount={restored}
   374→        onExit={() => setFinished(false)}
   375→        scrollRef={bodyRef}
   376→      >
   377→        <motion.div initial={{ opacity: 0, y: 18 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }}>
   378→          <div className="exam-result-card">
   379→            <div className="exam-score-ring" style={{ borderColor: pct >= 60 ? "#8fae5d" : "#e0a45e" }}>
   380→              <span className="exam-score-num">{pct}%</span>
   381→              <span className="exam-score-sub">
   382→                {score.correct} / {score.total}
   383→              </span>
   384→            </div>
   385→            <div>
   386→              <h3 className="font-display text-2xl font-bold text-[#2b1d10]">Result report</h3>
   387→              <p className="mt-1 text-[.95rem] font-semibold" style={{ color: verdict.c }} role="status">
   388→                {verdict.t}
   389→              </p>
   390→              <div className="mt-3 flex flex-wrap gap-2 text-[.72rem] font-bold">
   391→                <span className="exam-chip" style={{ background: "#eef4dd", color: "#44621d" }}>✓ {score.correct} correct</span>
   392→                <span className="exam-chip" style={{ background: "#fdecec", color: "#8c2f2f" }}>✗ {score.wrong} wrong</span>
   393→                <span className="exam-chip" style={{ background: "#f3e8d2", color: "#6f5438" }}>• {score.total - score.answered} unattempted</span>
   394→                <span className="exam-chip" style={{ background: mcqMeta.bg, color: mcqMeta.color }}>
   395→                  <Timer className="mr-1 inline h-3 w-3" />
   396→                  {fmt(seconds)}
   397→                </span>
   398→              </div>
   399→            </div>
   400→          </div>
   401→
   402→          {/* per-type breakdown */}
   403→          <div className="mt-5 grid gap-3 sm:grid-cols-2">
   404→            {typeKeys.map((t) => {
   405→              const meta = theme.typeMeta[t];
   406→              const d = score.byType[t];
   407→              if (!meta || !d) return null;
   408→              return (
   409→                <div key={t} className="flex items-center justify-between rounded-xl border border-[#e8dcc0] bg-[#fffdf9] px-4 py-3">
   410→                  <span className="text-[.8rem] font-bold" style={{ color: meta.color }}>
   411→                    {meta.label}
   412→                  </span>
   413→                  <span className="text-[.8rem] font-extrabold text-[#2b1d10]">
   414→                    {d.c} / {d.t}
   415→                  </span>
   416→                </div>
   417→              );
   418→            })}
   419→          </div>
   420→
   421→          {/* review the misses */}
   422→          {score.correct < score.total ? (
   423→            <div className="mt-6">
   424→              <p className="exam-eyebrow">
   425→                <Printer className="h-3.5 w-3.5" />
   426→                Review &amp; rewrite these by hand
   427→              </p>
   428→              <div className="mt-3 grid gap-3">
   429→                {qs.map((qq) => {
   430→                  const s = states[qq.id]?.status ?? "unattempted";
   431→                  if (s === "correct") return null;
   432→                  const model =
   433→                    qq.type === "mcq"
   434→                      ? (qq.options?.[qq.answer ?? 0] ?? "")
   435→                      : qq.answerText ?? qq.solution ?? "";
   436→                  return (
   437→                    <div key={qq.id} className="rounded-xl border border-[#e8dcc0] bg-[#fffdf9] p-4">
   438→                      <p className="text-[.88rem] font-bold leading-snug text-[#2b1d10]">{qq.q}</p>
   439→                      {qq.code ? <pre className="exam-code mt-2">{qq.code}</pre> : null}
   440→                      <p className="mt-2 text-[.82rem] leading-relaxed" style={{ color: s === "wrong" ? "#8c2f2f" : "#6f5438" }}>
   441→                        <strong>{s === "wrong" ? "Missed —" : "Unattempted —"}</strong>{" "}
   442→                        <span className="whitespace-pre-wrap font-mono text-[.78rem] text-[#2f2013]">{model}</span>
   443→                      </p>
   444→                      {qq.explain ? <p className="mt-1 text-[.78rem] leading-relaxed text-[#6f5438]">{qq.explain}</p> : null}
   445→                    </div>
   446→                  );
   447→                })}
   448→              </div>
   449→            </div>
   450→          ) : null}
   451→
   452→          <div className="mt-7 flex flex-wrap justify-center gap-3">
   453→            <button type="button" onClick={retake} className="exam-btn exam-btn-primary">
   454→              <RotateCcw className="h-4 w-4" />
   455→              Retake this test
   456→            </button>
   457→            <button type="button" onClick={() => setFinished(false)} className="exam-btn">
   458→              <ListChecks className="h-4 w-4" />
   459→              Back to questions
   460→            </button>
   461→          </div>
   462→        </motion.div>
   463→      </ExamShell>
   464→    );
   465→  }
   466→
   467→  /* ---------------- question view ---------------- */
   468→  const meta = theme.typeMeta[q.type] ?? theme.typeMeta.mcq ?? Object.values(theme.typeMeta)[0];
   469→  const mcqLocked = st.status === "correct" || st.status === "wrong";
   470→
   471→  return (
   472→    <ExamShell
   473→      theme={theme}
   474→      bank={bank}
   475→      score={score}
   476→      seconds={seconds}
   477→      onClose={closeWindow}
   478→      restoredCount={restored}
   479→      onExit={() => finish()}
   480→      scrollRef={bodyRef}
   481→    >
   482→      <motion.div key={q.id} initial={{ opacity: 0, x: 24 }} animate={{ opacity: 1, x: 0 }} transition={{ duration: 0.22, ease: "easeOut" }}>
   483→        {/* type chip + counter */}
   484→        <div className="flex flex-wrap items-center gap-2">
   485→          <span className="exam-type-chip" style={{ color: meta.color, background: meta.bg }}>
   486→            {meta.label}
   487→          </span>
   488→          <span className="text-[.72rem] font-extrabold uppercase tracking-[.16em] text-[#8a6d4f]">
   489→            Question {idx + 1} of {qs.length}
   490→          </span>
   491→        </div>
   492→
   493→        {/* question */}
   494→        <h3 className="font-display mt-3 text-xl font-bold leading-snug text-[#2b1d10] md:text-[1.35rem]">{q.q}</h3>
   495→
   496→        {q.code ? <pre className="exam-code mt-4">{q.code}</pre> : null}
   497→
   498→        {/* answer area */}
   499→        {q.type === "mcq" ? (
   500→          <div className="mt-4 grid gap-2">
   501→            {q.options?.map((opt, i) => {
   502→              const isAnswer = i === q.answer;
   503→              const chosen = st.chosen === i;
   504→              let cls = "exam-opt";
   505→              if (mcqLocked && isAnswer) cls += " exam-opt-correct";
   506→              else if (mcqLocked && chosen) cls += " exam-opt-wrong";
   507→              return (
   508→                <button key={i} type="button" disabled={mcqLocked} onClick={() => chooseMCQ(i)} className={cls}>
   509→                  <span className="exam-opt-letter">{String.fromCharCode(65 + i)}</span>
   510→                  <span className="font-mono text-[.86rem]">{opt}</span>
   511→                  {mcqLocked && isAnswer ? <Check className="ml-auto h-4 w-4" /> : null}
   512→                </button>
   513→              );
   514→            })}
   515→          </div>
   516→        ) : (
   517→          <div className="mt-4">
   518→            <label htmlFor={`ta-${q.id}`} className="text-[.74rem] font-bold uppercase tracking-[.14em] text-[#8a6d4f]">
   519→              {theme.paperLabel}
   520→            </label>
   521→            <textarea
   522→              id={`ta-${q.id}`}
   523→              value={st.typed ?? ""}
   524→              onChange={(e) => patch(q.id, { typed: e.target.value })}
   525→              spellCheck={false}
   526→              rows={q.type === "short" ? 4 : 6}
   527→              placeholder={q.type === "output" ? theme.outputPlaceholder : q.type === "code" ? "# write your program here…" : "your answer…"}
   528→              className="exam-ta"
   529→            />
   530→            {!st.revealed ? (
   531→              <button type="button" onClick={() => patch(q.id, { revealed: true })} className="exam-btn mt-2">
   532→                <Eye className="h-4 w-4" />
   533→                Reveal the model answer
   534→              </button>
   535→            ) : (
   536→              <motion.div
   537→                initial={{ opacity: 0, y: 10 }}
   538→                animate={{ opacity: 1, y: 0 }}
   539→                className="mt-3"
   540→                role="status"
   541→              >
   542→                <div className="exam-answer">
   543→                  <p className="exam-eyebrow">
   544→                    <GraduationCap className="h-3.5 w-3.5" />
   545→                    Model answer
   546→                  </p>
   547→                  {q.solution ? <pre className="exam-code mt-2">{q.solution}</pre> : null}
   548→                  {q.answerText ? (
   549→                    <p className={`mt-2 whitespace-pre-wrap font-mono text-[.82rem] leading-relaxed text-[#2f2013] ${q.solution ? "text-[.78rem]" : ""}`}>
   550→                      {q.answerText}
   551→                    </p>
   552→                  ) : null}
   553→                  {q.explain ? <p className="mt-2 text-[.82rem] leading-relaxed text-[#6f5438]">{q.explain}</p> : null}
   554→                </div>
   555→                {st.status !== "correct" && st.status !== "wrong" ? (
   556→                  <div className="mt-3 flex flex-wrap items-center gap-2">
   557→                    <span className="text-[.74rem] font-bold uppercase tracking-[.12em] text-[#8a6d4f]">Self-check:</span>
   558→                    <button type="button" onClick={() => selfCheck(true)} className="exam-btn" style={{ color: "#44621d" }}>
   559→                      <Check className="h-4 w-4" /> I got it
   560→                    </button>
   561→                    <button type="button" onClick={() => selfCheck(false)} className="exam-btn" style={{ color: "#8c2f2f" }}>
   562→                      <X className="h-4 w-4" /> Not yet
   563→                    </button>
   564→                  </div>
   565→                ) : null}
   566→              </motion.div>
   567→            )}
   568→          </div>
   569→        )}
   570→
   571→        {/* MCQ explanation */}
   572→        {q.type === "mcq" && mcqLocked && q.explain ? (
   573→          <motion.div initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} className="exam-answer mt-3" role="status">
   574→            <p className="text-[.88rem] font-bold" style={{ color: st.status === "correct" ? "#44621d" : "#8c2f2f" }}>
   575→              {st.status === "correct" ? "Correct." : "Not quite."}
   576→            </p>
   577→            <p className="mt-1 text-[.85rem] leading-relaxed text-[#2f2013]">{q.explain}</p>
   578→          </motion.div>
   579→        ) : null}
   580→
   581→        {/* navigator */}
   582→        <div className="exam-nav">
   583→          <div className="flex flex-wrap items-center justify-between gap-x-4 gap-y-1">
   584→            <p className="exam-eyebrow">
   585→              <ListChecks className="h-3.5 w-3.5" />
   586→              Jump to a question
   587→            </p>
   588→            <p className="hidden items-center gap-2.5 text-[.58rem] font-bold uppercase tracking-[.1em] text-[#8a6d4f] sm:flex" aria-hidden>
   589→              <span>
   590→                <kbd className="exam-kbd">1–4</kbd> answer
   591→              </span>
   592→              <span>
   593→                <kbd className="exam-kbd">↵</kbd> next
   594→              </span>
   595→              <span>
   596→                <kbd className="exam-kbd">←→</kbd> move
   597→              </span>
   598→            </p>
   599→          </div>
   600→          <div className="mt-2 flex flex-wrap gap-1.5">
   601→            {qs.map((qq, i) => {
   602→              const s = states[qq.id]?.status ?? "unattempted";
   603→              const cls =
   604→                i === idx
   605→                  ? "exam-dot exam-dot-current"
   606→                  : s === "correct"
   607→                  ? "exam-dot exam-dot-ok"
   608→                  : s === "wrong"
   609→                  ? "exam-dot exam-dot-bad"
   610→                  : s === "revealed"
   611→                  ? "exam-dot exam-dot-rev"
   612→                  : "exam-dot";
   613→              return (
   614→                <button key={qq.id} type="button" onClick={() => go(i)} className={cls} aria-label={`Go to question ${i + 1}`}>
   615→                  {i + 1}
   616→                </button>
   617→              );
   618→            })}
   619→          </div>
   620→        </div>
   621→
   622→        {/* prev / next */}
   623→        <div className="mt-6 flex items-center justify-between pb-2">
   624→          <button type="button" onClick={() => go(Math.max(0, idx - 1))} disabled={idx === 0} className="exam-btn">
   625→            <ChevronLeft className="h-4 w-4" />
   626→            Prev
   627→          </button>
   628→          {idx < qs.length - 1 ? (
   629→            <button type="button" onClick={() => go(idx + 1)} className="exam-btn exam-btn-primary">
   630→              Next
   631→              <ChevronRight className="h-4 w-4" />
   632→            </button>
   633→          ) : (
   634→            <button type="button" onClick={() => finish()} className="exam-btn exam-btn-primary">
   635→              <Trophy className="h-4 w-4" />
   636→              Finish &amp; see score
   637→            </button>
   638→          )}
   639→        </div>
   640→      </motion.div>
   641→    </ExamShell>
   642→  );
   643→}
