   183→   183→/* ============================================================
   184→   184→   Python syntax highlighter (tiny, warm palette)
   185→   185→   ============================================================ */
   186→   186→const PY_KEYWORDS = new Set([
   187→   187→  "False","None","True","and","as","assert","break","class","continue","def",
   188→   188→  "del","elif","else","except","finally","for","from","global","if","import",
   189→   189→  "in","is","lambda","nonlocal","not","or","pass","raise","return","try",
   190→   190→  "while","with","yield",
   191→   191→]);
   192→   192→const PY_BUILTINS = new Set([
   193→   193→  "print","len","range","list","sorted","min","max","sum","int","str","float",
   194→   194→  "input","append","insert","pop","remove","sort","reverse","title","upper",
   195→   195→  "lower","strip","lstrip","rstrip","keys","values","items","get","set",
   196→   196→  "dict","tuple","enumerate","zip","abs","round","type","clear","copy","count","index",
   197→   197→]);
   198→   198→
   199→   199→type Tok = { text: string; cls: string };
   200→   200→
   201→   201→function tokenizePython(code: string): Tok[][] {
   202→   202→  const pattern =
   203→   203→    /(#.*$)|("""[\s\S]*?"""|'''[\s\S]*?'''|"(?:[^"\\]|\\.)*"|'(?:[^'\\]|\\.)*')|\b(\d[\d_]*\.?\d*)\b|([A-Za-z_][A-Za-z0-9_]*)|(\s+)|([^\sA-Za-z0-9_]+)/gm;
   204→   204→  const lines: Tok[][] = [[]];
   205→   205→  let m: RegExpExecArray | null;
   206→   206→  while ((m = pattern.exec(code)) !== null) {
   207→   207→    const [full, comment, str, num, word, ws, sym] = m;
   208→   208→    if (comment !== undefined) push(comment, "tk-c");
   209→   209→    else if (str !== undefined) push(str, "tk-s");
   210→   210→    else if (num !== undefined) push(num, "tk-n");
   211→   211→    else if (word !== undefined) {
   212→   212→      if (PY_KEYWORDS.has(word)) push(word, "tk-k");
   213→   213→      else if (PY_BUILTINS.has(word)) push(word, "tk-b");
   214→   214→      else push(word, "");
   215→   215→    } else if (ws !== undefined) push(ws, "ws");
   216→   216→    else if (sym !== undefined) push(sym, "tk-d");
   217→   217→  }
   218→   218→  function push(text: string, cls: string) {
   219→   219→    const parts = text.split("\n");
   220→   220→    parts.forEach((p, i) => {
   221→   221→      if (i > 0) lines.push([]);
   222→   222→      if (p) lines[lines.length - 1].push({ text: p, cls: cls === "ws" ? "" : cls });
   223→   223→    });
   224→   224→  }
   225→   225→  return lines;
   226→   226→}

   228→   228→function HighlightedCode({ code }: { code: string }) {
   229→   229→  const lines = tokenizePython(code);
   230→   230→  return (
   231→   231→    <pre className="code-body">
   232→   232→      {lines.map((toks, i) => (
   233→   233→        <div key={i} className="flex">
   234→   234→          <span className="mr-4 w-6 shrink-0 select-none text-right text-[#6d5236]">
   235→   235→            {i + 1}
   236→   236→          </span>
   237→   237→          <span className="flex-1">
   238→   238→            {toks.length === 0 ? " " : null}
   239→   239→            {toks.map((t, j) =>
   240→   240→              t.cls ? (
   241→   241→                <span key={j} className={t.cls}>
   242→   242→                  {t.text}
   243→   243→                </span>
   244→   244→              ) : (
   245→   245→                <React.Fragment key={j}>{t.text}</React.Fragment>
   246→   246→              )
   247→   247→            )}
   248→   248→          </span>
   249→   249→        </div>
   250→   250→      ))}

   255→   255→/* ============================================================
   256→   256→   CodeBlock — syntax-highlighted snippet with an optional LIVE
   257→   257→   Python runtime: ▶ Run executes the code in the browser (Pyodide),
   258→   258→   ✎ lets the learner edit the code, ⟲ restores the original.
   259→   259→   `pre`   → hidden Python executed before the visible code
   260→   260→   `files` → virtual .py modules written before the run
   261→   261→   ============================================================ */
   262→   262→type RunPhase = "idle" | "running" | "done" | "error";
   263→   263→
   264→   264→export function CodeBlock({
   265→   265→  title = "main.py",
   266→   266→  code,
   267→   267→  output,
   268→   268→  note,
   269→   269→  className = "",
   270→   270→  runnable = true,
   271→   271→  pre,
   272→   272→  files,
   273→   273→  dark = false,
   274→   274→}: {
   275→   275→  title?: string;
   276→   276→  code: string;
   277→   277→  output?: string;
   278→   278→  note?: React.ReactNode;
   279→   279→  className?: string;
   280→   280→  runnable?: boolean;
   281→   281→  pre?: string;
   282→   282→  files?: Record<string, string>;
   283→   283→  dark?: boolean;
   284→   284→}) {
   285→   285→  const [current, setCurrent] = useState(code);
   286→   286→  const [editing, setEditing] = useState(false);
   287→   287→  const [phase, setPhase] = useState<RunPhase>("idle");
   288→   288→  const [bootMsg, setBootMsg] = useState("");
   289→   289→  const [out, setOut] = useState("");
   290→   290→  const [err, setErr] = useState("");
   291→   291→  const [elapsed, setElapsed] = useState(0);
   292→   292→  const [copied, setCopied] = useState(false);
   293→   293→  const termRef = useRef<HTMLPreElement>(null);
   294→   294→  const dirty = current !== code;
   295→   295→  const live = phase !== "idle";
   296→   296→  const busy = phase === "running";
   297→   297→
   298→   298→  useEffect(() => {
   299→   299→    if (termRef.current) termRef.current.scrollTop = termRef.current.scrollHeight;
   300→   300→  }, [out, err]);
   301→   301→
   302→   302→  async function run() {
   303→   303→    if (busy) return;
   304→   304→    setEditing(false);
   305→   305→    setPhase("running");
   306→   306→    setOut("");
   307→   307→    setErr("");
   308→   308→    setBootMsg("");
   309→   309→    setElapsed(0);
   310→   310→    const t0 = performance.now();
   311→   311→    let py: PyodideLike | null = null;
   312→   312→    try {
   313→   313→      py = await getPyodide((m) => setBootMsg(m));
   314→   314→      const pyr = py;
   315→   315→      setBootMsg("");
   316→   316→      let lastOut = "";
   317→   317→      // capped push: a print flood can't grow the buffer unbounded
   318→   318→      const push = makeCappedPush((full) => {
   319→   319→        lastOut = full;
   320→   320→        setOut(full);
   321→   321→      });
   322→   322→      py.setStdout({ batched: push });
   323→   323→      py.setStderr({ batched: push });
   324→   324→      py.setStdin({
   325→   325→        stdin: () => {
   326→   326→          try {
   327→   327→            return window.prompt("Python input() is asking for a value — cancel = stop the program:");
   328→   328→          } catch {
   329→   329→            return null; // prompt blocked (e.g. sandboxed iframe) → treat as EOF
   330→   330→          }
   331→   331→        },
   332→   332→      });
   333→   333→      // Fresh module cache + virtual files for import demos.
   334→   334→      await py.runPythonAsync(PURGE_USER_MODULES);
   335→   335→      if (files) {
   336→   336→        for (const [name, content] of Object.entries(files)) py.FS.writeFile(name, content);
   337→   337→      }
   338→   338→      const wf = current.match(/^%%writefile\s+([^\s#]+)/);
   339→   339→      if (wf) {
   340→   340→        // Notebook magic: save the cell as a real file inside the runtime.
   341→   341→        const name = wf[1];
   342→   342→        const body = current.slice(current.indexOf("\n") + 1).replace(/^\n+/, "");
   343→   343→        py.FS.writeFile(name, body);
   344→   344→        push(`✓ saved ${name} (${body.split("\n").length} lines) — import blocks can now use it`);
   345→   345→      } else {
   346→   346→        const ns = py.globals.get("dict")() as { destroy?: () => void };
   347→   347→        try {
   348→   348→          // step-guard: cooperative runaway-loop killer (see lib/pyguard.ts)
   349→   349→          await withStepGuard(pyr, async () => {
   350→   350→            if (pre) await pyr.runPythonAsync(pre, { globals: ns });
   351→   351→            await pyr.runPythonAsync(current, { globals: ns });
   352→   352→          });
   353→   353→        } finally {
   354→   354→          ns.destroy?.();
   355→   355→        }
   356→   356→      }
   357→   357→      setElapsed((performance.now() - t0) / 1000);
   358→   358→      setPhase("done");
   359→   359→    } catch (e) {
   360→   360→      const msg = e instanceof Error ? e.message : String(e);
   361→   361→      setErr(cleanTraceback(msg) || msg);
   362→   362→      setElapsed((performance.now() - t0) / 1000);
   363→   363→      setPhase("error");
   364→   364→    } finally {
   365→   365→      if (py) {
   366→   366→        try {
   367→   367→          py.setStdout();
   368→   368→          py.setStderr();
   369→   369→          py.setStdin();
   370→   370→        } catch {
   371→   371→          /* already reset */
   372→   372→        }
   373→   373→      }
   374→   374→    }
   375→   375→  }
   376→   376→
   377→   377→  function copy() {
   378→   378→    try {
   379→   379→      navigator.clipboard.writeText(current).then(
   380→   380→        () => {
   381→   381→          setCopied(true);
   382→   382→          window.setTimeout(() => setCopied(false), 1600);
   383→   383→        },
   384→   384→        () => undefined
   385→   385→      );
   386→   386→    } catch {
   387→   387→      /* clipboard unavailable */
   388→   388→    }
   389→   389→  }
   390→   390→
   391→   391→  return (
   392→   392→    <div className={`code-shell ${className}`}>
   393→   393→      <div className="code-bar">
   394→   394→        <span className="code-dot" style={{ background: "#e4674a" }} />
   395→   395→        <span className="code-dot" style={{ background: "#e8a13c" }} />
   396→   396→        <span className="code-dot" style={{ background: "#8fae5d" }} />
   397→   397→        <span className="code-title ml-2">{title}</span>
   398→   398→        {runnable ? (
   399→   399→          <div className="ml-auto flex items-center gap-1">
   400→   400→            <button
   401→   401→              type="button"
   402→   402→              onClick={copy}
   403→   403→              className="code-icon-btn"
   404→   404→              title="Copy code"
   405→   405→              aria-label="Copy code"
   406→   406→            >
   407→   407→              {copied ? <Check className="h-3.5 w-3.5" style={{ color: "#8fae5d" }} /> : <Copy className="h-3.5 w-3.5" />}
   408→   408→            </button>
   409→   409→            <button
   410→   410→              type="button"
   411→   411→              onClick={() => {
   412→   412→                setCurrent(code);
   413→   413→                setEditing((v) => !v);
   414→   414→              }}
   415→   415→              className={`code-icon-btn ${editing ? "code-icon-btn-on" : ""}`}
   416→   416→              title={dirty ? "Editing — click ⟲ to restore the original" : editing ? "Done editing" : "Edit the code, then run it"}
   417→   417→              aria-label="Edit code"
   418→   418→            >
   419→   419→              <Pencil className="h-3.5 w-3.5" />
   420→   420→            </button>
   421→   421→            {dirty ? (
   422→   422→              <button
   423→   423→                type="button"
   424→   424→                onClick={() => setCurrent(code)}
   425→   425→                className="code-icon-btn"
   426→   426→                title="Restore the original code"
   427→   427→                aria-label="Restore original code"
   428→   428→              >
   429→   429→                <RotateCcw className="h-3.5 w-3.5" />
   430→   430→              </button>
   431→   431→            ) : null}
   432→   432→            <button
   433→   433→              type="button"
   434→   434→              onClick={run}
   435→   435→              disabled={busy}
   436→   436→              className="run-btn"
   437→   437→              title="Run this code live — real Python 3.12 in your browser"
   438→   438→            >
   439→   439→              {busy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Play className="h-3.5 w-3.5" />}
   440→   440→              {busy ? "running" : "Run"}
   441→   441→            </button>
   442→   442→          </div>
   443→   443→        ) : (
   444→   444→          <span className="readonly-chip" title="Illustrative snippet — not executable">
   445→   445→            <EyeOff className="h-3 w-3" />
   446→   446→            read-only
   447→   447→          </span>
   448→   448→        )}
   449→   449→      </div>
   450→   450→
   451→   451→      {editing ? (
   452→   452→        <textarea
   453→   453→          value={current}
   454→   454→          onChange={(e) => setCurrent(e.target.value)}
   455→   455→          spellCheck={false}
   456→   456→          aria-label="Editable code"

   458→   458→          rows={Math.max(6, current.split("\n").length + 1)}

   460→   460→      ) : (
   461→   461→        <HighlightedCode code={current} />
   462→   462→      )}
   463→   463→
   464→   464→      <AnimatePresence>
   465→   465→        {live ? (

   472→   472→            style={{ overflow: "hidden" }}
   473→   473→          >
   474→   474→            <div className="live-term">
   475→   475→              <div className="live-term-bar">
   476→   476→                <Terminal className="h-3 w-3" />

   486→   486→                  </span>
   487→   487→                ) : null}
   488→   488→                {out ? out : !bootMsg && busy ? "\u2026" : ""}
   489→   489→                {!bootMsg && busy ? <span className="term-caret" aria-hidden /> : null}
   490→   490→                {err ? <span className="tb-err">{err}\n</span> : null}
   491→   491→              </pre>

   519→   519→        </div>
   520→   520→      ) : null}
   521→   521→    </div>
   522→   522→  );

   529→   529→  const [show, setShow] = useState(false);
