   145→/* ============================================================
   146→   useSeenDisplay — the read-checkmark data for every menu.
   147→   Marking is per-lecture; display spans the whole shelf.
   148→   ============================================================ */
   149→function useSeenDisplay(lecture: MktgLecture): ReadonlySet<string> {
   150→  const ids = useMemo(() => lecture.chapters.map((c) => c.id), [lecture]);
   151→  const live = useSectionProgress("mktg", ids);
   152→  const persisted = useMemo(() => {
   153→    try {
   154→      return new Set(Object.keys(loadProgress().mktg ?? {}));
   155→    } catch {
   156→      return new Set<string>();
   157→    }
   158→  }, []);
   159→  return useMemo(() => new Set<string>([...persisted, ...live]), [persisted, live]);
   160→}

   258→          >
   259→            <Map className="h-3.5 w-3.5" />
   260→            <span className="hidden sm:inline">Atlas</span>
   261→          </button>
   262→
   263→          {/* brand */}
   264→          <a
   265→            href="#top"
   266→            className="flex shrink-0 items-center gap-2 rounded-full border border-white/70 bg-white/60 py-1.5 pl-2 pr-3.5 shadow-[0_4px_16px_-8px_rgba(15,50,48,.35)] backdrop-blur-md transition-colors duration-200 hover:bg-white/80"
   267→          >
   268→            <span
   269→              className="flex h-6 w-6 items-center justify-center rounded-full text-white"
   270→              style={{ background: `linear-gradient(135deg,${TEAL_BRIGHT},${TEAL})`, boxShadow: "0 4px 10px -3px rgba(15,118,110,.55)" }}

   950→export function MktgApp({ onExitAtlas }: { onExitAtlas: () => void }) {
   951→  const [lectureId, setLectureId] = useState<string>(MKTG_DEFAULT_LECTURE);
   952→  const [examGroup, setExamGroup] = useState<MktgExamGroup | null>(null);
   953→  const [hubOpen, setHubOpen] = useState(false);
   954→  const lectureIdRef = useRef(lectureId);
   955→  useEffect(() => {
   956→    lectureIdRef.current = lectureId;
   957→  }, [lectureId]);
   958→
   959→  const lecture = lectureById(lectureId) ?? MKTG_LECTURES[0];
   960→  const seen = useSeenDisplay(lecture);
   961→
   962→  const selectLecture = useCallback((id: string) => {
   963→    setLectureId(id);
   964→    requestAnimationFrame(() =>
   965→      requestAnimationFrame(() => {
   966→        const el = document.getElementById("notes-top");
   967→        if (!el) return;
   968→        const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   969→        el.scrollIntoView({ behavior: reduce ? "auto" : "smooth", block: "start" });
   970→      })
   971→    );
   972→  }, []);
   973→
   974→  const goToChapter = useCallback((chapterId: string) => {
   975→    const target = lectureOfChapter(chapterId);
   976→    if (!target) return;
   977→    const scrollWhenReady = () => {
   978→      let frames = 0;
   979→      const tick = () => {
   980→        const el = document.getElementById(chapterId);
   981→        if (el) {
   982→          jumpTo(el);
   983→          return;
   984→        }
   985→        if (++frames < 240) requestAnimationFrame(tick);
   986→      };
   987→      requestAnimationFrame(tick);
   988→    };
   989→    if (lectureIdRef.current === target.id) {
   990→      scrollWhenReady();
   991→      return;
   992→    }
   993→    setLectureId(target.id);
   994→    requestAnimationFrame(() => requestAnimationFrame(scrollWhenReady));
   995→  }, []);
   996→
   997→  useOpenSection("mktg", goToChapter);
   998→
   999→  const openExam = useCallback((g: MktgExamGroup) => setExamGroup(g), []);
  1000→  const openHub = useCallback(() => setHubOpen(true), []);
  1001→
  1002→  return (
  1003→    <div id="top" className="page-wrap mktg-canvas">
  1004→      <TopNav onOpenExam={openExam} onExitAtlas={onExitAtlas} onJumpChapter={goToChapter} seen={seen} />
  1005→      <main className="pt-14">
  1006→        <PartBoundary label="Hero">
  1007→          <MktgHero
  1008→            activeId={lectureId}
  1009→            onSelect={selectLecture}
  1010→            onPractice={openHub}
  1011→            seen={seen}
  1012→            totalQuestions={MKTG_TOTAL_QUESTIONS}
  1013→          />
  1014→        </PartBoundary>
  1015→
  1016→        {/* the notes start here */}
  1017→        <div id="notes-top" aria-hidden className="relative scroll-mt-16" />
  1018→
  1019→        <PartBoundary key={lecture.id} label={`Lecture ${lecture.code} — ${lecture.title}`}>
  1020→          <motion.div
  1021→            initial={{ opacity: 0, y: 18 }}
  1022→            animate={{ opacity: 1, y: 0 }}
  1023→            transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
  1024→          >
  1025→            <MktgLectureHead lecture={lecture} />
  1026→            {lecture.chapters.length > 0 ? (
  1027→              lecture.chapters.map((c) => (
  1028→                <PartBoundary key={c.id} label={c.label}>
  1029→                  {SECTION_COMPONENTS[c.id]}
  1030→                </PartBoundary>
  1031→              ))
  1032→            ) : (
  1033→              <div className="mx-auto max-w-3xl px-4 py-16 sm:px-6">
  1034→                <div className="card-warm p-8 text-center">
  1035→                  <span className="chip mx-auto">
  1036→                    <BookOpen className="h-3.5 w-3.5" /> Slot reserved
  1037→                  </span>
  1038→                  <h2 className="mt-4 text-2xl font-extrabold text-[#1f2b2b]">
  1039→                    Lecture {lecture.n} hasn&rsquo;t been summarised yet
  1040→                  </h2>
  1041→                  <p className="mx-auto mt-3 max-w-md text-[.92rem] leading-relaxed text-[#5f7472]">
  1042→                    This lecture&rsquo;s notes haven&rsquo;t been uploaded, so the shelf keeps the
  1043→                    slot open. The moment its summary arrives, the chapters fill in right here.
  1044→                  </p>
  1045→                </div>
  1046→              </div>
  1047→            )}
  1048→            <LecturePager current={lecture} onJump={selectLecture} />
  1049→          </motion.div>
  1050→        </PartBoundary>
  1051→      </main>
  1052→      <MktgFooter />
  1053→      <PracticeHub open={hubOpen} onClose={() => setHubOpen(false)} onOpenExam={openExam} />
  1054→      {examGroup ? <MktgExamWindow group={examGroup} onClose={() => setExamGroup(null)} /> : null}
  1055→    </div>
  1056→  );
