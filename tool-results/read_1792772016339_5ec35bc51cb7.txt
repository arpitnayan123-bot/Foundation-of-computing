    78→    78→function Stat({
    79→    79→  value,
    80→    80→  render,
    81→    81→  label,
    82→    82→  delay = 0,
    83→    83→}: {
    84→    84→  value: number;
    85→    85→  render?: (n: number) => React.ReactNode;
    86→    86→  label: string;
    87→    87→  delay?: number;
    88→    88→}) {
    89→    89→  const ref = useRef<HTMLDivElement>(null);
    90→    90→  const inView = useInView(ref, { once: true, margin: "-40px" });
    91→    91→  const [n, setN] = useState(0);
    92→    92→
    93→    93→  useEffect(() => {
    94→    94→    if (!inView) return;
    95→    95→    const controls = animate(0, value, {
    96→    96→      duration: 1.5,
    97→    97→      delay,
    98→    98→      ease: EASE,
    99→    99→      onUpdate: (v) => setN(Math.round(v)),
   100→   100→    });
   101→   101→    return () => controls.stop();
   102→   102→  }, [inView, value, delay]);
   103→   103→
   104→   104→  return (
   105→   105→    <div ref={ref} className="flex flex-col items-center gap-1">
   106→   106→      <span className="font-display text-3xl font-black tracking-tight text-[#f7ecdd] md:text-4xl">
   107→   107→        {render ? render(n) : n}
   108→   108→      </span>
   109→   109→      <span className="text-[.62rem] font-extrabold uppercase tracking-[.22em] text-[#8d7b63]">{label}</span>
   110→   110→    </div>
   111→   111→  );
   112→   112→}
